\section{Concepts and Definitions}
\begin{definition}{State}
    A quantum system is described at each instant by a \underline{state} $\state$ which belongs to a complex vector space $V$. Any non-zero element of this space $V$ is a possible state.
\end{definition}
Since these are in a vector space, for $\state, \ket{\phi} \in V$, $\alpha \state + \beta \ket{\phi} \in V$ for any complex constants $\alpha, \beta$.

Physically, this is a superposition principle leading to wave-like behaviour, but these states are not wavefunctions. There are also dual or conjugate states $\bra{\psi}$ which belong to the dual space $V^\dagger$. By definition, states and duals can be combined to give a complex number:
\begin{equation*}
    \bra{\phi} \text{ combined with }\state \mapsto \inn{\phi}{\psi}
\end{equation*}
The spaces $V$ and $V^\dagger$ are equipped with an inner product which can be described as a one-to-one correspondence between the spaces:
\begin{align*}
    V &\mapsto V^\dagger \\
    \state &\mapsto \bra{\psi} = \left(\state\right)^\dagger
\end{align*}
Then the inner product requires the first state to be mapped to its dual, then applied to the second:
\begin{equation*}
    \ket{\phi},~~\state \mapsto \inn\phi\psi = \left(\ket{\phi}\right)^T \state \in \C
\end{equation*}

\begin{remarks}
    \item Knowing $\inn\phi\psi$ for all $\bra\phi$ determines $\ket\psi$. \label{remInnDefinesState}
    \item The physical content of any state is unchanged by $\state \mapsto \alpha \state$ for $\alpha \in \C \backslash \{0\}$.
    \item We will usually normalise the state to make $\norm\state = 1$, but this still allows for relative phases $\state \mapsto e^{i\theta}\state$, which can be important for superposition.
    \item The space $V$ is complete. We will assume that appropriate sequences and series converge.
    \item The space of states is also known as a Hilbert space $\hilb$.
\end{remarks}
\begin{definition}{Operator}
    An \underline{operator} $Q$ is a linear map on states:
    \begin{align*}
        Q : V &\mapsto V\\
        \state &\mapsto Q\state
    \end{align*}
    which must obey linearity.
\end{definition}
The same operator can also be thought of acting to the left on dual states:
\begin{align*}
    Q : V^\dagger &\mapsto V^\dagger\\
    \bra{\psi} &\mapsto \bra{\psi}Q
\end{align*}
where the RHS is defined by remark~\ref{remInnDefinesState} as:
\begin{equation*}
    \left(\ket{\phi}Q\right)\state = \ket{\phi} \left(Q\state\right) = \opbraket{\phi}{Q}{\psi}
\end{equation*}
\begin{remark}
    States and operators $\state, \bra{\phi}, Q$ can be dealt with by rules that mirror those for complex vectors. Identifying these, in some basis, with $\vec{u}, \vec{v}^\dagger, A$ where $A$ is a matrix, we can say:
    \begin{equation*}
        \opbraket{\phi}{Q}{\psi} = \vec{v}^\dagger A \vec{u}
    \end{equation*}
\end{remark}
\begin{definition}{Eigenstate}
    For any operator $Q$, we say that a state $\state \neq 0$ is an \underline{eigenstate} with eigenvalue $\lambda$ if $Q\state = \lambda\state$. Equivalently, $\bra{\psi} Q^\dagger = \lambda^* \bra{\psi}$.
\end{definition}
For general $Q$, $\lambda$ may be complex.
\begin{definition}{Hermitian conjugate}
    For any operator $Q$, the \underline{Hermitian conjugate} or adjoint $Q^\dagger$ is defined by:
    \begin{equation*}
        \bra{\phi} Q^\dagger = \left(Q\ket{\phi}\right)^\dagger
    \end{equation*}
    for any $\phi$.
\end{definition}
We can equivalently define it as $\opbraket{\phi}{Q^\dagger}{\psi} = \opbraket{\psi}{Q}{\phi}^*$.
\begin{propositions}{
        For operators $A, B$ and complex numbers $\alpha, \beta$,
        \label{propsHermConjProps}
    }
    \item $(\alpha A + \beta B)^\dagger = \alpha^* A^\dagger + \beta^* B^\dagger$.
    \item $(AB)^\dagger = B^\dagger A^\dagger$.
    \item $(A^\dagger)^\dagger = A$.
\end{propositions}
\begin{definition}{Hermitian}
    An operator $Q$ is \underline{Hermitian} or \underline{self-adjoint} if $Q^\dagger = Q$. Such operators are called \underline{observables} because they correspond to physical quantities that can be measured.
\end{definition}
To define a quantum system, we must specify the \underline{Hamiltonian}, $H$, whose eigenvalues are the permissible energy levels of the system.

\begin{propositions}{
        Let $Q$ be a Hermitian operator
        \label{propsHermitian}
    }
    \item All eigenvalues of $Q$ are real.
    \item All eigenstates of $Q$ are orthogonal.
    \item Any state can be written as a linear combination (which may be an infinite sum) of eigenstates.
\end{propositions}
\begin{proof}
    The first two can be seen in IB Linear Algebra. The third we will assume to be true for the purposes of this course.
\end{proof}
\section{The Quantum Harmonic Oscillator}
A one-dimensional harmonic oscillator of mass $m$ and frequency $\omega$ has Hamiltonian:
\begin{equation*}
    H = \hat{p}^2 + \frac12 m\omega^2\hat{x}^2 
\end{equation*}
\subsection{Construction and Annihilation Operators}
We seek eigenstates and eigenvalues of $H$. We will do this by introducing new operators:
\begin{align*}
    a &= \left(\frac{m\omega}{2\hbar}\right)^\frac12 \left(\hat{x} + \frac{i\hat{p}}{m\omega}\right) \\
    a^\dagger &= \left(\frac{m\omega}{2\hbar}\right)^\frac12 \left(\hat{x} - \frac{i\hat{p}}{m\omega}\right)
\end{align*}
We do this in order to reduce $H$ into something involving $a^\dagger a$. Note that the constants in $a$ are chosen such that it is dimensionless. We can now write:
\begin{align*}
    \hat{x} &= \left(\frac{\hbar}{2m\omega}\right)^\frac12 (a + a^\dagger) \\
    \hat{p} &= \left(\frac{\hbar m\omega}{2}\right)^\frac12 i(a^\dagger - a)
\end{align*}
We also get that $\cmmt{a}{a^\dagger} = 1$. The products of $a$ with its conjugate are:
\begin{align*}
    aa^\dagger &= \frac{1}{\hbar \omega} H + \frac12 \\
    a^\dagger a &= \frac{1}{\hbar \omega} H - \frac12
\end{align*}
so now we can write the Hamiltonian:
\begin{equation*}
    H = \hbar \omega \left(N + \frac12\right)
\end{equation*}
where $N$ is the \underline{number operator}, $a^\dagger a$. This form shows that the eigenstates of $N$ are exactly the eigenstates of $H$, with eigenvalues adjusted as required.

Now consider an eigenstate of $N$, $\ket{\lambda}$, with eigenvalue $\lambda$. Assume that $\ket{\lambda}$ is normalised. Therefore:
\begin{align*}
    \lambda &= \opbraket{\lambda}{N}{\lambda} \\
    &= \opbraket{\lambda}{a^\dagger a}{\lambda} \\
    &= \left(a\ket{\lambda}\right)^\dagger \left(a\ket\lambda\right) \\
    &= \norm{a\ket\lambda}^2.
\end{align*}
This tells us that the eigenvalues are non-negative. In fact, we have a zero eigenvalue if and only if $a\ket\lambda = 0$.

Now consider:
\begin{align*}
    N\left(a^\dagger \ket\lambda\right) &= \left(\cmmt{N}{a^\dagger} + a^\dagger N\right)\ket\lambda \\
    &= \left(a^\dagger + a^\dagger \lambda\right) \ket\lambda \\
    &= (\lambda + 1) \left(a^\dagger \ket\lambda\right)
\end{align*}
and by similar algebra, $N\left(a \ket\lambda\right) = (\lambda - 1) a\ket\lambda$. Note that we used:
\begin{equation*}
    \cmmt{N}{a^\dagger} = a^\dagger,~~\cmmt{N}{a} = -a~~\text{ which follow from } \cmmt{a}{a^\dagger} = 1.
\end{equation*}
In order to declare these new eigenstates, we must check they are non-zero. To do this we look at the norms:
\begin{align*}
    \norm{a\ket\lambda}^2 &= \lambda~~\text{ as seen before} \\
    \norm{a^\dagger \ket\lambda}^2 &= \opbraket{\lambda}{aa^\dagger}{\lambda} \\
    &= \opbraket\lambda{a^\dagger a + 1}\lambda \\
    &= \lambda + 1
\end{align*}
$a^\dagger$ and $a$ are called creation (or raising) and annihilation (or lowering) operators.

These two operators give us exactly the possible set of eigenvalues. We want to argue that these are all integers so suppose we have a non-integer eigenvalue. In this case we can use the lowering operator to decrease the eigenvalue by one until we end up at a negative eigenstate -- this is a contradiction. We avoid this contradiction in the case of integer eigenvalues because we cannot lower to $0$ (because then $a$ gives the zero state). This means our set of eigenvalues is $\N_0$.

By the transformation above, the oscillator has eigenvalues:
\begin{equation*}
    E_n = \hbar \omega \left(n + \frac12\right),~~n \in \N_0.
\end{equation*}
By considering our norm calculations, we can choose normalised eigenstates $\ket{n}$ related by:
\begin{align*}
    a^\dagger \ket{n} &= \sqrt{n+1}~\ket{n+1} \\
    a\ket{n} &= \sqrt{n}~\ket{n-1}.
\end{align*}
Stating from any state $\ket{n}$, we can generate all others. It is customary to start from the ground state $\ket0$ which is characterised by $a\ket0 = 0$. Then the rest are given by:
\begin{equation*}
    \ket{n} = \frac{1}{\sqrt{n!}} \left(a^\dagger\right)^n\ket0
\end{equation*}
\subsection{Importance of the Oscillator}
\begin{itemize}
    \item A smooth potential $V(x)$ with an equilibrium at $x = x_0$ (so $V
    (x_0) = 0$) can be expanded as:
        \begin{equation*}
            V(x + x_0) = V(x_0) + \frac12 V''(x_0) x^2 + O(x^3).
        \end{equation*}
        Therefore, we can set $V''(x) = m\omega^2$ and for displacements $x$ that are not too large, we expect the system to be well-approximated by the quantum harmonic oscillator.
    \item More complicated systems can be analysed in terms of normal modes: coherent motion in which all degrees of freedom oscillate with some common frequency $\omega$. The normal modes for a classical system with variables $x_N$ are solutions of the form:
        \begin{equation*}
            x_N(t) = Au_Ne^{-i\omega t} + A^* u_n^* e^{i\omega t}
        \end{equation*}
        which is a superposition. In quantum mechanics, we also treat the normal modes as independent oscillators. For each mode we have an annihilation operator $a(\omega)$, the creation operator $a(\omega)^\dagger$, the ground state $\ket0$ satisfying $a\ket0 = 0$, and the other states formed via the creation operator.
        
        We find that these other states have energy and momentum and behave like particles. These are phonons in a crystal lattice, or photos in the EM field. We say that $a$ destroys particles and $a^\dagger$ creates particles. For example, $\ket3$ can be understood as a state with $3$ photons/phonons.
\end{itemize}
\section{Bases and Measurements}
\subsection{Bases of Eigenstates}
We have already seen that for an observable $Q$, we have an orthonormal basis of eigenstates $\{\ket{n}\}$ for the vector space $V$.
\begin{align*}
    Q\ket{n} &= q_n \ket{n} \\
    \inn{m}{n} &= \delta_{mn} \\
    \state &= \sum_n \alpha_n \ket{n},~\text{where }\alpha_n = \inn{n}{\psi}
\end{align*}
for any state $\state$. Also, normalisation is given by:
\begin{equation*}
    \norm{\state}^2 = 1 \iff \sum_n \abs{\alpha_n}^2 = 1.
\end{equation*}
Note that we can write:
\begin{align*}
    1 &= \sum_n \ket{n}\bra{n} \\
    \text{because } \state &= \left(\sum_{n} \ket{n} \bra{n}\right) \state \\
    &= \sum_n \ket{n} \inn{n}{\psi} \\
    &= \sum_n \alpha_n \ket{n}.
\end{align*}
In the above there may be several independent states with the same eigenvalue $\lambda$.
\begin{definition}{Eigenspace}
    The \underline{eigenspace} of an eigenvalue $\lambda$ is the set:
    \begin{equation*}
        V_\lambda = \subsetselect{\state}{Q\state = \lambda \state}
    \end{equation*}
\end{definition}
\begin{definition}{Degeneracy}
    The \underline{degeneracy} of an eigenvalue $\lambda$ is $\dim(V_\lambda)$. If $\dim(V_\lambda) = 1$ then the eigenvalue is \underline{non-degenerate}.
\end{definition}
\subsection{Functions of Operators}
It is straightforward to define $f(Q)$ for some operator $Q$ if $f$ is a polynomial or power series, assuming convergence.

Some important examples:
\begin{itemize}
    \item The time evolution operator, $e^{-iHt/\hbar}$, where $H$ is the Hamiltonian.
    \item The translation operator, $e^{-ia\hat{p} / \hbar}$.
\end{itemize}
For observables $Q$, we can define more general $f$ by how it acts on the basis states of $Q$.

Let $\{\ket{n}\}_{n \in \N}$ be a basis of eigenstates of $Q$, with eigenvalues $q_n$. Then $f$ is uniquely defined by:
\begin{equation*}
    f(Q)\ket{n} = f(q_n)\ket{n}
\end{equation*}
provided $f(q_n)$ makes sense. $f(Q)$ applied to any other state is then found by linear combination.

Additionally, note that:
\begin{equation*}
    Q = \sum_n q_n \ket{n}\bra{n}
\end{equation*}
and so:
\begin{equation*}
    f(Q) = \sum_n f(q_n) \ket{n}\bra{n}.
\end{equation*}

\begin{definition}{Projection operator}
    The \underline{projection operator} for the eigenvalue $\lambda$, $P_\lambda$, can be defined by:
    \begin{equation*}
        P_\lambda \ket{n} =
        \begin{cases}
            \ket{n} & \text{ if } q_n = \lambda \\
            0 & \text{otherwise}
        \end{cases}
    \end{equation*}
\end{definition}
The projection operator can be written:
\begin{equation*}
    P_\lambda = \sum_{n : q_n = \lambda} \ket{n}\bra{n}
\end{equation*}
\subsection{Measurement Axioms}
Consider a measurement of an observable $Q$ when the system is in normalised state $\state$. Then:
\begin{enumerate}
    \item the result of this measurement is an eigenvalue $\lambda$ of $Q$; \label{axMeasureIsEVal}
    \item this result is obtained with probability $p(\lambda) = \norm{P_\lambda \state}^2$; \label{axMeasureProb}
    \item immediately after the measurement, the state is $P_\lambda \state$ (up to a constant of normalisation). \label{axStateCollapse}
\end{enumerate}
In the case with non-degenerate $\lambda$, and eigenstate $\ket{n}$, we have:
\begin{itemize}
    \item $p(\lambda) = \abs{\inn{n}{\phi}}^2$
    \item the resulting state is $\ket{n}$.
\end{itemize}
\begin{example} \label{exOrthoStates}
    Consider a system with space $V$ and orthonormal basis states $\ket1, \ket2, \ket3$. Let these be eigenvalues of an operator $Q$ with eigenvalues $q_1 = q_2 = 0, q_3 = 1$. Consider a state:
    \begin{equation*}
        \state = \frac1{\sqrt{6}} \left(2~\ket1+\ket2+\ket3\right)
    \end{equation*}
    Then the measurement probability $p$ is given by $p(1) = \frac16$ and $p(0) = \frac56$. The state after measuring $1$ is $\ket3$, the state after measuring $0$ is:
    \begin{equation*}
        \frac1{\sqrt5} \left(2~\ket1 + \ket2\right)
    \end{equation*}
\end{example}
\begin{definition}{Expectation}
    The \underline{expectation value} of $Q$ in a state $\state$ is:
    \begin{equation*}
        \E{Q}_\psi = \opbraket\psi{Q}\psi = \sum_\lambda \lambda p(\lambda) = \sum_n q_n \abs{a_n}^2.
    \end{equation*}
\end{definition}
\begin{definition}{Uncertainty}
    The \underline{uncertainty} (also known as spread or variance) is:
    \begin{equation*}
        (\Delta Q)^2 = \E{(Q - \E{Q}_\psi)^2}_\psi = \E{Q^2}_\psi - \E{Q}_\psi^2
    \end{equation*}
\end{definition}
\begin{remark}
    If $\state$ is an eigenstate of $Q$ with eigenvalue $\lambda$, then the expectation is $\lambda$ and variance is $0$.
\end{remark}
Quantum mechanical behaviour arises from the fact that certain observables do not commute. In general, for any state $\state$,
\begin{equation*}
    (\Delta A)_\psi (\Delta B)_\psi \geq \frac12 \abs{\E{\cmmt{A}{B}}_\psi}.
\end{equation*}
For the proof, see the course IB Quantum Mechanics.
\subsection{Simultanous Measurement}
If $A$ and $B$ do not commute, then we cannot expect to measure $A$ and $B$ precisely at the same time. The classic example of this is position and momentum:
\begin{equation*}
    (\Delta x)(\Delta p) \geq \frac\hbar2
\end{equation*}
In 3 dimensions, $\hat{x_i}$ and $\hat{p_i}$ have:
\begin{equation*}
    \cmmt{\hat{x}_i}{\hat{p}_j} = i\hbar \delta_{ij}
\end{equation*}
and so uncertainty only applies when we measure along the same direction.

Our aim with measurement is to be able to describe the quantum system. Our process is to choose an operator $Q_1$. If it has degenerate eigenstates, then we aim to choose another operator $Q_2$ which commutes with $Q_1$ and can distinguish between the degenerate eigenstates. We continue until we can distinguish between all the eigenstates. 

\begin{definition}{Complete set of operators}
    A set of operators $\{Q_k\}_{k=1}^n$ is \underline{complete} if:
    \begin{itemize}
        \item the $Q_k$ all commute;
        \item each normalised eigenstate is uniquely determined by its eigenvalues of the $Q_k$.
    \end{itemize}
\end{definition}
Equivalently, if $A$ is another observable that commutes with the $Q_k$ then it must be a function of them.
\begin{example}
    Return to the example~\ref{exOrthoStates}, with states $\ket1, \ket2, \ket3$ with eigenvalues $0, 0, 1$. We want to find an observable $Q'$ that commutes with $Q$ and can differentiate the states $\ket1$ and $\ket2$. Suppose that we can find an operator:
    \begin{align*}
        Q'\ket1 &= \ket2 \\
        Q'\ket2 &= \ket1 \\
        Q'\ket3 &= 0
    \end{align*}
    Note that this will have eigenvalues:
    \begin{equation*}
        \ket\pm = \frac1{\sqrt{2}} \left(\ket1 \pm \ket2\right)
    \end{equation*}
    with eigenvalues $\pm 1$. Under $Q$, these are still eigenstates with eigenvalue $0$. Therefore, $\{Q, Q'\}$ is a complete commuting set with joint eigenstate basis $\ket\pm, \ket3$.
\end{example}
\section{Bases, Representations and Wavefunctions}
\subsection{Discrete and Continuous Bases}
A choice of basis provides a representation of the quantum states and operators. Under this representation, a discrete basis gives (possibly infinite) vectors and matrices for states and operators. However, we must also consider continuous bases to represent these quantum states and operators.

Consider a basis $\{\ket\lambda\}_{\lambda \in E}$ where $\lambda$ is continuous. Here states are given by:
\begin{equation*}
    \state = \int_E \alpha(\lambda)\ket\lambda~d\lambda
\end{equation*}
where $\alpha(\lambda)$ is some coefficient function, $\alpha(\lambda) = \inn{\psi}{\lambda}$.

For $\mu \in E$, the orthogonality relation is $\inn{\lambda}{\mu} = \delta(\lambda - \mu)$. The completeness relation becomes:
\begin{equation*}
    1 = \int_E \ket\lambda\bra\lambda~d\lambda
\end{equation*}

If these basis states $\ket\lambda$ arise as eigenstates of an observable, then:
\begin{equation*}
    \abs{\alpha(\lambda)}^2 = \abs{\inn{\lambda}{\psi}}^2 \text{ is the probability density.}
\end{equation*}
\subsection{Position and Momentum Bases -- Wavefunctions}
Consider a particle in one dimension with position operator $\hat{x}$ and momentum operator $\hat{p}$ obeying $\cmmt{\hat{x}}{\hat{p}} = i\hbar$.

Let $\ket{x}$ be position eigenstates with eigenvalue $x$. We see that this is a continuous basis of eigenvalues. Any state can now be written as a linear combination:
\begin{equation*}
    \state = \int_\R \psi(x) \ket{x}~dx,\quad \psi(x) = \inn{x}{\psi}.
\end{equation*}
This $\psi(x)$ is the position space wavefunction that was discussed in IB Quantum Mechanics. $\abs{\psi(x)}^2$ is a probability density for measuring position. We note that the inner product is as expected:
\begin{align*}
    \inn{\phi}{\psi} &= \left(\int_{\R} \phi(x) \ket{x}~dx\right)^\dagger \left(\int_{\R} \psi(y) \ket{y}~dy\right) \\
    &= \left(\int_{\R} \phi(x)^* \bra{x}~dx\right) \left(\int_{\R} \psi(y) \ket{y}~dy\right) \\
    &= \int_{\R^2} \phi(x)^*\psi(y) \inn{x}{y}~dx~dy \\
    &= \int_{\R^2} \phi(x)^*\psi(y) \delta(x - y)~dx~dy \\
    &= \int_{\R} \phi^*(x) \psi(x)~dx.
\end{align*}
We can similarly define momentum eigenstates $\ket{p}$, again a continuous basis. Now a state is given by:
\begin{equation*}
    \state{\psi} = \int_{\R} \tilde{\psi}(p) \ket{p}~dp
\end{equation*}
with $\tilde{\psi}(p) = \inn{p}{\psi}$ the momentum space wavefunction. $\abs{\tilde{\psi}(p)}^2$ is the probability density for measurements of momentum.

We would now like to understand how $\psi(x)$ and $\tilde{\psi}(p)$ relate.
\begin{proposition}
    \begin{align*}
        \inn{x}{p} &= \frac{1}{\sqrt{2\pi \hbar}} e^{ipx / \hbar} \\
        \inn{p}{x} &= \frac{1}{\sqrt{2\pi \hbar}} e^{-ipx / \hbar} \\
    \end{align*}
    \label{propXPProducts}
\end{proposition}

\end{proof}
We already know:
\begin{equation*}
    \hat{x}\psi \text{ acts as } \opbraket{x}{\hat{x}}{\psi} = x\inn{x}{\psi} = x\psi(x).
\end{equation*}
Now, using \thmref{propXPProducts}, applying $\hat{p}$ gives:
\begin{equation*}
    p\inn{x}{p} = -i\hbar \frac{\partial}{\partial x} \inn{x}{p}
\end{equation*}
\begin{align*}
    \opbraket{x}{\hat{p}}{\psi} &= \int_{\R} \opbraket{x}{\hat{p}}{p} \inn{p}{\psi}~dp \\
    &= \int_{\R} p\inn{x}{p} \inn{p}{\psi}~dp \\
    &= -i\hbar \frac{\partial }{\partial x} \int_{\R} \inn{x}{p} \inn{p}{\psi}~dp \\
    &= -i\hbar \frac{\partial }{\partial x} \inn{x}{\psi} \\
    &= -i\hbar \psi'(x)
\end{align*}
so $\hat{p}$ acts as $-i\hbar \frac{\partial }{\partial x}$ on position space wavefunctions.

Similarly, on momentum space wavefunctions we find:
\begin{align*}
    \hat{x} &\mapsto i\hbar \frac{\partial}{\partial p} \\
    \hat{p} &\mapsto p
\end{align*}
We can now express the momentum-space wavefunction in terms of the position-space wavefunction:
\begin{align*}
    \tilde{\psi}(p) &= \inn{p}{\psi} \\
    &= \int_{\R} \inn{p}{x} \inn{x}{\psi}~dx \\
    &= \frac{1}{\sqrt{2\pi\hbar}} \int_{\R} e^{-ipx/\hbar} \psi(x)~dx
\end{align*}
which is a Fourier transform!
