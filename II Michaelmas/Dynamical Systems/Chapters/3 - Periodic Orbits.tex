\section{Poincaré Indices}
Consider a general dynamical system:
\begin{equation*}
    \dvec{x} = \vec{f}(\vec{x}) = \begin{pmatrix}f_1(x, y) \\ f_2(x, y)\end{pmatrix}
\end{equation*}
At any given point, the angle the trajectory passing through that point makes with the horizontal is:
\begin{equation*}
    \psi(x, y) = \tan^{-1} \left(\frac{f_2(x, y)}{f_1(x, y)}\right)
\end{equation*}
Note this is not defined at fixed points.

\begin{definition}{Poincaré index}
    The \underline{Poincaré index} of a simple, closed curve $\Gamma$ not passing through any fixed points is:
    \begin{equation*}
        I(\Gamma) = \frac{1}{2\pi} \oint_\Gamma d\psi = \frac1{2\pi} \oint_\Gamma \frac{f_1 df_2 - f_2 df_1}{f_1^2 + f_2^2}
    \end{equation*}
\end{definition}
\begin{propositions}{
        Let $\Gamma$ be a simple, closed curve, traversed anticlockwise and not passing through any fixed points.
        \label{propsPoincIndxProps}
    }
    \item $I(\Gamma)$ is an integer. \label{propPoincIndxInt}
    \item $I(\Gamma)$ is unaffected by smooth, continuous deformations of $\Gamma$ as long as this deformation does not cross any fixed points. \label{propPoincIndxDeform}
    \item If $\Gamma$ does not enclose any fixed points, $I(\Gamma) = 0$. \label{propPoincIndxZero}
    \item Suppose $\Gamma = \Gamma_1 + \Gamma_2$, where $\Gamma_i$ are smooth and do not pass through fixed points. Then $I(\Gamma) = I(\Gamma_1) + I(\Gamma_2)$. \label{propPoincIndxAdd}
    \item $\Gamma$ is symmetric in $t \mapsto -t$. \label{propPoincIndxTimeReverse}
    \item If $\Gamma$ is a periodic orbit, $I(\Gamma) = 1$. \label{propPoincIndxPeriodic}
\end{propositions}
\begin{proof}
    \begin{enumerate}
        \item Parameterise the curve $\Gamma$ as $\vec{x}(t)$, where $t \in [0, 1]$. Evaluate:
            \begin{equation*}
                \frac1{2\pi} \oint_\Gamma d\psi = \frac1{2\pi} \int_{0}^{1} \psi'(t) dt = \frac1{2\pi} \left(\psi(1) - \psi(0)\right)
            \end{equation*}
            Therefore, the integral is the difference in the angle at the start point and the end point. $\Gamma$ is closed so this is the same point. Angles are defined up to modulo $2\pi$ so the difference is a multiple of $2\pi$.
        \item This is a consequence of the deformation property of integrals in $\C$. Note that the integrand (of $x + iy$) is analytic at all points other than fixed points, which means the deformation fails when crossing them.
        \item A consequence of \thmref{propPoincIndxDeform}. Deform $\Gamma$ to an infinitesimal contour on which the field is uniform, so there is no change in the index.
        \item This is simply due to the additive property of line integrals.
        \item We can see this in the definition:
            \begin{align*}
                I(\Gamma) &\mapsto \frac1{2\pi} \oint_\Gamma \frac{(-f_1) (-df_2) - (-f_2) (-df_1)}{f_1^2 + f_2^2} \\
                &= \frac1{2\pi} \oint_\Gamma \frac{f_1 df_2 - f_2 df_1}{f_1^2 + f_2^2} = I(\Gamma).
            \end{align*}
        \item On a periodic orbit $\psi$ is simply a tangent to the orbit. In order for this to close (in an anticlockwise sense) we must have a total change in angle of $2\pi$.
    \end{enumerate}
\end{proof}
Based on these definitions, it is sensible to define:
\begin{definition}{Poincaré index of a fixed point}
    The \underline{Poincaré index} of a fixed point $\vec{x_0}$ is the Poincaré index of any curve that encircles only $\vec{x_0}$.
\end{definition}
\begin{proposition}
    For a curve encircling multiple fixed points, the Poincaré index is equal to the sum of Poincaré indices of the fixed points.
    \label{propPoincIndxManyPts}
\end{proposition}
\begin{proof}
    We consider deforming the contour as in figure~\ref{figDeformedContour}. Then the only remaining contributions are those around the fixed point.
\end{proof}

\begin{proposition}
    The Poincaré index of a hyperbolic sink or source is $1$.
    \label{propPoincIndHyp}
\end{proposition}
\begin{proof}
    Note that for such a point all the trajectories point inwards to the fixed point. We can therefore consider a contour $\gamma$ such that $\psi$ is normal to the curve at every point. This gives a contribution $2\pi$ when integrating around the contour, so the Poincaré index is 1.
\end{proof}

\begin{example}[Poincaré index of a hyperbolic saddle]
    Consider a hyperbolic saddle as in figure~\ref{figSaddleAndContour}, and the green tangent lines. We see that as we go around the blue contour, $\psi$ travels clockwise around which contributes $-2\pi$. Therefore, the Poincaré index of a hyperbolic saddle is $-1$.
\end{example}
For non-hyperbolic points, we must proceed case-by-case. For example, the saddle-node has a Poincaré index of $0$, and a centre has a Poincaré index of $1$.
\begin{proposition}
    Any periodic orbit must encircle at least one fixed point. Furthermore, the sum of their Poincaré indices must be $1$.
    \label{propPeriodicHasFP}
\end{proposition}
\begin{proof}
    Note that the Poincaré index of a periodic orbit is $1$. If there were no fixed point encircled, we could shrink the contour to a tiny point which would require a Poincaré index of $0$. By \thmref{propPoincIndxManyPts}, the sum of the Poincaré indices of the encircled fixed points must be $1$.
\end{proof}
\begin{example}
    Return to the Lotka-Volterra system of example~\ref{exLotkaVolterra}:
    \begin{equation*}
        \dot{r} = r(3-r-2s),\qquad \dot{s} = s(2 - r - s)
    \end{equation*}
    We know that there are nodes at $(0, 0)$, $(3, 0)$, $(0, 2)$. However, the axes are invariant and so we know no trajectories can cross the axis. There is a saddle point at $(1, 1)$. If we wanted a periodic orbit, we would have to encircle at least one node. We cannot encircle any of the nodes because we cannot cross the axes, and we also cannot encircle the point $(1, 1)$ because this has Poincaré index $-1$.

    We can therefore conclude that there are no periodic orbits for this system.
\end{example}
\section{Negative Tests for Periodic Orbits}
\begin{theorem}[Dulac's Criterion]
    If there exists a continuously differentiable function $\phi(x, y)$ such that $\nabla \cdot (\phi \vec{f})$ has the same sign everywhere on a simply-connected domain $D \subseteq \R^2$, then there are no periodic orbits lying entirely in $D$.
    \label{thmDulac}
\end{theorem}
\begin{proof}
    Take $\nabla \cdot (\phi \vec{f}) > 0$, and suppose a periodic orbit $\gamma$ exists. Let $\vec{n}$ be the outward normal on $\gamma$. Note that $\vec{f}$ points normal to the trajectory, so:
    \begin{equation*}
        \oint_\gamma \phi \vec{f} \cdot \vec{n}~dl = 0
    \end{equation*}
    However, by Green's Theorem in the Plane:
    \begin{equation*}
        \oint_\gamma \phi \vec{f} \cdot \vec{n}~dl = \int_A \nabla \cdot (\phi \vec{f})~dS 
    \end{equation*}
    where $A$ is the area enclosed by $\gamma$. However, this second integral must be positive, which is a contradiction.

    The proof is similar for $\nabla \cdot (\phi \vec{f}) < 0$.
\end{proof}
\begin{remark}
    The \underline{divergence test} is \thmref{thmDulac} with $\phi = 1$.
\end{remark}
\begin{corollary}
    If $\nabla \cdot (\phi\vec f)$ maintains its sign in a doubly-connected domain (domain with a hole in it), then there is at most one periodic orbit in $D$ and it must enclose the hole.
    \label{corDblConnected}
\end{corollary}
\begin{proof}
    Suppose there are two periodic orbits. We argue as in \thmref{thmDulac} with $\gamma$ traversing both orbits for a contradiction.
\end{proof}
\begin{example}
    Returning to the more general form of the Lotka-Volterra equations,
    \begin{align*}
        \dot{r} &= r(a - br - c s) \\
        \dot{s} &= s(d - er - fs)
    \end{align*}
    where $a$ to $f$ are constants, and $b, f$ are positive. We consider a function:
    \begin{equation*}
        \phi = \frac1{rs}
    \end{equation*}
    and take the divergence of it and $\vec{f}$:
    \begin{align*}
        \nabla \cdot (\phi\vec f) &= \frac{\partial}{\partial r} \left(\frac{a-br-cs}{s}\right) + \frac{\partial }{\partial s}\left(\frac{d-er-fs}{r}\right) \\
        &= -\frac{b}{s} - \frac f r
    \end{align*}
    which is negative on $r, s > 0$. This implies that there are no periodic orbits by \thmref{thmDulac}.
\end{example}
\begin{remark}
    For $\R^n$, we can adapt \thmref{thmDulac} to exclude the existence of invariant manifolds of dimension $(n-1)$.
\end{remark}
\begin{example}[Lorentz system]
    Consider a dynamical system in $\R^3$:
    \begin{equation*}
        \begin{pmatrix}\dot{x} \\ \dot y \\ \dot z\end{pmatrix} = \begin{pmatrix}\sigma(y-x) \\ rz-y-zx \\ xy-bz\end{pmatrix}
    \end{equation*}
    for $b, \sigma, r > 0$.
    We can consider the divergence test:
    \begin{equation*}
        \nabla \cdot \vec{f} = -\sigma-1-b < 0
    \end{equation*}
    Therefore we cannot have any invariant 2D manifolds.
\end{example}
\begin{theorem}[Gradient Criterion]
    If there exists a positive function $\rho(x, y)$ such that $\rho \vec{f} = \nabla \psi$ for some single-valued $\psi(x, y)$ on a simply-connected domain $D$, then there are no periodic orbits in $D$.
    \label{thmGradient}
\end{theorem}
\begin{proof}
    Suppose we have a periodic orbit $\gamma$. Note that $\vec{dl}$ and $\vec{f}$ are parallel on trajectories:
    \begin{equation*}
        \oint_\gamma \rho \vec{f} \cdot \vec{dl} > 0
    \end{equation*}
    but now substitute in $\psi$:
    \begin{equation*}
        \oint_\gamma \rho \vec{f} \cdot \vec{dl} = \oint_\gamma \nabla \psi \cdot \vec{dl} = 0
    \end{equation*}
    which is a contradiction.
\end{proof}
\section{Positive Tests for Periodic Orbits}
\begin{theorem}[Poincaré-Bendixson Theorem]
    If the forward orbit $O^+(\vec{x})$ of some point $\vec{x}$ enters and remains within a closed, bounded set $D \subseteq \R^2$ that contains no fixed points, then $\omega(\vec{x})$ is a periodic orbit.
    \label{thmPoincBendix}
\end{theorem}
\begin{remark}
    We require $D$ to be closed and bounded so that it contains all of its limit points.
\end{remark}
\begin{proof}
    The proof of this theorem is \textbf{non-examinable}.
\end{proof}
\begin{remarks}
    \item This theorem is only true in $\R^2$.
    \item The set $D$ cannot be simply-connected! Otherwise, we would require a fixed point for the periodic orbit to go around.
\end{remarks}
\begin{example}
    Consider:
    \begin{align*}
        \dot{x} &= x - y - 2x(x^2 + y^2) \\
        \dot{y} &= y + x - y(x^2 + y^2)
    \end{align*}
    We can find an unstable focus at $0$. Also, converting to polar coordinates gives:
    \begin{align*}
        \dot{r} &= r \left[1 - r^2 (1+\cos^2\theta)\right] \\
        \dot{\theta} &= 1 + \frac12 r^2 \sin(2\theta)
    \end{align*}
    A nullcline in $r$ is:
    \begin{align*}
        1 - r^2 (1 + \cos^2\theta) &= 0 \\
        \implies r^2(\sin^2 \theta + 2\cos^2\theta) &= 1 \\
        \implies 2x^2 + y^2 &= 1
    \end{align*}
    which is an ellipse.

    We conclude:
    \begin{equation*}
        \begin{cases}
            \dot{r} = 0 & \text{on }\left(\frac{x}{1 / \sqrt{2}}\right)^2 + y^2 = 1 \\
            \dot{r} < 0 & \text{for }r > 1 \\
            \dot{r} > 0 & \text{for }r < \frac{1}{\sqrt{2}} \\
        \end{cases}
    \end{equation*}
    This means that any $O^+(\vec{x})$ enters and remains in:
    \begin{equation*}
        D = \subsetselect{\vec{x}}{\frac1{\sqrt{2}} < r < 1}
    \end{equation*}
    On $D$, $\dot{\theta > 0}$ and so there are no fixed points. This means we can apply \thmref{thmPoincBendix} to show that there must exist a periodic orbit in this region. Furthermore, by applying the divergence test:
    \begin{equation*}
        \nabla \cdot \vec{f} = (1-6x^2) + (1-3y^2) = 2 - 6x^2 - 3y^2
    \end{equation*}
    which is negative in $D$, and so we conclude that there can only be one periodic orbit in $D$.
\end{example}
\section{Near-Hamiltonian Systems}
\label{secNearHamiltonian}
Consider:
\begin{equation*}
    \begin{pmatrix}\dot{x} \\ \dot{y}\end{pmatrix} = \begin{pmatrix}f_1(x, y) + \epsilon g_1(x, y) \\ f_2(x, y) + \epsilon g_2(x, y)\end{pmatrix}
\end{equation*}
where $f_1 = \frac{\partial H}{\partial y}$ and $f_2 = -\frac{\partial H}{\partial x}$. This means that $\epsilon = 0$ gives a Hamiltonian system, which can have periodic orbits on the contours of $H$.

If $\epsilon \neq 0$,
\begin{equation*}
    \dot{H} = (f_1 + \epsilon g_1)(-f_2) + (f_2 + \epsilon g_2) f_1 = \epsilon(f_1 g_2 - f_2 g_1)
\end{equation*}
which is not identically zero. However, if there exists a periodic orbit $\Gamma$, the integral over this contour $dH$ must be zero, so:
\begin{equation*}
    \epsilon \oint_\Gamma (f_1 g_2 - f_2 g_1) dt = 0.
\end{equation*}
This already tells us that no periodic orbit can lie in a region on which $f_1 g_2 - f_2 g_1$ maintains its sign.

\begin{remark}
    We can understand a Hamiltonian system as one where energy is conserved. Then the small addition of order $\epsilon$ represents dissipation of energy or some increase of energy into the system. In general, we do not expect periodic orbits when we have only dissipation of energy. However, when there is also addition of energy to the system, we can find periodic orbits again.
\end{remark}

If $\epsilon$ is small and $\Gamma_\epsilon$ is a perturbation of the periodic orbit $\Gamma_0$, our aim is to estimate $\Delta H(\Gamma_\epsilon)$ using unperturbed orbits of the Hamiltonian system ($\epsilon = 0$).

Note that $\Gamma_\epsilon = \Gamma_0 + O(\epsilon)$, and the period of the orbit is $T_\epsilon = T_0 + O(\epsilon)$. Over one revolution, the change in the Hamiltonian is:
\begin{align*}
    \Delta H(\Gamma_\epsilon) &= \int_{0}^{T_\epsilon} \dot{H}(t) dt \\
    &= \epsilon\int_{0}^{T_0} (f_1 g_2 - f_2 g_1) dt + O(\epsilon^2)
\end{align*}
Now we have an integral in terms of a known contour (the periodic orbits of the Hamiltonian system). The \underline{energy balance method} is then:
\begin{enumerate}
    \item Find the periodic orbits for $\epsilon = 0$, parameterised by $t$.
    \item Calculate the energy-balance function for the orbit with Hamiltonian equal to $E$:
        \begin{equation*}
            M(E) = \oint_{\Gamma_0(E)} \left(f_1 g_2 - f_2 g_1\right) dt = \oint_{\Gamma_0(E)} \left(g_2 dx - g_1 dy\right)
        \end{equation*}
        Note that $\Delta H = \epsilon M(E) + O(\epsilon^2)$.
    \item Solve $M(E) = 0$ to get values of $E$ that can represent periodic orbits.
    \item~(*) Any double roots of $M(E) = 0$ do not necessarily represent periodic orbits, because then we need higher-order terms.
\end{enumerate}
By Green's Theorem, we can evaluate:
\begin{equation*}
    M(E) = \int_S(E) \left(\frac{\partial g_1}{\partial x} + \frac{\partial g_2}{\partial y}\right) dx~dy
\end{equation*}
where $S(E)$ is the surface enclosed by $\Gamma_0(E)$.

Any candidate energies will then yield periodic orbits that are slight perturbations of $\Gamma_0(E)$.
\begin{example}
    Consider:
    \begin{equation*}
        \begin{pmatrix}\dot{x} \\ \dot y\end{pmatrix} = \begin{pmatrix}y \\ -x+x^2 + \epsilon y (a - x)\end{pmatrix}
    \end{equation*}
    which corresponds to the second-order differential equation $\ddot x - \epsilon \dot x (a-x) + x - x^2 = 0$. We see that the small $\epsilon$ term represents input/dissipation of energy.

    For $\epsilon = 0$ we find the Hamiltonian:
    \begin{equation*}
        H(x, y) = \frac12 (x^2 + y^2) - \frac13 y^3.
    \end{equation*}
    We have fixed points at $(0, 0)$ -- a centre -- and at $(1, 0)$ -- a saddle. We also find a homoclinic orbit passing through $(1, 0)$.

    For $\epsilon \neq 0$ we find $g_1 = 0, g_2 = y(a-x)$.
    \begin{equation*}
        \dot{H} = \epsilon y^2 (a-x)
    \end{equation*}
    and so:
    \begin{equation*}
        \delta H(H_0) = \epsilon \oint_{H = H_0} \epsilon y^2(a-x) = \epsilon \int_0^T y^2(a-x) dt
    \end{equation*}
    \begin{itemize}
        \item This integrand has a fixed sign if $x < a$ or $x > a$. Thus, any periodic orbit must go into both regions.
        \item By the Poincaré index test, we also know that any periodic orbit must go around $(0, 0)$ and not $(1, 0)$.
        \item Considering $\dot{x} = 0$ we require $y = 0$, so the minimum and maximum values of $x$ must lie on the $x$ axis.
        \item The above gives $x_{min} < a < x_{max} < 1$, so $a < 1$ for periodic orbits.
    \end{itemize}
    Calculating $\Delta H$ in terms of $H_0$:
    \begin{align*}
        \Delta H (H_0) &= \epsilon \int_{\Gamma_0} g_2 f_1 dt \\
        &= 2\epsilon \int_{x_{min}(H_0)}^{x_{max}(H_0)} y (a-x) dx
    \end{align*}
    We express $y$ using $H(x, y)$:
    \begin{equation*}
        \Delta H(H_0) = 2\epsilon \int_{x_{min}(H_0)}^{x_{max}(H_0)} (a-x)\sqrt{2H_0 - x^2 + \frac23 x^3} dx
    \end{equation*}
    Studying this numerically, we find:
    \begin{itemize}
        \item no periodic orbits for $a < 0$;
        \item no periodic orbits for $a > \frac17$;
        \item periodic orbits for $a \in \left(0, \frac17\right)$;
        \item sufficiently close to the homoclinic orbit, where $H_0 \approx \frac16$, the integrand factorises as $\frac1{\sqrt{3}} (1-x) \sqrt{1+2x}$ which means the integral can be evaluated exactly (for any $a$) to require $a = \frac17$.
    \end{itemize}
\end{example}
\section{Stability of Periodic Orbits}
Suppose that the system $\dvec{x} = \vec{f}(\vec{x})$ (where $\vec{x} \in \R^n$) has a periodic orbit $\vec{x} = \vec{X}(t)$ with period $T$. Considering a perturbation:
\begin{equation*}
    \vec{x}(t) = \vec X(t) + \vec\eta(t) \implies \dvec{x} = \vec{f}(\vec{X} + \vec\eta) = \vec{f}(\vec X) + \vec \eta \cdot \nabla f(\vec X) + O(|\vec \eta|^2)
\end{equation*}
To linear order, then, $\dvec{\eta} = A\vec\eta$ where $A_{ij} = \frac{\partial f_i}{\partial x_j}$ evaluated at $\vec{X}(t)$. This is a linear ODE with $T$-periodic coefficients: $A_{ij}(t + T) = A_{ij}(t)$. By linearity, introduce a mapping:
\begin{equation*}
    \phi_{ij}(t) : \vec\eta(0) \mapsto \vec\eta(t)
\end{equation*}
We can consider $\phi_{ij}$ to be a time evolution operator. It evolves in time as:
\begin{equation*}
    \dot{\phi}_{ij} \eta_j (0) = A_{ik} \phi_{kj} \eta_j(0)
\end{equation*}
and since this is true for any $\vec{\eta}(t)$,
\begin{equation*}
    \dot\phi_{ij} = A_{ij} \phi_{kj}
\end{equation*}
We can solve this using the matrix exponential:
\begin{equation*}
    \phi(t) = \underbrace{e^{\int_0^t A(t') dt'}}_B \phi(0)
\end{equation*}
and since $\phi(0)$ must be the identity, $\phi = B$.

We are particularly interested in how $\vec\eta(kT)$ behaves. First note that:
\begin{equation*}
    \vec\eta(kT) = \phi(kT)\vec\eta(0) = \left(\phi(T)\right)^k \vec\eta(t)
\end{equation*}
and so the stability of the periodic orbit is decided by the eigenvalues of $\phi(T)$. At least one eigenvalue of $\phi(T)$ is equal to 1, because this corresponds to the direction around the circle.
\begin{definition}{Floquet Multiplier}
    The \underline{Floquet multipliers} of a periodic orbit are the eigenvalues $\lambda_i$ of the matrix $\phi(T)$. The \underline{Floquet exponents} are given by $\mu_i = \frac1T \log\abs{\lambda_i}$, such that $\abs{\lambda_i} = e^{\mu_i T}$.
\end{definition}
\begin{definition}{Hyperbolic Periodic Orbit}
    A periodic orbit is \underline{hyperbolic} if all $\abs{\lambda_i}$ are not equal to $1$ (except the one along the orbit).
\end{definition}
If $\abs{\lambda_i} < 1$ for all $\lambda_i$ except the one along the orbit, the periodic orbit is asymptotically stable. If $\abs{\lambda_i} > 1$ for any $\lambda$, then the periodic orbit is unstable.

In $\R^2$, only one eigenvalue is not equal to $1$. In this case the determinant is $1 \times \lambda$, which determines the stability.
\begin{proposition}
    Let $C$ be a diagonalisable square matrix. Then:
    \begin{equation*}
        \det{e^C} = e^{\tr(C)}
    \end{equation*}
    \label{propDetDiag}
\end{proposition}
\begin{proof}
    Suppose $C = QDQ^{-1}$, where $D$ is diagonal.
    \begin{align*}
        \det{e^C} &= \det \left[\sum_{k=0}^{\infty} \frac{C^k}{k!}\right] \\
        &= \det\left[\sum_{k=0}^{\infty} \frac{(QDQ^{-1})^k}{k!}\right] \\
        &= \det\left[\sum_{k=0}^{\infty} Q\frac{D^k}{k!}Q\right] \\
        &= \det(Q)\det\left[\sum_{k=0}^{\infty} \frac{D^k}{k!}\right] \det(Q^{-1}) \\
        &= \det(e^D)
    \end{align*}
    then because $D$ is diagonal, with diagonal elements $\lambda_i$:
    \begin{align*}
        \det(e^D) &= e^{\sum_{i=1}^{n}\lambda_k} \\
        &= e^{\tr(D)} = e^{\tr(C)}
    \end{align*}
\end{proof}
We can use \thmref{propDetDiag} for $\phi(T)$:
\begin{align*}
    \det(\phi(T)) &= \det \left[e^{\int_{0}^{T} A(t) dt}\right] \\
    &= \exp\left[\tr\left(\int_{0}^{t} A(t) dt \right)\right] \\
    &= \exp\left[\int_{0}^{t} \tr(A(t)) dt\right]
\end{align*}
which is true in any number of dimensions, but is most useful in 2D. First, the trace of $A$ is:
\begin{equation*}
    A_{ij} = \frac{\partial f_i}{\partial x_j} \implies \tr(A) = \frac{\partial f_i}{\partial x_i} = \nabla \cdot \vec{f}
\end{equation*}
This means our non-1 eigenvalue is:
\begin{equation*}
    \lambda = \exp\left[\int_{0}^{T} \nabla \cdot \vec{f}(t) dt\right]
\end{equation*}
In $\R^n$ for $n \geq 2$, all we would get is the product of eigenvalues which is not sufficient to say whether one has modulus greater than 1.

We can summarise our $\R^2$ criterion:
\begin{equation*}
    \int_{0}^{T} \nabla \cdot \vec f(t) dt =
    \begin{cases}
        < 0 & \implies \abs\lambda < 1,\text{ periodic orbit is stable} \\
        > 0 & \implies \abs\lambda > 1,\text{ periodic orbit is unstable} \\
        = 0 & \implies \abs\lambda = 1,\text{ periodic orbit is non-hyperbolic.}
    \end{cases}
\end{equation*}
\begin{example}
    Consider:
    \begin{equation*}
        \dot{r} = r(1-r^2),\qquad\dot\theta = \frac1r
    \end{equation*}
    A periodic orbit has $r = 1, \theta = t$, and so $T = 2\pi$. We consider a perturbation:
    \begin{equation*}
        r(t) = 1 + \delta(t),\qquad \theta(t) = t + \gamma(t).
    \end{equation*}
    Substituting this into the differential equations:
    \begin{align*}
        \dot{r} &= (1 + \delta)(1 - (1 + \delta)^2) \\
        &= (1 + \delta)(1 - 1 - 2\delta - \delta^2) \\
        &= -2\delta + O(\delta^2)
    \end{align*}
    similarly,
    \begin{align*}
        \dot\theta &= \frac{1}{1 + \delta} \\
        &= 1 - \delta + O(\delta^2)
    \end{align*}
    To leading order, then $\dot\delta = -2\delta$ and $\dot\gamma = -\delta$. Solving gives:
    \begin{equation*}
        \begin{pmatrix}\delta(t) \\ \gamma(t)\end{pmatrix} =
        \begin{pmatrix}
            e^{-2t} & 0 \\
            \frac12(e^{-2t} - 1) & 1
        \end{pmatrix}
        \begin{pmatrix}\delta(0) \\ \gamma(0)\end{pmatrix}
    \end{equation*}
    Then taking the determinant of $\phi(T)$ (the matrix above) gives $e^{-2T} < 1$ so the periodic orbit is stable.
\end{example}
\section{The Van der Pol Oscillator}
This section is devoted entirely to the dynamical system:
\begin{equation}
    \ddot{x} + \mu (x^2 - 1) \dot{x} + x = 0
    \label{eqnVanDerPol}
\end{equation}
where $\mu > 0$ is a parameter.

This is a simple harmonic oscillator and a more complicated damping term. For $\abs{x} < 1$, we have energy input (bumping), and for $\abs{x} > 1$ we have damping. However, note that $x = 1$ is not itself a solution.

This is itself a special case of a Lienard equation:
\begin{equation*}
    \ddot{x} + f(x) \dot{x} + g(x) = 0.
\end{equation*}
Using the Lienard transformation $y = \dot{x} + F(x)$ where $F(x) = \int_{0}^{x} f(s) ds$, we can transform this to a dynamical system:
\begin{equation*}
    \dot{x} = y - F(x),\qquad \dot{y} = -g(x).
\end{equation*}
Our Van der Pol oscillator can be transformed in this way to:
\begin{equation}
    \dot{x} = y - \mu \left(\frac{x^3}{3} - x\right),\qquad \dot{y} = -x.
    \label{eqnVanDerPolDS}
\end{equation}
\subsection{Small Values of \texorpdfstring{$\mu$}{the Parameter}}
For $\mu \ll 1$, we have a near-Hamiltonian system with Hamiltonian $H(x, y) = \frac12(x^2 + y^2)$. With $\mu = 0$ the orbits are circular, with period $2\pi$. We can use the energy balance method to see if any periodic orbits survive. In the notation of section~\ref{secNearHamiltonian},
\begin{equation*}
    f_1(x, y) = y,\quad f_2(x, y) = -x,\quad \epsilon = \mu \text{ and } g_1(x, y) = -F(x),\quad g_2 = 0.
\end{equation*}
We parameterise a periodic orbit by $x = \sqrt{2H} \cos(t)$, $y = \sqrt{2H} \sin(t)$, and apply the energy balance method:
\begin{align*}
    \delta H(H) &= - \oint_{\Gamma_0} \epsilon g_1 f_2 dt \\
    &= \mu \int_{0}^{2\pi} \left(x^2 - \frac{x^4}{3}\right) dt \\
    &= \mu \int_{0}^{2\pi} \left(2H\cos^2(t) - \frac43 \cos^4(t)\right) dt \\
    &= 2\pi \mu \left(H - \frac{H^2}{2}\right)
\end{align*}
therefore we find $H = 2$ is a stable periodic orbit.
\subsection{Large Values of \texorpdfstring{$\mu$}{the Parameter}}
Now consider $\mu \gg 1$ and rescale with the variable $Y = \frac Y \mu$.
\begin{equation*}
    \dot{x} = \mu\left[Y - \left(\frac{x^3}{3} - x\right)\right],\qquad \dot{Y} = -\frac x \mu
\end{equation*}
We can consider the phase space of $x$ and $Y$, see figure~\ref{figVanDerPolPS}.

As the orbit approaches the cubic, reaching separation $O(\mu^{-2})$, we find $\dot{x} \sim \dot{Y} \sim O(\mu^{-1})$. Note $\dot{Y}$ is negative, so we cross the curve. Once crossing the curve, $\dot{x}$ is negative and the trajectory follows the cubic.

At $x = 1$, $\dot{Y}$ is still negative but the cubic starts moving upwards. Therefore, the trajectory must separate from the curve, and will again travel almost horizontally (but this time will move left).

We see a rotational symmetry about the origin, and so we will repeat the process indefinitely (see figure~\ref{figVanDerPolPS}).

This is an example of relaxation oscillation - we have a slow buildup (motion along the cubic) and a fast discharge (horizontal segments). This is more obvious by considering the plot of $x$ against $t$, see figure~\ref{figVanDerPolXOssc}. We see that the time of the buildup is $O(\mu)$, and the time of the discharge is $O(\mu^{-1})$. The period is:
\begin{align*}
    P &= 2 \int_{A}^{B} dt \\
    &= 2 \int_{-\frac23}^{\frac23} \frac1{\dot{Y}} dY \\
    &= 2\mu \int_{2}^{1} \frac{1-x^2}{x} dx \\
    &= \mu (3 - 2\log(2))
\end{align*}

\begin{proof}[\textbf{non-examinable}, though the techniques are useful]
    We want to use \thmref{thmPoincBendix}, so we need a doubly-connected domain. We use a bounding function:
    \begin{equation*}
        H_k(x, y) = \frac12\left[x^2 + (y-k)^2\right]
    \end{equation*}
    which has time derivative:
    \begin{align*}
        \dot{H}_k(x, y) &= \dot{x} \frac{\partial H_k}{\partial x} + \dot{y} \frac{\partial H_k}{\partial y} \\
        &= kx - \mu x^2 \left(\frac{x^3}{3} - 1\right).
    \end{align*}
    The simplest case is $k = 0$, for which:
    \begin{equation*}
        \dot{H_0} = -\mu x^2 \left(\frac{x^3}{3} - 1\right).
    \end{equation*}
    This is a quartic function with a double root at the origin, and roots at $\pm \sqrt{3}$. We see that $\dot{H_0} < 0$ for $\abs{x} > \sqrt{3}$, and $\dot{H_0} > 0$ for $\abs{x} < \sqrt{3}$, except at $0$ where $\dot{H}(0, y) = 0$. Note also that $H_0 = \frac12 r^2$. We therefore have that any orbits starting with $0 < r < \sqrt{3}$, the orbit must leave this region.

    Consider now a point $(x_0, y_0)$ with $\abs{x_0} > \sqrt{3}$. Without loss of generality, suppose that $x_0 < 0$. Therefore, $\dot{H_0}$ is negative and so the trajectory is attracted towards $\vec{x_B} = (-\sqrt{3}, y_B)$. Furthermore, $\abs{\vec{x_0}} > \abs{\vec{x_B}}$.

    The orbit must continue to some point $\vec{x_C} = (\sqrt{3}, y_c)$. In this region we know $H_0 > 0$ so the distance from the origin is increasing. We can estimate the change in $H$ in a similar way to the energy balance principle:
    \begin{align*}
        \delta H &= \mu \int_{-\sqrt{3}}^{\sqrt{3}} \left(x^2 - \frac{x^4}{3}\right) \frac{dt}{dx} dt \\
        &= \mu \int_{-\sqrt{3}}^{\sqrt{3}} \frac{x^2 - \frac{x^4}{3}}{y - F(x)} dx 
    \end{align*}
    Now, if $y$ is sufficiently large,
    \begin{align*}
        \frac{1}{y - F(x)} &< \frac{1}{y - \max_{-\sqrt{3} < x < \sqrt{3}} F(x)} \\
        &= \frac{1}{y - \frac23 \mu} \\
        &< \frac{1}{y_0 - \frac23 \mu}.
    \end{align*}
    Returning to $\Delta H$:
    \begin{align*}
        \Delta H &< \mu \int_{-\sqrt{3}}^{\sqrt{3}} \frac{x^2 - \frac{x^4}{3}}{y_0 - \frac23 \mu} dx \\
        &= \frac{4\sqrt{3} \mu}{5 \left(y_0 - \frac23 \mu\right)}
    \end{align*}
    This means:
    \begin{align*}
        H_0(\sqrt{3}, y_C) &= H_0(-\sqrt{3}, y_B) + \Delta H_0 \\
        &< H_0 (x_0, y_0) + \frac{4\sqrt{3} \mu}{5 \left(y_0 - \frac23 \mu\right)} = H_0(C^*)
    \end{align*}
    for some point $C^*$ with:
    \begin{equation*}
        \frac12 r_{c^*} = \frac12 \left(x_0^2 + y_0^2\right) + \frac{4\sqrt{3} \mu}{5 \left(y_0 - \frac23 \mu\right)}.
    \end{equation*}
    Let $D^*$ be the point with the same radius as $C^*$ but on the line $x = \sqrt{3}$. %TODO: Continue from notes.
\end{proof}
