\section{Poincaré Indices}
Consider a general dynamical system:
\begin{equation*}
    \dvec{x} = \vec{f}(\vec{x}) = \begin{pmatrix}f_1(x, y) \\ f_2(x, y)\end{pmatrix}
\end{equation*}
At any given point, the angle the trajectory passing through that point makes with the horizontal is:
\begin{equation*}
    \psi(x, y) = \tan^{-1} \left(\frac{f_2(x, y)}{f_1(x, y)}\right)
\end{equation*}
Note this is not defined at fixed points.

\begin{definition}{Poincaré index}
    The \underline{Poincaré index} of a simple, closed curve $\Gamma$ not passing through any fixed points is:
    \begin{equation*}
        I(\Gamma) = \frac{1}{2\pi} \oint_\Gamma d\psi = \frac1{2\pi} \oint_\Gamma \frac{f_1 df_2 - f_2 df_1}{f_1^2 + f_2^2}
    \end{equation*}
\end{definition}
\begin{propositions}{
        Let $\Gamma$ be a simple, closed curve, traversed anticlockwise and not passing through any fixed points.
        \label{propsPoincIndxProps}
    }
    \item $I(\Gamma)$ is an integer. \label{propPoincIndxInt}
    \item $I(\Gamma)$ is unaffected by smooth, continuous deformations of $\Gamma$ as long as this deformation does not cross any fixed points. \label{propPoincIndxDeform}
    \item If $\Gamma$ does not enclose any fixed points, $I(\Gamma) = 0$. \label{propPoincIndxZero}
    \item Suppose $\Gamma = \Gamma_1 + \Gamma_2$, where $\Gamma_i$ are smooth and do not pass through fixed points. Then $I(\Gamma) = I(\Gamma_1) + I(\Gamma_2)$. \label{propPoincIndxAdd}
    \item $\Gamma$ is symmetric in $t \mapsto -t$. \label{propPoincIndxTimeReverse}
    \item If $\Gamma$ is a periodic orbit, $I(\Gamma) = 1$. \label{propPoincIndxPeriodic}
\end{propositions}
\begin{proof}
    \begin{enumerate}
        \item Parameterise the curve $\Gamma$ as $\vec{x}(t)$, where $t \in [0, 1]$. Evaluate:
            \begin{equation*}
                \frac1{2\pi} \oint_\Gamma d\psi = \frac1{2\pi} \int_{0}^{1} \psi'(t) dt = \frac1{2\pi} \left(\psi(1) - \psi(0)\right)
            \end{equation*}
            Therefore, the integral is the difference in the angle at the start point and the end point. $\Gamma$ is closed so this is the same point. Angles are defined up to modulo $2\pi$ so the difference is a multiple of $2\pi$.
        \item This is a consequence of the deformation property of integrals in $\C$. Note that the integrand (of $x + iy$) is analytic at all points other than fixed points, which means the deformation fails when crossing them.
        \item A consequence of \thmref{propPoincIndxDeform}. Deform $\Gamma$ to an infinitesimal contour on which the field is uniform, so there is no change in the index.
        \item This is simply due to the additive property of line integrals.
        \item We can see this in the definition:
            \begin{align*}
                I(\Gamma) &\mapsto \frac1{2\pi} \oint_\Gamma \frac{(-f_1) (-df_2) - (-f_2) (-df_1)}{f_1^2 + f_2^2} \\
                &= \frac1{2\pi} \oint_\Gamma \frac{f_1 df_2 - f_2 df_1}{f_1^2 + f_2^2} = I(\Gamma).
            \end{align*}
        \item On a periodic orbit $\psi$ is simply a tangent to the orbit. In order for this to close (in an anticlockwise sense) we must have a total change in angle of $2\pi$.
    \end{enumerate}
\end{proof}
Based on these definitions, it is sensible to define:
\begin{definition}{Poincaré index of a fixed point}
    The \underline{Poincaré index} of a fixed point $\vec{x_0}$ is the Poincaré index of any curve that encircles only $\vec{x_0}$.
\end{definition}
\begin{proposition}
    For a curve encircling multiple fixed points, the Poincaré index is equal to the sum of Poincaré indices of the fixed points.
    \label{propPoincIndxManyPts}
\end{proposition}
\begin{proof}
    We consider deforming the contour as in figure~\ref{figDeformedContour}. Then the only remaining contributions are those around the fixed point.
\end{proof}

\begin{proposition}
    The Poincaré index of a hyperbolic sink or source is $1$.
    \label{propPoincIndHyp}
\end{proposition}
\begin{proof}
    Note that for such a point all the trajectories point inwards to the fixed point. We can therefore consider a contour $\gamma$ such that $\psi$ is normal to the curve at every point. This gives a contribution $2\pi$ when integrating around the contour, so the Poincaré index is 1.
\end{proof}

\begin{example}[Poincaré index of a hyperbolic saddle]
    Consider a hyperbolic saddle as in figure~\ref{figSaddleAndContour}, and the labelled points $1-7$. We see that as we go round $1$ to $7$, $\psi$ travels clockwise around which contributes $-2\pi$. Therefore, the Poincaré index of a hyperbolic saddle is $-1$.
\end{example}
For non-hyperbolic points, we must proceed case-by-case. For example, the saddle-node has a Poincaré index of $0$, and a centre has a Poincaré index of $1$.
\begin{proposition}
    Any periodic orbit must encircle at least one fixed point. Furthermore, the sum of their Poincaré indices must be $1$.
    \label{propPeriodicHasFP}
\end{proposition}
\begin{proof}
    Note that the Poincaré index of a periodic orbit is $1$. If there were no fixed point encircled, we could shrink the contour to a tiny point which would require a Poincaré index of $0$. By \thmref{propPoincIndxManyPts}, the sum of the Poincaré indices of the encircled fixed points must be $1$.
\end{proof}
\begin{example}
    Return to the Lotka-Volterra system of example~\ref{exLotkaVolterra}:
    \begin{equation*}
        \dot{r} = r(3-r-2s),\qquad \dot{s} = s(2 - r - s)
    \end{equation*}
    We know that there are nodes at $(0, 0)$, $(3, 0)$, $(0, 2)$. However, the axes are invariant and so we know no trajectories can cross the axis. There is a saddle point at $(1, 1)$. If we wanted a periodic orbit, we would have to encircle at least one node. We cannot encircle any of the nodes because we cannot cross the axes, and we also cannot encircle the point $(1, 1)$ because this has Poincaré index $-1$.

    We can therefore conclude that there are no periodic orbits for this system.
\end{example}
\section{Negative Tests for Periodic Orbits}
\begin{theorem}[Dulac's Criterion]
    If there exists a continuously differentiable function $\phi(x, y)$ such that $\nabla \cdot (\phi \vec{f})$ has the same sign everywhere on a simply-connected domain $D \subseteq \R^2$, then there are no periodic orbits lying entirely in $D$.
    \label{thmDulac}
\end{theorem}
\begin{proof}
    Take $\nabla \cdot (\phi \vec{f}) > 0$, and suppose a periodic orbit $\gamma$ exists. Let $\vec{n}$ be the outward normal on $\gamma$. Note that $\vec{f}$ points normal to the trajectory, so:
    \begin{equation*}
        \oint_\gamma \phi \vec{f} \cdot \vec{n}~dl = 0
    \end{equation*}
    However, by Green's Theorem in the Plane:
    \begin{equation*}
        \oint_\gamma \phi \vec{f} \cdot \vec{n}~dl = \int_A \nabla \cdot (\phi \vec{f})~dS 
    \end{equation*}
    where $A$ is the area enclosed by $\gamma$. However, this second integral must be positive, which is a contradiction.

    The proof is similar for $\nabla \cdot (\phi \vec{f}) < 0$.
\end{proof}
\begin{remark}
    The \underline{divergence test} is \thmref{thmDulac} with $\phi = 1$.
\end{remark}
\begin{corollary}
    If $\nabla \cdot (\phi\vec f)$ maintains its sign in a doubly-connected domain (domain with a hole in it), then there is at most one periodic orbit in $D$ and it must enclose the hole.
    \label{corDblConnected}
\end{corollary}
\begin{proof}
    Suppose there are two periodic orbits. We argue as in \thmref{thmDulac} with $\gamma$ traversing both orbits for a contradiction.
\end{proof}
\begin{example}
    Returning to the more general form of the Lotka-Volterra equations,
    \begin{align*}
        \dot{r} &= r(a - br - c s) \\
        \dot{s} &= s(d - er - fs)
    \end{align*}
    where $a$ to $f$ are constants, and $b, f$ are positive. We consider a function:
    \begin{equation*}
        \phi = \frac1{rs}
    \end{equation*}
    and take the divergence of it and $\vec{f}$:
    \begin{align*}
        \nabla \cdot (\phi\vec f) &= \frac{\partial}{\partial r} \left(\frac{a-br-cs}{s}\right) + \frac{\partial }{\partial s}\left(\frac{d-er-fs}{r}\right) \\
        &= -\frac{b}{s} - \frac f r
    \end{align*}
    which is negative on $r, s > 0$. This implies that there are no periodic orbits by \thmref{thmDulac}.
\end{example}
\begin{remark}
    For $\R^n$, we can adapt \thmref{thmDulac} to exclude the existence of invariant manifolds of dimension $(n-1)$.
\end{remark}
\begin{example}[Lorentz system]
    Consider a dynamical system in $\R^3$:
    \begin{equation*}
        \begin{pmatrix}\dot{x} \\ \dot y \\ \dot z\end{pmatrix} = \begin{pmatrix}\sigma(y-x) \\ rz-y-zx \\ xy-bz\end{pmatrix}
    \end{equation*}
    for $b, \sigma, r > 0$.
    We can consider the divergence test:
    \begin{equation*}
        \nabla \cdot \vec{f} = -\sigma-1-b < 0
    \end{equation*}
    Therefore we cannot have any invariant 2D manifolds.
\end{example}
\begin{theorem}[Gradient Criterion]
    If there exists a positive function $\rho(x, y)$ such that $\rho \vec{f} = \nabla \psi$ for some single-valued $\psi(x, y)$ on a simply-connected domain $D$, then there are no periodic orbits in $D$.
    \label{thmGradient}
\end{theorem}
\begin{proof}
    Suppose we have a periodic orbit $\gamma$. Note that $\vec{dl}$ and $\vec{f}$ are parallel on trajectories:
    \begin{equation*}
        \oint_\gamma \rho \vec{f} \cdot \vec{dl} > 0
    \end{equation*}
    but now substitute in $\psi$:
    \begin{equation*}
        \oint_\gamma \rho \vec{f} \cdot \vec{dl} = \oint_\gamma \nabla \psi \cdot \vec{dl} = 0
    \end{equation*}
    %TODO: Why is this zero??
    which is a contradiction.
\end{proof}
\section{Positive Tests for Periodic Orbits}
\begin{theorem}[Poincaré-Bendixson Theorem]
    If the forward orbit $O^+(\vec{x})$ of some point $\vec{x}$ enters and remains within a closed, bounded set $D \subseteq \R^2$ that contains no fixed points, then $\omega(\vec{x})$ is a periodic orbit.
    \label{thmPoincBendix}
\end{theorem}
\begin{remark}
    We require $D$ to be closed and bounded so that it contains all of its limit points.
\end{remark}
\begin{proof}
    The proof of this theorem is \textbf{non-examinable}.
\end{proof}
\begin{remarks}
    \item This theorem is only true in $\R^2$.
    \item The set $D$ cannot be simply-connected! Otherwise, we would require a fixed point for the periodic orbit to go around.
\end{remarks}
\begin{example}
    Consider:
    \begin{align*}
        \dot{x} &= x - y - 2x(x^2 + y^2) \\
        \dot{y} &= y + x - y(x^2 + y^2)
    \end{align*}
    We can find an unstable focus at $0$. Also, converting to polar coordinates gives:
    \begin{align*}
        \dot{r} &= r \left[1 - r^2 (1+\cos^2\theta)\right] \\
        \dot{\theta} &= 1 + \frac12 r^2 \sin(2\theta)
    \end{align*}
    A nullcline in $r$ is:
    \begin{align*}
        1 - r^2 (1 + \cos^2\theta) &= 0 \\
        \implies r^2(\sin^2 \theta + 2\cos^2\theta) &= 1 \\
        \implies 2x^2 + y^2 &= 1
    \end{align*}
    which is an ellipse.

    We conclude:
    \begin{equation*}
        \begin{cases}
            \dot{r} = 0 & \text{on }\left(\frac{x}{1 / \sqrt{2}}\right)^2 + y^2 = 1 \\
            \dot{r} < 0 & \text{for }r > 1 \\
            \dot{r} > 0 & \text{for }r < \frac{1}{\sqrt{2}} \\
        \end{cases}
    \end{equation*}
    This means that any $O^+(\vec{x})$ enters and remains in:
    \begin{equation*}
        D = \subsetselect{\vec{x}}{\frac1{\sqrt{2}} < r < 1}
    \end{equation*}
    On $D$, $\dot{\theta > 0}$ and so there are no fixed points. This means we can apply \thmref{thmPoincBendix} to show that there must exist a periodic orbit in this region. Furthermore, by applying the divergence test:
    \begin{equation*}
        \nabla \cdot \vec{f} = (1-6x^2) + (1-3y^2) = 2 - 6x^2 - 3y^2
    \end{equation*}
    which is negative in $D$, and so we conclude that there can only be one periodic orbit in $D$.
\end{example}
\section{Near-Hamiltonian Systems}
Consider:
\begin{equation*}
    \begin{pmatrix}\dot{x} \\ \dot{y}\end{pmatrix} = \begin{pmatrix}f_1(x, y) + \epsilon g_1(x, y) \\ f_2(x, y) + \epsilon g_2(x, y)\end{pmatrix}
\end{equation*}
where $f_1 = \frac{\partial H}{\partial y}$ and $f_2 = -\frac{\partial H}{\partial x}$. This means that $\epsilon = 0$ gives a Hamiltonian system, which can have periodic orbits on the contours of $H$.

If $\epsilon \neq 0$,
\begin{equation*}
    \dot{H} = (f_1 + \epsilon g_1)(-f_2) + (f_2 + \epsilon g_2) f_1 = \epsilon(f_1 g_2 - f_2 g_1)
\end{equation*}
which is clearly non-zero. However, if there exists a periodic orbit $\Gamma$, the integral over this contour $dH$ must be zero, so:
\begin{equation*}
    \epsilon \oint_\Gamma (f_1 g_2 - f_2 g_1) dt = 0.
\end{equation*}
This already tells us that no periodic orbit can lie in a region on which $f_1 g_2 - f_2 g_1$ maintains its sign.

For very small positive $\epsilon$, we can use the energy balance method.

\begin{remark}
    We can understand a Hamiltonian system as one where energy is conserved. Then the small addition of order $\epsilon$ represents dissipation of energy or some increase of energy into the system. In general, we do not expect periodic orbits when we have only dissipation of energy. However, when there is also addition of energy to the system, we can find periodic orbits again.
\end{remark}

If $\Gamma_\epsilon$ is a perturbation of the periodic orbit $\Gamma_0$, our aim is to estimate $\Delta H(\Gamma_\epsilon)$ using unperturbed orbits of the Hamiltonian system ($\epsilon = 0$).

Note that $\Gamma_\epsilon = \Gamma_0 + O(\epsilon)$, and the period of the orbit is $T_\epsilon = T_0 + O(\epsilon)$. This means that the change in the Hamiltonian is:
\begin{equation*}
    \Delta h = \underbrace{\int_0^{T_\epsilon} H dt}_{\text{on }\Gamma_\epsilon} = \epsilon\underbrace{\int_0^{T_0} H dt}_{\text{on }\Gamma_0} + O(\epsilon^2)
\end{equation*}
\begin{example}
    Consider:
    \begin{equation*}
        \begin{pmatrix}\dot{x} \\ \dot y\end{pmatrix} = \begin{pmatrix}y \\ -x+x^2 + \epsilon y (a - x)\end{pmatrix}
    \end{equation*}
    which corresponds to the second-order differential equation $\ddot x - \epsilon \dot x (a-x) + x - x^2 = 0$. We see that the small $\epsilon$ term represents input/dissipation of energy.

    For $\epsilon = 0$ we find the Hamiltonian:
    \begin{equation*}
        H(x, y) = \frac12 (x^2 + y^2) - \frac13 y^3.
    \end{equation*}
    We have fixed points at $(0, 0)$ -- a centre -- and at $(1, 0)$ -- a saddle. We also find a homoclinic orbit passing through $(1, 0)$.

    For $\epsilon \neq 0$ we find $g_1 = 0, g_2 = y(a-x)$.
    \begin{equation*}
        \dot{H} = \epsilon y^2 (a-x)
    \end{equation*}
    and so:
    \begin{equation*}
        \delta H(H_0) = \epsilon \oint_{H = H_0} \epsilon y^2(a-x) = \epsilon \int_0^T y^2(a-x) dt
    \end{equation*}
    \begin{itemize}
        \item This integrand has a fixed sign if $x < a$ or $x > a$. Thus, any periodic orbit must go into both regions.
        \item By the Poincaré index test, we also know that any periodic orbit must go around $(0, 0)$ and not $(1, 0)$.
        \item Considering $\dot{x} = 0$ we require $y = 0$, so the minimum and maximum values of $x$ must lie on the $x$ axis.
        \item The above gives $x_{min} < a < x_{max} < 1$, so $a < 1$ for periodic orbits.
    \end{itemize}
    Calculating $\Delta H$ in terms of $H_0$:
    \begin{align*}
        \Delta H (H_0) &= \epsilon \int_{\Gamma_0} g_2 f_1 dt \\
        &= 2\epsilon \int_{x_{min}(H_0)}^{x_{max}(H_0)} y (a-x) dx
    \end{align*}
    We express $y$ using $H(x, y)$:
    \begin{equation*}
        \Delta H(H_0) = 2\epsilon \int_{x_{min}(H_0)}^{x_{max}(H_0)} (a-x)\sqrt{2H_0 - x^2 + \frac23 x^3} dx
    \end{equation*}
    Studying this numerically, we find:
    \begin{itemize}
        \item no periodic orbits for $a < 0$;
        \item no periodic orbits for $a > \frac17$;
        \item periodic orbits for $a \in \left(0, \frac17\right)$;
        \item sufficiently close to the homoclinic orbit, where $H_0 \approx \frac16$, the integrand factorises as $\frac1{\sqrt{3}} (1-x) \sqrt{1+2x}$ which means the integral can be evaluated exactly (for any $a$) to require $a = \frac17$.
    \end{itemize}
\end{example}
\section{Stability of Periodic Orbits}
Suppose that the system $\dvec{x} = \vec{f}(\vec{x})$ (where $\vec{x} \in \R^n$) has a periodic orbit $\vec{x} = \vec{X}(t)$ with period $T$. Considering a perturbation:
\begin{equation*}
    \vec{x}(t) = \vec X(t) + \vec\eta(t) \implies \dvec{x} = \vec{f}(\vec{X} + \vec\eta) = \vec{f}(\vec X) + \vec \eta \cdot \nabla f(\vec X) + O(|\vec \eta|^2)
\end{equation*}
To linear order, then, $\dvec{\eta} = A\vec\eta$ where $A_{ij} = \frac{\partial f_i}{\partial x_j}$ evaluated at $\vec{X}(t)$. This is a linear ODE with $T$-periodic coefficients: $A_{ij}(t + T) = A_{ij}(t)$. By linearity, introduce a mapping:
\begin{equation*}
    \phi_{ij}(t) : \vec\eta(0) \mapsto \vec\eta(t)
\end{equation*}
We can consider $\phi_{ij}$ to be a time evolution operator. It evolves in time as:
\begin{equation*}
    \dot{\phi}_{ij} \eta_j (0) = A_{ik} \phi_{kj} \eta_j(0)
\end{equation*}
and since this is true for any $\vec{\eta}(t)$,
\begin{equation*}
    \dot\phi_{ij} = A_{ij} \phi_{kj}
\end{equation*}
We can solve this using the matrix exponential:
\begin{equation*}
    \phi = \underbrace{e^{\int_0^t A(t') dt'}}_B \phi(0)
\end{equation*}
and since $\phi(0)$ must be the identity, $\phi = B$.
