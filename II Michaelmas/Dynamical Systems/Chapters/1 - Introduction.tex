\section{Motivation and Examples}
\begin{definition}{Dynamical system}
    A \underline{dynamical system} is a mathematical entity described by a set of equations for a set of variables with respect to a time-like variable.
\end{definition}
In this course we will want to understand such systems. In particular, we will study:
\begin{itemize}
    \item long term behaviours of dynamical systems;
    \item periodic solutions;
    \item stability of solutions and chaos;
    \item how parameters of a system affect the solutions and behaviour.
\end{itemize}
\subsection{Continuous Time}
When our time-like variable is continuous, the dynamical system is described by a set of ODEs.
\begin{example}
    Consider a simple 1D system:
    \begin{equation*}
        \dot{x} = x(1-x^2)
    \end{equation*}
    This is a very easy example and in fact we find an analyical solution:
    \begin{equation*}
        x(t) = \pm \left(1-2e^2(t_0 - t)\right)^{-\frac12}
    \end{equation*}
    However, to practice the methods we would like to apply to more difficult systems, we will forget about this solution.

    We begin with analysis of fixed points. Solving the RHS gives stationary points at $x = 0, \pm 1$. Then we can also see that the RHS is positive on $(-\infty, -1)$ and $(0, 1)$, negative on the rest. This means that the fixed points at $-1$ and $1$ are stable (and attract solutions), whereas the fixed point at $0$ is unstable (and repels solutions). Therefore we can draw a diagram like in figure~\ref{figSimpleSystem}.
\end{example}

    \label{figSimpleSystem}
    \caption{Diagram of fixed points in a simple system}
\end{figure}
\begin{example}[Lotka-Volterra equations]
    Let $r$ be the number of rabbits in a population and $s$ be the number of sheep. Then for $a, b, c, \cdots, f > 0$, the Lotka-Volterra equations for these populations are:
    \begin{align*}
        \dot{r} &= r(a - br - cs) \\
        \dot{s} &= s(d - er - fs)
    \end{align*}
    The \underline{state space} of these equations is $(r, s) \in [0, \infty) \times [0, \infty)$.

    The constants $a$ and $d$ model unconstrained birth rate, and then $b, c, e, f$ model scarcity of food (where we assume that the rabbits and sheep both eat grass, which is constrained).
    Consider the example with $a = 3, d = 2, b=c=e=f=1$.
    \begin{align*}
        \dot{r} &= r(3 - r - s) \\
        \dot{s} &= s(2 - r - s)
    \end{align*}
    Then we can identify fixed points:
    \begin{equation*}
        (0, 0),~~(3, 0),~~(0, 2)
    \end{equation*}
    We also have fixed points for the individual equations when $r+s \in \{2, 3\}$.

    Now consider the signs of the derivatives. For large $r$ and $s$, both derivatives are negative. We can also see that $\dot{s} < 0$ when $r + s = 3$, and $\dot{r} > 0$ when $r + s = 2$. By then interpolating the rest, we get the diagram in figure~\ref{figLotkaVolterra}. We see that the behaviour as $t \to \infty$ is:
    \begin{equation*}
        (r, s) \to
        \begin{cases}
            (0, 0) & r = s = 0 \\
            (0, 2) & r = 0, s > 0 \\
            (3, 0) & \text{otherwise}.
        \end{cases}
    \end{equation*}
\end{example}
\begin{figure}
    \label{figLotkaVolterra}
    \caption{Diagram of solutions to Lotka-Volterra equations}
\end{figure}
\begin{example}[Predator-prey system]
    Consider a number of sheep $s$ and number of wolves $w$. These populations can be modelled by the system:
    \begin{align*}
        \dot{w} &= w(-a + bs) \\
        \dot{s} &= s(c - dw)
    \end{align*}
    for positive constants $a, b, c, d$.

    The fixed points are now $(0, 0)$ and $\left(\frac{a}{b}, \frac{c}{d}\right)$. By drawing our diagram, we see that we find periodic solutions around the non-zero fixed point.
\end{example}
\begin{example}\label{exPolarSpirals}
    Consider:
    \begin{align*}
        \dot{x} &= -y + \epsilon x(\mu - x^2 - y^2) \\
        \dot{y} &= x + \epsilon y (\mu - x^2 - y^2)
    \end{align*}
    We change coordinate system to plane polar coordinates to achieve:
    \begin{align*}
        \dot{r} &= \frac{x\dot{x} + y\dot{y}}{r} \\
        \dot{\theta} &= \frac{x\dot{y} - y\dot{x}}{r^2}
    \end{align*}
    then substituting this into the original equations gives:
    \begin{equation*}
        \dot{r} = \epsilon r(\mu - r^2),\qquad\dot{\theta} = 1.
    \end{equation*}
    Now our system depends sensitively on $\epsilon$ and $\mu$. We can clearly see that $\epsilon = 0$ gives circular solutions, since $\dot{r} = 0$.

    In the case $\epsilon > 0, \mu \leq 0$ we have $\dot{r} < 0$ and the solution spirals in towards the origin. The case $\epsilon > 0, \mu \leq 0$ gives spirals outwards.

    The case $\mu > 0$ is more interesting. We now have a radial fixed point at $r = \sqrt{\mu}$.
    \begin{itemize}
        \item For $\epsilon > 0$, this fixed point is stable and the solution spirals towards it.
        \item For $\epsilon < 0$, this fixed point is unstable and the solution spirals away from it.
    \end{itemize}
\end{example}
\subsection{Discrete Time}
When our time-like variable is discrete, the dynamical system is described by a map. For example, the logistic map is:
\begin{equation*}
    x_{n+1} = \mu x_n(1-x_n),~~\mu \in [0, 1]
\end{equation*}

We will often like to think of maps like this as discretised versions of differential equations.

\begin{example}[Poincare Map]
    Consider a trajectory in $\R^3$. Label the points on the plane $z = 0$ that the trajectory crosses as $x_1, x_2, \cdots$. Draw directional lines between the points on a plane. Then this is a Poincare map of the $\R^3$ trajectory to $\R^2$ space. Poincare maps are useful to diagnose chaos in a system.
\end{example}
\section{Defining a Dynamical System}
\subsection{Basic Definitions}
\begin{definition}{State space}
    A \underline{state space} $E \subseteq \R^n$ is the set of possible states of a dynamical system.
\end{definition}
\begin{remark}
    A state space is a vector field.
\end{remark}
Then $\vec{x} \in E$ evolves according to a set of ODEs:
\begin{equation*}
    \dvec{x} = \frac{d \vec{x}}{dt} = \vec{f}(\vec{x}, t)
\end{equation*}
where $f : E \times \R \mapsto E$. 
\begin{definition}{Autonomous system}
    A dynamical system where $\vec{f}$ (as given above) does not depend on $t$ is \underline{autonomous}.
\end{definition}
We can reduce a non-autonomous system to an autonomous one by considering $\vec{y} \in E \times \R$ described by:
\begin{equation*}
    \dvec{y} = \begin{pmatrix} \vec{f}(\vec{y}) \\ 1\end{pmatrix}
\end{equation*}
Therefore we can restrict attention only to autonomous systems.

Note also that $m$th-order ODEs can be reduced to $m$ first-order ODEs in $m$ derivatives, so it is also sufficient to consider only first-order ODEs.

Discrete-time dynamical systems can also be autonomous if the map does not depend on $t$.
\subsection{Initial-Value Problem}
We want to study the problem:
\begin{equation}
    \begin{cases}
        \dvec{x} = \vec{f}(\vec{x}, t) & t>t_0 \\
        \vec{x}(t_0) = \vec{x_0} & \vec{x_0} \in E
    \end{cases}
    \label{eqnInitialValue}
\end{equation}
The problem in \eqnref{eqnInitialValue} is the \underline{initial value problem}.

\begin{definition}{Flow (of a system)}
    For an autonomous system, define a \underline{flow} $\phi_t(\vec{x})$ such that $\phi_t(\vec{x_0})$ solves \eqnref{eqnInitialValue} with $\vec{x}(0) = \vec{x_0}$.
\end{definition}
\begin{remarks}
    \item $\phi$ is a vector field.
    \item $\phi_0(\vec{x_0}) = \vec{x_0}$.
    \item $\phi_s(\phi_t(\vec{x_0})) = \phi_{t+s}(\vec{x_0})$, because the system is autonomous.
\end{remarks}
We now want to understand the conditions for a solution to exist, and the conditions for it to be unique.
\begin{theorem}[Cauchy-Peano Theorem]
    Let $\dvec{x} = f(\vec{x}, t)$ (with $\vec{x}(0) = \vec{x_0}$) be a dynamical system. Then if $f$ is continuous and $\norm{f} < M$ on the domain $D = {\abs{t - t_0} < \alpha, \norm{\vec{x} - \vec{x_0}} < \beta}$, then the initial value problem has a solution for:
    \begin{equation*}
        \abs{t - t_0} < \min \left(\alpha, \frac{\beta}{M}\right).
    \end{equation*}
    \label{thmCauchyPeano}
\end{theorem}
\begin{example}[Theorem~\ref{thmCauchyPeano} is not sufficient for uniqueness]
    \begin{enumerate}
        \item Consider the 1D problem:
            \begin{equation*}
                \dot{x} = \abs{x},~~x(t_0) = x_0
            \end{equation*}
            which has solution:
            \begin{equation*}
                \begin{cases}
                    x_0 e^{t_0 - t} & x_0 < 0 \\
                    0 & x_0 = 0 \\
                    x_0 e^{t - t_0} & x_0 > 0.
                \end{cases}
            \end{equation*}
            Note that $f(x) = \abs{x}$ is everywhere continuous, but not differentiable at $0$. We have a unique solution for $x(t)$, so it is not necessary for $f$ to be differentiable to guarantee uniqueness.
        \item Consider the 1D problem:
            \begin{equation*}
                \dot{x} =
                \begin{cases}
                    \sqrt{x} & x \geq 0 \\
                    0 & x < 0
                \end{cases}
                \qquad x(0) = 0.
            \end{equation*}
            We see again that $f$ is continuous but not differentiable at $0$. We can check that:
            \begin{equation*}
                x(t) =
                \begin{cases}
                    0 & t < \tau \\
                    \frac14 (t - \tau)^2 & t \geq \tau
                \end{cases}
            \end{equation*}
            satisfies the equation and initial condition. However, we do not have uniqueness here! Indeed, $\tau$ can be chosen arbitrarily.
    \end{enumerate}
    We have now demonstrated two continuous but non-differentiable $f$, but only one gives unique solutions.
\end{example}
\begin{definition}{Lipschitz continuity}
    A function $f : E \subseteq \R^n \mapsto \R^n$ is $l$-\underline{Lipschitz} at a point $\vec{x_0}$ if there exists $a \in \R$ such that:
    \begin{equation*}
        \norm{\vec{x} - \vec{x_0}} < a, \norm{\vec{y} - \vec{x_0}} < a \implies \norm{f(\vec{x}) - f(\vec{y})} \leq l\norm{\vec{x} - \vec{y}}.
    \end{equation*}
\end{definition}
\begin{theorem}[Uniqueness of Solution to IVP]
    Let $\dvec{x} = \vec{f}(\vec{x})$, $\vec{x}(t) = \vec{x_0}$ be an initial value problem in autonomous form.

    If $f$ is $l$-Lipschitz at $x_0$ then the solution of a flow $\phi_{t - t_0}(\vec{x_0})$ exists and is unique in a neighbourhood of $(\vec{x_0}, t_0)$.
    \label{thmIVPUnique}
\end{theorem}
\begin{example}[Theorem~\ref{thmIVPUnique} does not guarantee uniqueness for all time]
    Consider:
    \begin{equation*}
        \dot{x} = x^2,\qquad x(0)=1.
    \end{equation*}
    Then we can see that $f(x) = x^2$ is Lipschitz, and we have solution:
    \begin{equation*}
        x(t) = \frac{1}{1-t}.
    \end{equation*}
    However, we have that $x(t) \to \infty$ as $t \to 1$, so we do not have a unique solution for all time.
\end{example}
We will focus on problems with differentiable $\vec{f}$, which ensures we have unique solutions.
\section{Orbits, Invariant and Limit Sets}
\begin{definition}{Orbit}
    The \underline{orbit} or \underline{trajectory} of a flow $\phi_t(\vec{x})$ through a point $\vec{x_0}$ is the set:
    \begin{equation*}
        O(\vec{x_0}) = \subsetselect{\phi_t(\vec{x_0})}{t \in (-\infty, \infty)}
    \end{equation*}
\end{definition}
The \underline{forward orbit} is the set $O^+(\vec{x_0})$, it is the orbit with $t > 0$. The \underline{backward orbit} is similarly defined for $t < 0$.
\begin{definition}{Invariant set}
    A set of points $\Lambda \in E$ is \underline{invariant} under $\vec{f}$ if:
    \begin{equation*}
        \vec{x} \in \Lambda \implies O(\vec{x}) \subseteq \Lambda
    \end{equation*}
\end{definition}
\begin{remark}
    Any orbit is an invariant set.
\end{remark}
\begin{definition}{Fixed point}
    A point $\vec{x_0}$ is a \underline{fixed point} if $\vec{f}(\vec{x_0}) = \vec{0}$.
\end{definition}
We see that $O(\vec{x_0}) = \{\vec{x_0}\}$.
\begin{definition}{Periodic point}
    A point $\vec{x_0}$ is a \underline{periodic point} with period $T$ if $\phi_T(\vec{x_0}) = \vec{x_0}$ but $\phi_t(\vec{x_0}) \neq \vec{x_0}$ for all $t \in (0, T)$. Then the set:
    \begin{equation*}
        \subsetselect{\phi_t(\vec{x_0})}{t \in [0, T)}
    \end{equation*}
    is the \underline{periodic orbit} through $\vec{x_0}$.
\end{definition}
\begin{example}
    Consider:
    \begin{equation*}
        \begin{pmatrix}\dot{x} \\ \dot{y}\end{pmatrix} = \begin{pmatrix}-y^3 \\ x^3\end{pmatrix}
    \end{equation*}
    which has solution $x^4 + y^4 = C$ constant.
    \label{exPeriodicCycle}
\end{example}
\begin{definition}{Limit cycle}
    A \underline{limit cycle} is an isolated periodic orbit, where there are no other periodic orbits within a small neighbourhood around it.
\end{definition}
We see that example~\ref{exPeriodicCycle} does not contain a limit cycle, because we can add an arbitrarily small amount to the constant to generate a nearby periodic orbit.
\begin{example}
    Consider the problem in example~\ref{exPolarSpirals} with $\epsilon = \mu = 1$. In polar coordinates we obtain:
    \begin{equation*}
        \dot{r} = r(1-r^2),\quad\dot\theta = 1.
    \end{equation*}
    We have a fixed point at the origin, and note that the periodic orbit at $r = 1$ is a limit cycle.
\end{example}
\begin{definition}{Homoclinic/heteroclinic orbits}
    For a point $\vec{y}$, suppose that:
    \begin{align*}
        \phi_t(\vec{y}) &\to \vec{x_0}\text{ as }t \to \infty\\
        \phi_t(\vec{y}) &\to \vec{x_1}\text{ as }t \to -\infty
    \end{align*}
    If $\vec{x_0} = \vec{x_1}$ then the orbit $O(\vec{y})$ is a \underline{homoclinic orbit}. If $\vec{x_0} \neq \vec{x_1}$ then the orbit is a \underline{heteroclinic orbit}.
\end{definition}
\begin{example}
    Consider a torus in 3 dimensions. Describe the motion as in figure %TODO
    by $\theta_1$ and $\theta_2$. Suppose that these increase as:
    \begin{equation*}
        \dot\theta_1 = \omega_1,\quad \dot\theta_2 = \omega_2.
    \end{equation*}
    Then if $\frac{\omega_1}{\omega_2}$ is rational, we will trace out a trajectory along the torus and it will be periodic. If it is irrational, then we will never return to the same place, and the whole torus is invariant.
\end{example}
\begin{definition}{$\omega$-/$\alpha$-limit set}
    An \underline{$\omega$-limit set} of $\vec{x_0}$ is given by:
    \begin{equation*}
        \omega(\vec{x_0}) = \subsetselect{\vec{y}}{\text{there exists a sequence $t_1, t_2, \cdots \to \infty$ with $\phi_{t_n}(\vec{x_0}) \to \vec{y}$}}
    \end{equation*}
    An \underline{$\alpha$-limit set} is given by:
    \begin{equation*}
        \alpha(\vec{x_0}) = \subsetselect{\vec{y}}{\text{there exists a sequence $t_1, t_2, \cdots \to -\infty$ with $\phi_{t_n}(\vec{x_0}) \to \vec{y}$}}
    \end{equation*}
\end{definition}
When $O(\vec{x})$ is bounded, $\omega(\vec{x})$ has some nice properties. It is:
\begin{itemize}
    \item non-empty;
    \item invariant under $\vec{f}$;
    \item closed;
    \item bounded;
    \item connected.
\end{itemize}
\section{Topological Equivalence}
\begin{definition}{Topological/geometrical equivalence}
    Two flows $\phi^f_t(\vec{x})$ and $\phi^g_t(\vec{x})$ are \underline{topologically equivalent} or \underline{geometrically} \underline{equivalent} if there is a homeomorphism (continuous bijection with continuous inverse) $\vec{h}(\vec{x}) : E^f \mapsto E^g$ and time-increasing function $\tau(\vec{x}, t)$ such that for all $\vec{x}, t$:
    \begin{equation*}
        \phi^f_t(\vec{x}) = \vec{h}^{-1} \circ \phi^g_\tau \circ \vec{h}(\vec{x})
    \end{equation*}
    and
    \begin{equation*}
        \tau(\vec{x}, t_1 + t_2) = \tau(\vec{x}, t_1) + \tau(\phi^f_{t_1}(\vec{x}), t_2)
    \end{equation*}
    for all $t_1, t_2$.
\end{definition}
\begin{remark}
    %TODO: Go back over start of lecture and understand. See diagram also.

    Less precisely, two flows are topologically equivalent when we can find nice maps of space and time between them.
\end{remark}
\begin{propositions}{
        Let $\phi^f_t(\vec{x})$ and $\phi^g_\tau(\vec{x})$ be two topologically equivalent flows.
        \label{propsTopEquiv}
    }
    \item If $\vec{x_1}$ is a fixed point of one, it is a fixed point of the other.
    \item If an orbit is periodic in one flow, it is periodic in another, but the period may differ.
\end{propositions}
\begin{definition}{Topological conjugacy}
    Two flows are \underline{topologically conjugate} if they are topologically equivalent and $t = \tau$ (where these are as defined above).
\end{definition}
\begin{example}
    Consider two flows in polar coordinates:
    \begin{align*}
        \phi^f_t: &~\dot{r} = -r,~~\dot{\theta} = 1 \\
        \phi^g_t: &~\dot{\rho} = i-2\rho,~~\dot{\phi} = 0 \\
    \end{align*}
    Solving gives:
    \begin{align*}
        \phi_t^f(r_0, \theta_0) &= \left(r_0 e^{-t}, \theta_0 + t\right) \\
        \phi_t^g(\rho_0, \phi_0) &= \left(\rho_0 e^{-2t},\phi_0\right).
    \end{align*}
    We would like to now show topological conjugacy. We consider:
    \begin{equation*}
        \vec{h}(\vec{x}) =
        \begin{cases}
            \vec0 & \vec{x} = \vec0 \\
            (r^2, \theta + \ln(r)) & \text{otherwise}
        \end{cases}
    \end{equation*}
    We will show topological conjugacy using the equivalent statement:
    \begin{equation*}
        \vec{h} \circ \phi^f_t = \phi_t^g \circ \vec{h}
        \label{RTP} %TODO: How to make this a (RTP) on the right?
    \end{equation*}
    \begin{align*}
        \vec{h} \circ \phi^f_t(\vec{x}) &= \left(r_0^2 e^{-2t}, \theta_0 + t + \ln\left(r_0 e^{-t}\right)\right) \\
        &= \left(r_0^2 e^{-2t}, \theta_0 + \ln(r_0)\right) \\
        \intertext{And similarly,}
        \phi^g_t(\vec{h}(\vec{x})) &= \left(r_0^2 e^{-2t}, \theta_0 + \ln(r_0)\right)
    \end{align*}
    so the flows are topologically conjugate, as required.
\end{example}
\begin{example}
    Consider two flows in polar coordinates:
    \begin{align*}
        \phi^f_t: &~\dot{r} = 0,~~\dot{\theta} = 1 \\
        \phi^g_\tau: &~\dot{\rho} = 0\rho,~~\dot{\phi} = \rho + \sin^2\phi \\
    \end{align*}
    In this case, the shapes of the orbits are the same and so $\vec{h}(\vec{x}) = \vec{x}$. We must find an appropriate $\tau(\vec{x}, t)$ for topological equivalence.

    We must solve:
    \begin{equation*}
        \frac{d}{d\tau} = (r + \sin^2(\theta)) \frac{d}{dt}
    \end{equation*}
    Note that $\phi_t^f(r_0, \theta_0) = (r_0, \theta_0 + t)$, and so we can find $\tau$:
    \begin{equation*}
        \tau = \int^t \frac{1}{r_0^2 + \sin^2(\theta_0 + t')} dt'
    \end{equation*}
\end{example}
\begin{definition}{Structural stability}
    A vector field $\vec{f}$ (or system $\dvec{x} = \vec{f}(\vec{x})$, or flow $\phi_t(\vec{x})$) \underline{structurally stable} if there exists $\epsilon > 0$ such that for any vector field $\vec{\delta}$ satisfying
    \begin{equation*}
        \norm{\delta} + \sum_i \norm{\frac{\partial \delta}{\partial x_i}} < \epsilon,
    \end{equation*}
    $\vec{f} + \vec{\delta}$ is topologically equivalent to $\vec{f}$.
\end{definition}
\begin{remark}
    We can say that a vector field is structurally stable when small perturbations to not break topological equivalence.
\end{remark}
