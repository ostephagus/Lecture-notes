In this chapter we will consider flows with $\Re \ll 1$ and so we neglect the inertial term in \eqnref{eqnNavierStokes}, to get:
\begin{equation}
    0 = \vec{f} - \nabla p + \mu \nabla^2 \vec{u}
    \label{eqnStokes}
\end{equation}
Note that $\Re = \frac{UL}{\nu}$. Therefore, the approximation holds for low speeds, small length scales or very large dynamic viscosities (such as lava or ice flows).

We will generally assume that $\vec{f} = \vec0$. This permits the case $\vec{f} = \rho \vec{g}$ as long as we consider the modified pressure $p \mapsto p - \rho g z$. In this case,
\begin{equation}
    0 = -\nabla p + \mu \nabla^2 \vec{u} = \nabla \cdot \sigma
    \label{eqnStokesNoF}
\end{equation}
\section{Simple Properties}
\begin{enumerate}
    \item \textbf{Instantaneity}: we have no $\frac{\partial}{\partial t}$ which means that the flow only depends on boundary conditions (and $\vec{f}$ if present) at each instant. It may depend on time parametrically (``quasi-steady'') if the boundary conditions depend on time.
    \item \textbf{Linearity}: we have no $(\vec{u} \cdot \nabla)\vec{u}$ term which means that velocity, pressure and stress are linear in the boundary conditions (and $\vec{f}$ if present).
    \item \textbf{Time reversibility}: if we change the sign of all the boundary conditions and $\vec{f}$ then we change the sign of $\vec{u}$. This reverses all the boundary conditions and the history of their application, so the fluid particles retrace their paths exactly.
    \item \textbf{Spatial symmetries}: consider a symmetry (such as rotation or reflection). This can sometimes be used (along with linearity and time reversibility) to show that a quantity is zero.
    \item \textbf{Harmonic properties}: Consider \eqnref{eqnStokesNoF}. Then taking the divergence gives $\nabla^2 p = 0$. Furthermore, taking the curl of the equation gives that $\nabla^2 \vec{\omega} = 0$, and taking the Laplacian of the equation gives that $\vec{u}$ is biharmonic: $\nabla^2 \nabla^2 \vec{u} = \nabla^4 \vec{u} = 0$.
\end{enumerate}
\begin{example}
    Consider an arbitrarily shaped body moving through a very viscous fluid. The drag vector will also affect the body, and by property 2 must be linearly related to the speed of the body. The most general linear relationship is:
    \begin{equation*}
        \text{drag} = -A\vec{U}
    \end{equation*}
    for some second-rank tensor $A$.

    If in fact the body is axisymmetric along some axis, we can then decompose the problem into one with $\vec{U}$ parallel to the axis of symmetry, and with $\vec{U}$ perpendicular to the axis of symmetry.
    \label{exParallelToWall}
\end{example}
\begin{example}
    Consider a sphere falling under $\vec{g}$ next to a vertical wall. Suppose that $\vec{U}$ has some component away from the wall, as in figure~\ref{figSphereFalling}. Now reversing the time direction means that the velocity is now point up and towards the wall, gravity is pointing upwards. Note that we also have a symmetry in $y \mapsto -y$ (if $y$ is the upwards direction), then applying this symmetry gives the same picture as figure~\ref{figSphereFalling}, but with $\vec{U}$ reflected along the $y$ axis. However, these picture should be the same so we require that $\vec{U}$ is unaffected by reflection in the $y$, axis, so it must in fact be parallel to the $\vec{y}$ axis.
\end{example}

\section{Further Properties}
We can give some further properties of the Stokes' equation, that need more proof.
\begin{proposition}
    Consider Stokes' equation with no force:
    \begin{equation*}
        \mu \nabla^2 \vec{u} = \nabla p
        \tag{\ref{eqnStokesNoF}}
    \end{equation*}
    Let $\vec{u_1}, \vec{u_2}$ be solutions to \eqnref{eqnStokes} in a domain $V$ and satisfy the same boundary conditions on $\partial V$.
    \label{propStokesUnique}
\end{proposition}
\begin{proof}
    Let $\vec{u} = \vec{u_1} - \vec{u_2}$. Similarly, let $e = e_1 - e_2$, same for $\sigma$. Consider:
    \begin{align*}
        2\mu \int e_{ij} e_{ij} dV &= \int_V \left(\sigma_{ij} + p \delta_{ij}\right)e_{ij}~dV \\
        &= \int_V \sigma_{ij} \frac{\partial u_i}{\partial x_j} dV \\
        &= \int_V \left(\frac{\partial}{\partial x_j} \left(u_i \sigma_{ij}\right) - u_i \frac{\partial \sigma_{ij}}{\partial x_j}\right) dV \\
        &= \int_V \left(\frac{\partial}{\partial x_j} \left(u_i \sigma_{ij}\right) - u_i \underbrace{\nabla \cdot \sigma}_{0}\right) dV \\
        &= \int_{\partial V} \vec{u} \cdot \sigma\vec{dS} \\
        &= 0
    \end{align*}
    because either $\vec{u_1} = \vec{u_2}$ on the boundary, or $\sigma_1 \vec{n} = \sigma_2\vec{n}$.
    Therefore, since this was the integral of a squared quantity ($e^2$), $e = 0$ in $V$. This means that $\vec{u}$ is zero in the case of velocity boundary conditions, and differs only by a solid body motion in the case of stress boundary conditions.
\end{proof}
\begin{theorem}[Minimum Dissipation Theorem]
    Let $\vec{u}^S$ be a Stokes flow in $V$ and $\vec{u}$ be any incompressible flow that has the same velocity on $\partial V$. Then:
    \begin{equation}
        2\mu \int_V e_{ij}^S e_{ij}^S dV \leq 2\mu \int_V e_{ij} e_{ij} dV
        \label{eqnMinimumDissp}
    \end{equation}
    \label{thmMinimumDissp}
\end{theorem}
\begin{proof}
    We construct the inequality as follows:
    \begin{align*}
        0 &\leq 2\mu \int_V \left(e - e^S\right)_{ij} \left(e - e^S\right)_{ij}~dV \\
        &= 2\mu \int_V \left(e_{ij} e_{ij} - e_{ij}^S e_{ij}^S\right) dV + 4\mu \int_V e^S_{ij} \left(e^S - e\right)_{ij}~dV
    \end{align*}
    If we can show that the last term is zero then we are done:
    \begin{align*}
        4\mu \int_V e^S_{ij} &\left(e^S - e\right)_{ij}~dV= 2\int_V \sigma_{ij}^S \left(e_{ij}^S - e_{ij}\right)dV \\
        &= 2 \int_V \frac{\partial}{\partial x_j} \left(\left(u_i^S - u_i\right)\sigma_{ij}^S - (u_i^S - u_i) \frac{\partial \sigma_{ij}}{\partial x_j}\right) dV \\
        &= 2 \int_V \frac{\partial}{\partial x_j} \left(\left(u_i^S - u_i\right)\sigma_{ij}^S\right) dV \\
        &- 2\int_{\partial V} \left(u_i^S - u_i\right) \sigma_{ij} dS_j \\
        &= 0
    \end{align*}
    because the velocities match on the boundary.
\end{proof}
\begin{example}[Bounding Drag on a Translating Rigid Body]
    Consider an arbitrary rigid body translating with velocity $\vec{U}$ and with Stokes flow $\vec{u}^S$ in the fluid volume $V$ around it. Consider a sphere $S$ that completely contains the body. Taking $\vec{n}$ into the body, the force $\vec{F_S}$ exerted by the body on the fluid is:
    \begin{equation*}
        \vec{F_S} = \int_{\partial V} \sigma\vec{dS}.
    \end{equation*}
    The dissipation must equal the rate of doing work by the body on the fluid:
    \begin{align*}
        2\mu \int_{V} e^S_{ij} e^S_{ij} dV &= \int_{\partial V} \vec{u}^S \cdot \sigma^S \vec{dS} \\
        &= \vec{U} \cdot \int_{\partial V} \sigma^S \vec{dS} = \vec{U} \cdot \vec{F_S}.
    \end{align*}
    Now suppose that $S$ is a sphere with radius $a$, and define a flow $\vec{u}$ in the original fluid volume $V$ such that:
    \begin{itemize}
        \item $\vec{u}$ is a Stokes flow outside a rigid sphere of radius $a$, when $\vec{x}$ is outside $S$,
        \item $\vec{u} = \vec{U}$ in the gap between $S$ and the body.
    \end{itemize}
    \begin{align*}
        2\mu \int_V e_{ij} e_{ij}~dV &= 2\mu \int_{V \backslash S} e_{ij} e_{ij}~~\text{ because $e = 0$ in $S$} \\
        &= \vec{U} \cdot \left(6\pi \mu a \vec{U}\right)
    \end{align*}
    where the $6\pi \mu a \vec{U}$ will be derived later. Applying \thmref{thmMinimumDissp},
    \begin{equation*}
        \vec{U} \cdot \vec{F^S} \leq 6\pi \mu a |\vec{U}|^2.
    \end{equation*}

    By similar argument for a sphere completely contained within the body, of radius $b$,
    \begin{equation*}
        \vec{U} \cdot \vec{F}^S \geq 6\pi \mu b |\vec{U}|^2
    \end{equation*}

    In the example of a cube of side lengths $2l$, we take $b = L$ and $a = \sqrt{3}l$. By the symmetries of the cube we know that $\vec{F}^S = A\vec{U}$ with the same $A$ for any orientation, and we deduce:
    \begin{equation*}
        1 \leq \frac{A}{6\pi \mu L} \sqrt{3}
    \end{equation*}
\end{example}
\begin{theorem}[Reciprocal Theorem]
    Let $\vec{u}^1$ and $\vec{u}^2$ be Stokes flows in some domain $V$ with different boundary conditions on $V$. Then:
    \begin{equation}
        \int_{\partial V} \vec{u}^1 \cdot \sigma^2 \vec{n} dS = \int_{\partial V} \vec{u}^2 \sigma^1 \vec{n} dS
        \label{eqnReciprocal}
    \end{equation}
    \label{thmReciprocal}
\end{theorem}
\begin{proof}
    Consider:
    \begin{align*}
        2\mu \int_V e^1_{ij} e^2_{ij}~dV &= \int_V \left(\sigma^1_{ij} + p^1 \delta_{ij}\right) e^2_{ij}~dV \\
        &= \int_{\partial V} u^2\cdot \sigma^1 \vec{n}dS
    \end{align*}
    by using symmetry of $\sigma^1$, $\nabla \cdot \vec{u}^2 = 0$ and $\nabla \cdot \sigma^1 = \vec{0}$.

    We can apply the same manipulations to the LHS to get the other way round.
\end{proof}
\begin{example}
    Suppose $\vec{u}^1$ and $\vec{u}^2$ are the Stokes flows due to rigid body motion of a certain body (same $V$ for both velocity fields). That is,
    \begin{equation*}
        \vec{u}^i = \vec{U}^i + \vec{\Omega}^i \times \vec{x}
    \end{equation*}
    on $\partial V$ with $\vec{U}^i$ and $\vec{\Omega}^i$ constants.

    \Thmref{thmReciprocal} gives:
    \begin{equation}
        \vec{U}^1 \cdot \vec{F}^2 + \vec{\Omega}^1 \cdot \vec{G}^2 = \vec{U}^2 \cdot \vec{F}^1 + \vec{\Omega}^2 \cdot \vec{G}^1
        \label{eqnSolidBodyReciproc}
    \end{equation}
    where $\vec{F} = \int_{\partial V} \sigma\vec{n}~dS$ is the force, $\vec{G} = \int_{\partial V} \vec{x} \times (\sigma \vec{n}) dS$ is the couple exerted.

    By linearity of the Stokes equations,
    \begin{align*}
        \vec{F} &= A\vec{U} + B\vec{\Omega} \\
        \vec{G} &= C\vec{U} + D\vec{\Omega}
    \end{align*}
    for 2nd-rank tensors $A, B, C, D$. \Eqnref{eqnSolidBodyReciproc} gives us that $A, D$ are symmetric, $B = C^T$.

    For spheres and cubes, their symmetries imply that $A = aI, D = dI, B=C=0$. For more interesting objects, like a helix, $B$ and $C$ are not zero. A translating helix will require a torque to stop it rotating, and similarly a rotating helix will require a linear force to stop it translating.
\end{example}

\subsection{Translating Sphere}
Consider a sphere translating with velocity $\vec{U}$ through fluid that is otherwise at rest. We can find the solution by changing to an inertial reference frame going at speed $-U$ (and then flipping $\vec{U}$) with respect to the solution of subsection~\ref{sscStationarySphere}, to get the solution:
\begin{equation*}
    \vec{u} = \vec{U} \left(\frac{3a}{4r} + \frac{a^4}{4r^3}\right) + (\vec{U} \cdot \vec{x}) \vec{x} \left(\frac{3a}{4r^3} - \frac{3a^3}{4r^5}\right)
\end{equation*}
and we can also find pressure and stress:
\begin{equation*}
    p = \frac{3\mu a}{2} \frac{\vec{U} \cdot \vec{x}}{r^3},\qquad \sigma \vec{n}|_{r = a} = \frac{-3\mu}{2a} \vec{U}
\end{equation*}
The fluid drag on the sphere is $-6\pi \mu a \vec{U}$, which is opposite and equal to the force exerted by the sphere.

In the case of a sedimenting sphere -- a sphere falling under its weight with no inertial term -- we have the force balance:
\begin{equation*}
    \vec{F} = \frac43 \pi a^2 (\rho_S - \rho) \vec{g}
\end{equation*}
and so balancing this with the drag gives:
\begin{equation*}
    \vec{U} = \frac29 \frac{\Delta \rho~a^2}{\mu}\vec{g}
\end{equation*}
\subsection{Far Field}
Our solution for $\vec{u}$ has terms $O(r^{-1})$ and $O(r^{-3})$. For $r \gg a$, the $O(r^{-1})$ terms dominate, and the sphere just looks like a point exerting a force on the fluid at $\vec{x} = \vec{0}$. The disturbance flow (either from uniform flow or from rest) can be written:
\begin{equation*}
    \vec{u}' = \frac{1}{8\pi\mu} \left(\frac1r\vec{F} + \frac1{r^3}(\vec{F} \cdot \vec{x}) \vec{x}\right)
\end{equation*}
Then this is the solution for a point force $\vec{f}(\vec{x}) = \vec{F}\delta(\vec{x})$, which is known as a ``Stokeslet''.
\section{Corner Flows}
There are many examples of 2D flow in a corner:
\begin{itemize}
    \item scraping fluid off a rigid boundary;
    \item hinged plates;
    \item source flow (stationary rigid plates at an angle to each other with a source at the join);
    \item flow past a corner (see figure~\ref{figFlowCorner})
\end{itemize}

Sufficiently close to the corner, we expect $L \sim r$ and $\Re \ll 1$ (for source flow we need $U \sim \frac1r and UL \sim Q \ll \nu$).

We can use a 2D streamfunction $\psi$ with $\vec{u} = \nabla \times \psi\vec{e_z}$ and plane polar coordinates:
\begin{equation*}
    \vec{u} = \begin{pmatrix}\frac1r \frac{\partial \psi}{\partial \theta} \\ -\frac{\partial \psi}{\partial r} \\ 0\end{pmatrix}
\end{equation*}
The vorticity is given by $\vec{\omega} = -\nabla^2\psi \vec{e_z}$. Vorticity must be harmonic in Stokes flow so the streamfunction must be biharmonic: $\nabla^4 \psi = 0$. We recover the pressure using the Stokes equation,
\begin{equation*}
    \nabla p = \mu \nabla^2\vec{u} = \mu \nabla \times (\nabla^2 \psi \vec{e_z}).
\end{equation*}
\begin{example}[Source flow]
    Consider the stup in figure~\ref{figSourceFlow}. Mass conservation suggests trying:
    \begin{equation*}
        u_r \propto \frac{Q}{r}.
    \end{equation*}
    We can try a streamfunction of the form:
    \begin{equation*}
        \psi = Qf(\theta)
    \end{equation*}
    Then the biharmonic equation gives:
    \begin{align*}
        0 &= \nabla^4 \psi \\
        &= \left(\frac1r \frac{\partial }{\partial r} \left(r \frac{\partial}{\partial r}\right) + \frac1{r^2} \frac{\partial^2 }{\partial \theta^2}\right) \frac{Qf''(\theta)}{r^2} \\
        \implies& f^{(4)}(\theta) + 4 f''(\theta) = 0 \\
        \implies& f(\theta) = A\cos(2\theta) + B \sin(2\theta) + C + D\theta
    \end{align*}
    Now since $u_r$ is an even function in $\theta$, and so $\psi$ is odd, we can immediately say $A = C = 0$.
    
    The no-slip condition is $u_r(\pm \alpha) = 0 \implies f'(\pm \alpha) = 0$. Applying this at $\alpha$:
    \begin{equation*}
        D = -2B\cos(2\alpha)
    \end{equation*}
    The final condition is that the volume flux at the origin is $Q$:
    \begin{equation*}
        Q = \int_{-\alpha}^{\alpha} u_r~r d\theta = Q(f(\alpha) - f(-\alpha))
    \end{equation*}
    so $f(\pm \alpha) = \pm \frac12$, and we have all the constants determined:
    \begin{equation*}
        f(\theta) = \frac12 \frac{\sin(2\theta) - 2\theta \cos(2\theta)}{\sin(2\alpha) - 2\alpha \cos(2\alpha)}
    \end{equation*}
    which gives our solution for $u_r$:
    \begin{equation*}
        u_r = \frac{Q}{r} \frac{\cos(2\theta) - \cos(2\alpha)}{\sin(2\alpha) - 2\alpha \cos(2\alpha)}
    \end{equation*}
    We note:
    \begin{itemize}
        \item As $\alpha \to 0$, $u_r$ has a parabolic profile.
        \item $u_r > 0$ on the whole domain only if $2\alpha < 180^\circ$. If $\alpha > 90^\circ$ then we get inflow by the walls.
        \item If the denominator is zero (happens when $2\alpha = 257^\circ$) then we have a problem.
    \end{itemize}
\end{example}
