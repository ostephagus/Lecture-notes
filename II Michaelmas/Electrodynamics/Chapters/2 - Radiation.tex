We will seek to understand the production of electromagnetic waves by oscillating charge distributions:
\begin{enumerate}
    \item macroscopic charge densities / currents involving non-relativistic motion, including scattering; and
    \item accelerated charges in arbitrary motion.
\end{enumerate}
\section{Retarded Potentials}
Recall Maxwell's equations:
\begin{align*}
    \partial_\mu F^{\mu\nu} &= -\mu_0 J^\nu \tag{\ref{eqnMaxForced}} \\
    F_{\mu\nu} &= \partial_\mu A_\nu - \partial_\nu A_\mu \tag{\ref{eqn4VecPot}}
\end{align*}
Combining the two,
\begin{equation*}
    \partial_\mu \left(\partial^\mu A^\nu - \partial^\nu A^\mu\right) = -\mu_0 J^\nu
\end{equation*}
and we can use the D'Alambertian operator $\Box = \eta^{\mu\nu} \partial_\mu \partial_\nu$:
\begin{equation}
    \Box A^\mu - \partial^\nu \partial_\mu A^\mu = -\mu_0 J^\nu
    \label{eqnDAlamBadGauge}
\end{equation}
We must choose a gauge to solve this. A convenient choice is Lorenz (not Lorentz!) gauge $\partial_\mu A^\mu = 0$, which can be found by the transformation:
\begin{equation*}
    A^\mu \mapsto A^\mu - \partial^\mu \chi, \text{ where } \Box \chi = - \partial_\nu A^\nu
\end{equation*}
In this gauge, \eqnref{eqnDAlamBadGauge} becomes:
\begin{equation}
    \Box A^\mu = -\mu_0 J^\mu
    \label{eqnDAlam}
\end{equation}
which is a sourced wave equation. We solve this using a Fourier transform in time:
\begin{equation*}
    A^\mu (t, \vec{x}) = \frac1{\sqrt{2\pi}} \int_{-\infty}^{\infty} \tilde{A}^\mu (\omega, \vec{x}) e^{-i\omega t} d\omega 
\end{equation*}
The Fourier transform of \eqnref{eqnDAlam} is:
\begin{equation*}
    (\nabla^2 + k^2) \tilde{A} = -\mu_0 \tilde{J}
\end{equation*}
where $k = \frac{\omega}{c}$. This can be solved with a Green's function $G(\vec{x};\vec{y})$ satisfying:
\begin{equation}
    (\nabla^2 + k^2) G(\vec{x};\vec{y}) = \delta(\vec{x} - \vec{y})
    \label{eqnGreens}
\end{equation}
which would give:
\begin{equation*}
    \tilde{A}(\omega, \vec{x}) = -\mu_0 \int_{\R^3} G(\vec{x}; \vec{y})\tilde{J}^\mu (\omega, \vec{x}) d^3\vec{y}.
\end{equation*}
We still need to give a boundary condition for $G$. For localised currents/charges, we want the free-space Green's function with boundary condition:
\begin{equation*}
    G(\vec{x}; \vec{y}) \to 0 \text{ as } \abs{\vec{x} - \vec{y}} \to \infty.
\end{equation*}
Note that \eqnref{eqnGreens} is homogeneous and isotropic, so $G$ may only depend on $\abs{\vec{x} - \vec{y}}$. To solve for $G$ we can take $r = \abs{\vec{x} - \vec{y}}$ and so \eqnref{eqnGreens} becomes:
\begin{equation*}
    \frac1{r^2} \frac{d}{dr} \left(r^2 \frac{dG}{dr}\right) + k^2 G = 0\qquad r \neq 0
\end{equation*}
which has solutions proportional to $\frac1r e^{ikr}$ and $\frac1r e^{-ikr}$. We take the positive exponent due to causality (see later on). The constant of proportionality is fixed by integrating \eqnref{eqnGreens} on a sphere of radius $\epsilon$ centred on $\vec{x} = 0$:
\begin{align*}
    \int_{B_\epsilon} \nabla \cdot (\nabla G) d^3\vec{x} + k^2 \int_{B_\epsilon} G~d^3\vec{x} &= 1 \\
    \int_{\partial B_\epsilon} (\nabla G) \cdot \vec{dS} + k^2 \int_{B_\epsilon} G~d^3\vec{x} &= 1 \\
    4\pi \epsilon^2 \left.\frac{dG}{dr}\right|_{\epsilon}+ k^2 \int_{B_\epsilon} G~d^3\vec{x} &= 1 \\
    %TODO: Put in dG/dr here and show orders
\end{align*}
and so taking the limit $\epsilon \to 0$ gives that the constant of proportionality must be $-\frac1{4\pi}$:
\begin{equation*}
    G(\vec{x};\vec{y}) = -\frac1{4\pi} \frac{e^{ik\abs{\vec{x} - \vec{y}}}}{\abs{\vec{x} - \vec{y}}}
\end{equation*}
and it follows that:
\begin{align}
    A^\mu (t, \vec{x}) &= \frac{\mu_0}{4\pi} \int_{-\infty}^{\infty} \frac1{\sqrt{2\pi}} e^{-i\omega t} \int_{\R^3} \frac{e^{ik\abs{\vec{x} - \vec{y}}}}{\abs{\vec{x} - \vec{y}}}\tilde{J}^\mu (\omega, \vec{y}) d^3\vec{y}~ d\omega \nonumber \\
    &= \frac{\mu_0}{4\pi} \int_{\R^3} \frac{J^\mu(t_{ret}, \vec{y})}{\abs{\vec{x} - \vec{y}}} d^3\vec{y} \label{eqnRetardedPotential}
\end{align}
where the \underline{retarded time} is:
\begin{equation*}
    t_{ret} = t - \frac1c \abs{\vec{x} - \vec{y}}
\end{equation*}
so we can think of this as the integral over a past light cone from $(t, \vec{x})$. This is why we had to discount the $e^{-ikr}$ term earlier, because otherwise we would be integrating over the future light cone which violates causality.

We want to show that \eqnref{eqnRetardedPotential} is covariant. To establish this, we rewrite as:
\begin{equation}
    A^\mu(X) = \frac{\mu_0}{4\pi} \int J^\mu(Y) \frac{\delta(Y^0 - X^0 + |\vec{x} - \vec{y}|)}{|\vec{x} - \vec{y}|} d^4Y
    \label{eqnRetardedPotentialRewrite}
\end{equation}
We need the following proposition.
\begin{proposition}
    Define $R^\mu = X^\mu - Y^\mu$. Then the fraction of \eqnref{eqnRetardedPotentialRewrite} is:
    \begin{equation*}
        \frac{\delta(Y^0 - X^0 - |\vec{x} - \vec{y}|)}{|\vec{x} - \vec{y}|} = 2\delta(\eta_{\mu \nu} R^\mu R^\nu) H(R^0)
    \end{equation*}
    where $H$ is the Heaviside function.
    \label{propRewriteGreens}
\end{proposition}
We can interpret this as follows:
\begin{itemize}
    \item The delta function forces $R^\mu$ to be null, and so $Y^\mu$ must be on the light cone through $X$.
    \item The Heaviside function requires $Y^0 < X^0$, and this restricts $Y^\mu$ to the past light cone.
\end{itemize}
\begin{proof}
    We think of $\delta(\eta_{\mu \nu} R^\mu R^\nu)$ as a function of $R^0$.
    \begin{align*}
        \delta(\eta_{\mu \nu} R^\mu R^\nu) &= \delta(\vec{R}^2 - (R^0)^2) \\
        &= \frac{\delta(\abs{\vec{R}} - R^0)}{2R^0} + \frac{\delta(\abs{\vec{R}} + R^0)}{-2R^0} \\
        &= \frac{1}{2\abs{\vec{R}}}\left(\delta(\abs{\vec{R}} - R^0) + \delta(\abs{\vec{R}} + R^0)\right)
    \end{align*}
    Then the Heaviside function removes the second term:
    \begin{equation*}
        2\delta(\eta_{\mu \nu} R^\mu R^\nu) H(R^0) = \frac{\delta(\abs{\vec R} - R^0)H(R^0)}{\abs{\vec R}}
    \end{equation*}
\end{proof}
\section{Dipole Radiation}
We consider radiation from a macroscopic charge distribution which is localised to a region of size $a$ and with speeds much less than $c$.

We will consider the origin inside the charge distribution, and want to calculate the fields at a point $\vec{x}$ far from the charge distribution ($\vec{x} \gg a$). $\vec{y}$ will be a point within the charge distribution (consistent with \eqnref{eqnRetardedPotential}). Some expansions are required:
\begin{align*}
    \abs{\vec{x} - \vec{y}}&= \left(\abs{\vec x}^2 - 2 \vec{x} \cdot \vec{y} + \abs{\vec y}^2\right)^{\frac12} \\
    &= r \left(1 - \frac{2 \vec{x} \cdot \vec{y}}{r^2} + \frac{|\vec{y}|^2}{r^2}\right)^{\frac12} \\
    &= r\left(1 - \frac{\vec{x} \cdot \vec{y}}{r^2} + O\left(\frac{a^2}{r^2}\right)\right) \text{ by Binomial expansion} \\
    &= r - \uvec{x} \cdot \vec{y} + O\left(\frac{a^2}{r}\right)
\end{align*}
where $\uvec{x} = \frac1r \vec{x}$.

We now substitute into \eqnref{eqnRetardedPotential}:
\begin{align*}
    A^\mu(t, \vec x) &= \frac{\mu_0}{4\pi} \int J^\mu(t_{ret}, \vec{y}) r \left(1 + \frac{\uvec{x} \cdot \vec y}{r} + O\left(\frac{a^2}{r^2}\right)\right) \\
    t_{ret} &= t - \frac r c + \frac{\uvec{x} \cdot \vec y}{c}
\end{align*}
We need to be careful with our retarded time. For a source oscillating with frequency $\omega$, we would need $\frac{a}{c} \ll \frac{1}{\omega}$ so that the change in the retarded time across the source is small compared to the period of the oscillation. We therefore need the dimensionless quantity $\frac{a\omega}{c} \ll 1$. Note that this is handled moving much slower than $c$, because $a\omega$ can be thought of as a typical speed scale. Furthermore, since the field will oscillate on the order $\omega$ for this distribution, the condition is equivalent to $a$ being much smaller than the wavelength of radiation.

Therefore, for sufficiently slow time variations of the source, we can ignore the variation of $t_{ret}$ with $\vec{y}$:
\begin{equation}
    \vec{A}(t, \vec{x}) \approx \frac{\mu_0}{4\pi r} \vec{J}(t - \frac r c, \vec{y}) d^3\vec{y}
    \label{eqnElecDipoleJ}
\end{equation}
We want this in terms of the electric dipole moment:
\begin{equation*}
    \vec p(t) = \int \vec x \rho(t, \vec{x}) d^3\vec x
\end{equation*}
Then the time derivative is:
\begin{align*}
    \dvec{p} &= \int_V \frac{\partial \rho}{\partial t} \vec{x} d^3\vec{x} \\
    \dot{p}_i &= -\int_V x_i \frac{\partial J_j}{\partial x_j} d^3 \vec x \text{ by current conservation} \\
    &= \int_{\partial V} x_i J_j dS_j + \int_V J_j \delta_{ij} d^3\vec{x} \\
    &= \int_{\partial V} x_i J_j dS_j + \int_V J_i d^3\vec{x} \\
    &= \int_V J_i d^3\vec{x}
\end{align*}
because we assume $\vec{J} \to 0$ if we take $V$ to be sufficiently large. It follows that:
\begin{equation}
    A(t, \vec{x}) \approx \frac{\mu_0}{4\pi r} \dvec{p}\left(t - \frac r c\right)
    \label{eqnElecDipole}
\end{equation}
which is the \underline{electric dipole approximation}.

For a source oscillating at frequency $\omega$,
\begin{align*}
    \vec{A}(t, \vec x) &\sim \frac{\mu_0}{4\pi r} e^{-i\omega \left(t - \frac r c\right)} \\
    &\sim \frac{\mu_0}{4\pi r} e^{i(kr-\omega t)}
\end{align*}
where the wave number $k = \frac \omega c$. This is what we expect - an outgoing spherical wave. We can calculate the magnetic field using $\vec B = \nabla \times \vec{A}$:
\begin{align*}
    \vec B&= \frac{\mu_0}{4\pi} \nabla \times \left[\frac1r \dvec{p}\left(t - \frac r c\right)\right] \\
    &= \frac{\mu_0}{4\pi} \left(\nabla \left(\frac1r\right) \times \vec{p}\left(t - \frac r c\right) - \frac1r \nabla \left(\frac r c\right) \times \ddvec{p}\left(t - \frac r c\right)\right) \\
    &= \frac{\mu_0}{4\pi} \left(\frac{\uvec{x}}{r^2} \times \vec{p}\left(t - \frac r c\right) - \frac1r \frac{\uvec{x}}{c} \times \ddvec{p}\left(t - \frac r c\right)\right) \\
    &= -\frac{\mu_0}{4\pi r^2} \uvec{x} \times \dvec{p} \left(t - \frac r c\right) - \frac{\mu_0}{4\pi r c} \uvec{x} \times \ddvec{p} \left(t - \frac r c\right)
\end{align*}
\begin{remarks}
    \item The cross products show that $\vec{B}$ is perpendicular to $\vec x$.
    \item If the charge distribution is a source oscillating at $\omega$, the first term goes as $\frac{\omega p}{r^2}$ and the second term goes as $\frac{\omega^2 p}{cr}$ which is of order $r / \lambda$ times the first term. 
\end{remarks}
We say that the area with $r \gg \lambda$ is the \underline{far field}. Therefore, in this far field,
\begin{equation*}
    \vec{B} \approx -\frac{\mu_0}{4\pi r c} \uvec{x} \times \ddvec{p}\left(t - \frac r c\right).
\end{equation*}
