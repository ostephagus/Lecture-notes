We will seek to understand the production of electromagnetic waves by oscillating charge distributions:
\begin{enumerate}
    \item macroscopic charge densities / currents involving non-relativistic motion, including scattering; and
    \item accelerated charges in arbitrary motion.
\end{enumerate}
\section{Retarded Potentials}
Recall Maxwell's equations:
\begin{align*}
    \partial_\mu F^{\mu\nu} &= -\mu_0 J^\nu \tag{\ref{eqnMaxForced}} \\
    F_{\mu\nu} &= \partial_\mu A_\nu - \partial_\nu A_\mu \tag{\ref{eqn4VecPot}}
\end{align*}
Combining the two,
\begin{equation*}
    \partial_\mu \left(\partial^\mu A^\nu - \partial^\nu A^\mu\right) = -\mu_0 J^\nu
\end{equation*}
and we can use the D'Alambertian operator $\Box = \eta^{\mu\nu} \partial_\mu \partial_\nu$:
\begin{equation}
    \Box A^\mu - \partial^\nu \partial_\mu A^\mu = -\mu_0 J^\nu
    \label{eqnDAlamBadGauge}
\end{equation}
We must choose a gauge to solve this. A convenient choice is Lorenz (not Lorentz!) gauge $\partial_\mu A^\mu = 0$, which can be found by the transformation:
\begin{equation*}
    A^\mu \mapsto A^\mu - \partial^\mu \chi, \text{ where } \Box \chi = - \partial_\nu A^\nu
\end{equation*}
In this gauge, \eqnref{eqnDAlamBadGauge} becomes:
\begin{equation}
    \Box A^\mu = -\mu_0 J^\mu
    \label{eqnDAlam}
\end{equation}
which is a sourced wave equation. We solve this using a Fourier transform in time:
\begin{equation*}
    A^\mu (t, \vec{x}) = \frac1{\sqrt{2\pi}} \int_{-\infty}^{\infty} \tilde{A}^\mu (\omega, \vec{x}) e^{-i\omega t} d\omega 
\end{equation*}
The Fourier transform of \eqnref{eqnDAlam} is:
\begin{equation*}
    (\nabla^2 + k^2) \tilde{A} = -\mu_0 \tilde{J}
\end{equation*}
where $k = \frac{\omega}{c}$. This can be solved with a Green's function $G(\vec{x};\vec{y})$ satisfying:
\begin{equation}
    (\nabla^2 + k^2) G(\vec{x};\vec{y}) = \delta(\vec{x} - \vec{y})
    \label{eqnGreens}
\end{equation}
which would give:
\begin{equation*}
    \tilde{A}(\omega, \vec{x}) = -\mu_0 \int_{\R^3} G(\vec{x}; \vec{y})\tilde{J}^\mu (\omega, \vec{x}) d^3\vec{y}.
\end{equation*}
We still need to give a boundary condition for $G$. For localised currents/charges, we want the free-space Green's function with boundary condition:
\begin{equation*}
    G(\vec{x}; \vec{y}) \to 0 \text{ as } \abs{\vec{x} - \vec{y}} \to \infty.
\end{equation*}
Note that \eqnref{eqnGreens} is homogeneous and isotropic, so $G$ may only depend on $\abs{\vec{x} - \vec{y}}$. To solve for $G$ we can take $r = \abs{\vec{x} - \vec{y}}$ and so \eqnref{eqnGreens} becomes:
\begin{equation*}
    \frac1{r^2} \frac{d}{dr} \left(r^2 \frac{dG}{dr}\right) + k^2 G = 0\qquad r \neq 0
\end{equation*}
which has solutions proportional to $\frac1r e^{ikr}$ and $\frac1r e^{-ikr}$. We take the positive exponent due to causality (see later on). The constant of proportionality is fixed by integrating \eqnref{eqnGreens} on a sphere of radius $\epsilon$ centred on $\vec{x} = 0$:
\begin{align*}
    \int_{B_\epsilon} \nabla \cdot (\nabla G) d^3\vec{x} + k^2 \int_{B_\epsilon} G~d^3\vec{x} &= 1 \\
    \int_{\partial B_\epsilon} (\nabla G) \cdot \vec{dS} + k^2 \int_{B_\epsilon} G~d^3\vec{x} &= 1 \\
    4\pi \epsilon^2 \left.\frac{dG}{dr}\right|_{\epsilon}+ k^2 \int_{B_\epsilon} G~d^3\vec{x} &= 1 \\
    %TODO: Put in dG/dr here and show orders
\end{align*}
and so taking the limit $\epsilon \to 0$ gives that the constant of proportionality must be $-\frac1{4\pi}$:
\begin{equation*}
    G(\vec{x};\vec{y}) = -\frac1{4\pi} \frac{e^{ik\abs{\vec{x} - \vec{y}}}}{\abs{\vec{x} - \vec{y}}}
\end{equation*}
and it follows that:
\begin{align}
    A^\mu (t, \vec{x}) &= \frac{\mu_0}{4\pi} \int_{-\infty}^{\infty} \frac1{\sqrt{2\pi}} e^{-i\omega t} \int_{\R^3} \frac{e^{ik\abs{\vec{x} - \vec{y}}}}{\abs{\vec{x} - \vec{y}}}\tilde{J}^\mu (\omega, \vec{y}) d^3\vec{y}~ d\omega \nonumber \\
    &= \frac{\mu_0}{4\pi} \int_{\R^3} \frac{J^\mu(t_{ret}, \vec{y})}{\abs{\vec{x} - \vec{y}}} d^3\vec{y} \label{eqnRetardedPotential}
\end{align}
where the \underline{retarded time} is:
\begin{equation*}
    t_{ret} = t - \frac1c \abs{\vec{x} - \vec{y}}
\end{equation*}
so we can think of this as the integral over a past light cone from $(t, \vec{x})$. This is why we had to discount the $e^{-ikr}$ term earlier, because otherwise we would be integrating over the future light cone which violates causality.

We want to show that \eqnref{eqnRetardedPotential} is covariant. To establish this, we rewrite as:
\begin{equation}
    A^\mu(X) = \frac{\mu_0}{4\pi} \int J^\mu(Y) \frac{\delta(Y^0 - X^0 + |\vec{x} - \vec{y}|)}{|\vec{x} - \vec{y}|} d^4Y
    \label{eqnRetardedPotentialRewrite}
\end{equation}
We need the following proposition.
\begin{proposition}
    Define $R^\mu = X^\mu - Y^\mu$. Then the fraction of \eqnref{eqnRetardedPotentialRewrite} is:
    \begin{equation*}
        \frac{\delta(Y^0 - X^0 - |\vec{x} - \vec{y}|)}{|\vec{x} - \vec{y}|} = 2\delta(\eta_{\mu \nu} R^\mu R^\nu) H(R^0)
    \end{equation*}
    where $H$ is the Heaviside function.
    \label{propRewriteGreens}
\end{proposition}
We can interpret this as follows:
\begin{itemize}
    \item The delta function forces $R^\mu$ to be null, and so $Y^\mu$ must be on the light cone through $X$.
    \item The Heaviside function requires $Y^0 < X^0$, and this restricts $Y^\mu$ to the past light cone.
\end{itemize}
\begin{proof}
    We think of $\delta(\eta_{\mu \nu} R^\mu R^\nu)$ as a function of $R^0$.
    \begin{align*}
        \delta(\eta_{\mu \nu} R^\mu R^\nu) &= \delta(\vec{R}^2 - (R^0)^2) \\
        &= \frac{\delta(\abs{\vec{R}} - R^0)}{2R^0} + \frac{\delta(\abs{\vec{R}} + R^0)}{-2R^0} \\
        &= \frac{1}{2\abs{\vec{R}}}\left(\delta(\abs{\vec{R}} - R^0) + \delta(\abs{\vec{R}} + R^0)\right)
    \end{align*}
    Then the Heaviside function removes the second term:
    \begin{equation*}
        2\delta(\eta_{\mu \nu} R^\mu R^\nu) H(R^0) = \frac{\delta(\abs{\vec R} - R^0)H(R^0)}{\abs{\vec R}}
    \end{equation*}
\end{proof}
\section{Dipole Radiation}
We consider radiation from a macroscopic charge distribution which is localised to a region of size $a$ and with speeds much less than $c$.

We will consider the origin inside the charge distribution, and want to calculate the fields at a point $\vec{x}$ far from the charge distribution ($\vec{x} \gg a$). $\vec{y}$ will be a point within the charge distribution (consistent with \eqnref{eqnRetardedPotential}).
\subsection{Dipole Approximation for the Magnetic Field}
Some expansions are required:
\begin{align*}
    \abs{\vec{x} - \vec{y}}&= \left(\abs{\vec x}^2 - 2 \vec{x} \cdot \vec{y} + \abs{\vec y}^2\right)^{\frac12} \\
    &= r \left(1 - \frac{2 \vec{x} \cdot \vec{y}}{r^2} + \frac{|\vec{y}|^2}{r^2}\right)^{\frac12} \\
    &= r\left(1 - \frac{\vec{x} \cdot \vec{y}}{r^2} + O\left(\frac{a^2}{r^2}\right)\right) \text{ by Binomial expansion} \\
    &= r - \uvec{x} \cdot \vec{y} + O\left(\frac{a^2}{r}\right)
\end{align*}
where $\uvec{x} = \frac1r \vec{x}$.

We now substitute into \eqnref{eqnRetardedPotential}:
\begin{align*}
    A^\mu(t, \vec x) &= \frac{\mu_0}{4\pi} \int J^\mu(t_{ret}, \vec{y}) r \left(1 + \frac{\uvec{x} \cdot \vec y}{r} + O\left(\frac{a^2}{r^2}\right)\right)d^3\vec y \\
    t_{ret} &= t - \frac r c + \frac{\uvec{x} \cdot \vec y}{c}
\end{align*}
We need to be careful with our retarded time. For a source oscillating with frequency $\omega$, we would need $\frac{a}{c} \ll \frac{1}{\omega}$ so that the change in the retarded time across the source is small compared to the period of the oscillation. We therefore need the dimensionless quantity $\frac{a\omega}{c} \ll 1$. Note that this is handled moving much slower than $c$, because $a\omega$ can be thought of as a typical speed scale. Furthermore, since the field will oscillate on the order $\omega$ for this distribution, the condition is equivalent to $a$ being much smaller than the wavelength of radiation.

Therefore, for sufficiently slow time variations of the source, we can ignore the variation of $t_{ret}$ with $\vec{y}$:
\begin{equation}
    \vec{A}(t, \vec{x}) \approx \int_{\R^3} \frac{\mu_0}{4\pi r} \vec{J}(t - \frac r c, \vec{y}) d^3\vec{y}
    \label{eqnElecDipoleJ}
\end{equation}
We want this in terms of the electric dipole moment:
\begin{equation*}
    \vec p(t) = \int \vec x \rho(t, \vec{x}) d^3\vec x
\end{equation*}
Then the time derivative is:
\begin{align*}
    \dvec{p} &= \int_V \frac{\partial \rho}{\partial t} \vec{x} d^3\vec{x} \\
    \dot{p}_i &= -\int_V x_i \frac{\partial J_j}{\partial x_j} d^3 \vec x \text{ by current conservation} \\
    &= \int_{\partial V} x_i J_j dS_j + \int_V J_j \delta_{ij} d^3\vec{x} \\
    &= \int_{\partial V} x_i J_j dS_j + \int_V J_i d^3\vec{x} \\
    &= \int_V J_i d^3\vec{x}
\end{align*}
because we assume $\vec{J} \to 0$ if we take $V$ to be sufficiently large. It follows that:
\begin{equation}
    A(t, \vec{x}) \approx \frac{\mu_0}{4\pi r} \dvec{p}\left(t - \frac r c\right)
    \label{eqnElecDipole}
\end{equation}
which is the \underline{electric dipole approximation}.

For a source oscillating at frequency $\omega$,
\begin{align*}
    \vec{A}(t, \vec x) &\sim \frac{\mu_0}{4\pi r} e^{-i\omega \left(t - \frac r c\right)} \\
    &\sim \frac{\mu_0}{4\pi r} e^{i(kr-\omega t)}
\end{align*}
where the wave number $k = \frac \omega c$. This is what we expect - an outgoing spherical wave. We can calculate the magnetic field using $\vec B = \nabla \times \vec{A}$:
\begin{align*}
    \vec B&= \frac{\mu_0}{4\pi} \nabla \times \left[\frac1r \dvec{p}\left(t - \frac r c\right)\right] \\
    &= \frac{\mu_0}{4\pi} \left(\nabla \left(\frac1r\right) \times \vec{p}\left(t - \frac r c\right) - \frac1r \nabla \left(\frac r c\right) \times \ddvec{p}\left(t - \frac r c\right)\right) \\
    &= \frac{\mu_0}{4\pi} \left(\frac{\uvec{x}}{r^2} \times \vec{p}\left(t - \frac r c\right) - \frac1r \frac{\uvec{x}}{c} \times \ddvec{p}\left(t - \frac r c\right)\right) \\
    &= -\frac{\mu_0}{4\pi r^2} \uvec{x} \times \dvec{p} \left(t - \frac r c\right) - \frac{\mu_0}{4\pi r c} \uvec{x} \times \ddvec{p} \left(t - \frac r c\right)
\end{align*}
\begin{remarks}
    \item The cross products show that $\vec{B}$ is perpendicular to $\vec x$.
    \item If the charge distribution is a source oscillating at $\omega$, the first term goes as $\frac{\omega p}{r^2}$ and the second term goes as $\frac{\omega^2 p}{cr}$ which is of order $r / \lambda$ times the first term. 
\end{remarks}
We say that the area with $r \gg \lambda$ is the \underline{far field}. Therefore, in this far field,
\begin{equation}
    \vec{B} \approx -\frac{\mu_0}{4\pi r c} \uvec{x} \times \ddvec{p}\left(t - \frac r c\right).
    \label{eqnMagnetDipoleB}
\end{equation}
For a set of charges $q_i$ at $\vec{x_i}(t)$, then:
\begin{equation*}
    \vec{p} = \sum_i q_i \vec{x_i}(t) \implies \ddvec{p} = \sum_i q_i \ddvec{x_i}
\end{equation*}
which tells us that it is accelerated charges that give rise to this magnetic field.
\subsection{Dipole Approximation for the Electric Field}
For $\vec{E} = -\frac{\partial \vec{A}}{\partial t} - \nabla \phi$, we need $\phi$. We have to work to higher order terms in $a / \lambda$ and $a / r$ to get time-dependent field, since if we approximate $\phi$ to first order:
\begin{equation*}
    \phi \approx \frac{\mu_0}{4\pi r} \int \rho\left(t - \frac r c, \vec{x}\right) d^3\vec{x}
\end{equation*}
which is a constant in terms of the charge in the region.

We start with $\nabla \times \vec{B} = \frac1{c^2} \frac{\partial E}{\partial t}$, which is valid for the source-free region ($r > a$). Using \eqnref{eqnMagnetDipoleB},
\begin{align*}
    \frac{\partial E}{\partial t} &= c^2 \nabla \times \left(-\frac{\mu_0}{4\pi r c}\uvec{x} \times \ddvec{p}\left(t - \frac r c\right)\right) \\
    &\approx \frac{\mu_0}{4\pi} \uvec{x} \times \left(\uvec{x} \times \dddvec{p}(t - r / c)\right) \text{ in the far field}
\end{align*}
Integrating:
\begin{equation}
    \vec{E} = \frac{\mu_0}{4\pi r} \uvec{x} \times \left(\uvec{x} \times \ddvec{p}(t - r / c)\right) = -c \uvec{x} \times \vec{B}
    \label{eqnElecDipoleE}
\end{equation}

\begin{remark}
    $\vec{E}$ and $\vec{B}$ are related in the far field as for a plane wave propagating along $\uvec{x}$. That is to say, for $r \gg \lambda$, we can neglect the curvature of the wavefronts and everything looks planar.
\end{remark}
\subsection{Power Radiated by a Charge Distribution}
\label{sscRadiatedPower}%
Considering equations~\ref{eqnMagnetDipoleB} and \ref{eqnElecDipoleE}, we see that there is a non-zero energy being radiated out from the current distribution. The Poynting vector over a sphere at radius $r$ in the far field is:
\begin{align*}
    \vec{N} &=\frac{1}{\mu_0} \vec{E} \times \vec{B} \\
    &= \frac{c}{\mu_0} \vec{B} \times (\uvec{x} \times \vec B) \\
    &= \frac{c}{\mu_0} |\vec{B}|^2 \uvec{x} \\
    &= \frac{\mu_0}{(4\pi r)^2} \abs{\uvec{x} \times \ddvec{p}} \uvec{x}
\end{align*}
which varies as $r^{-2}$. Note that $\vec{N}$ is the power per unit area, so integrating over an area would give a constant.

\begin{example}[Linearly polarised radiation]
    If $\ddvec{p}$ is along $\vec{e_z}$, and $\uvec{x}$ is at an angle $\theta$ to this, we find that $\vec{B}$ oscillates in the direction of $\vec{e_\phi}$ (adopting spherical coordinates) and $\vec{E}$ oscillates in the direction $\vec{e_\theta}$. The Poynting vector is:
    \begin{equation*}
        \vec N = \frac{\mu_0}{(4\pi r)^2} \abs{\ddvec{p}}^2 \sin^2 \theta \uvec{x}
    \end{equation*}
    which is strongest when $\theta = \frac\pi 2$ and weakest when $\uvec{x}$ is aligned with $\ddvec{p}$.

    The total radiated power is:
    \begin{align*}
        \int \vec{N} \cdot \vec{dS} &= \frac{\mu_0}{(4\pi)^2 c} \abs{\ddvec{p}}^2 \int_0^{2\pi} \int_0^\pi \sin^2\theta \sin\theta~d\theta~d\phi \\
        \int \vec{N} \cdot \vec{dS} &= \frac{\mu_0}{(4\pi)^2 c} \abs{\ddvec{p}}^2 \times \frac{8\pi}{3} \\
        &= \frac{\mu_0}{6\pi c} \abs{\ddvec{p}}^2
    \end{align*}
\end{example}
\section{Scattering Radiation}
Consider a plane electromagnetic wave incident on a stationary charge distribution. This will cause the free charges to oscillate and cause scattering of radiation from oscillating dipoles.
\subsection{Thompson Scattering}
We begin with scattering off a free electron (or free charge) with field weak enough that the motion is non-relativistic. Because $\abs{\vec E} = c\abs{\vec{B}}$ in the electromagnetic wave, we can ignore the magnetic force (because $vB \sim \frac v c E \ll E$, and so it is dominated by the electric force). We choose a frame in which the time-averaged velocity of the electron is zero, and so the equation of motion is:
\begin{equation*}
    m_e \ddvec{x} = (-e) \vec{E}(\vec{x}(t), t)
\end{equation*}
where we must evaluate the electric field at the location of the electron. This is difficult, so we evaluate the electric field at the time-averaged location of the electron (which we take to be the origin) because the amplitude of oscillation of the electron is much smaller than $\lambda$, the wavelength of the incident radiation. We can quantify these:
\begin{equation*}
    \text{amplitude} \sim \frac{e E_0}{m_e \omega^2} \ll \lambda \sim \frac{c}{\omega}
\end{equation*}
which is in fact the condition $a \ll \lambda$ as before, because the characteristic size of the charge distribution is exactly the size of the oscillations our single electron undergoes. The speed of the electron is this amplitude times $\omega$ and we require this to be much less than $c$.

The acceleration of our particle is:
\begin{align*}
    \ddvec{x} &= -\frac{e}{m_e} \vec{E} \\
    \implies \ddvec{p} &= \frac{e^2}{m_e} \vec{E}
\end{align*}
then the radiated power (from subsection~\ref{sscRadiatedPower}) is:
\begin{equation*}
    P = \frac{\mu_0}{6\pi c} \abs{\ddvec{p}}^2 \implies \tAvg{P} = \frac{\mu_0}{6\pi c} \frac{e^4 \abs{\vec{E_0}}^2}{2m_e^2}
\end{equation*}
This factor $\abs{\vec{E_0}}^2$ is the same as in the power incident on the charge distribution, so we see that the radiated power is proportional to the incident power.
\begin{definition}{Scattering cross-section}
    The \underline{scattering cross-section}, $\sigma$, is the scattered power divided by the incident energy flux.
\end{definition}

The time-averaged Poynting vector of an incident plane wave is:
\begin{equation*}
    \tAvg{\vec N} = \frac12 \frac{\vec{E_0}^2}{\mu_0 c} \uvec{k}
\end{equation*}
and so, for Thompson scattering,
\begin{equation*}
    \sigma_T = \frac{\mu_0^2 e^4}{4\pi m_0^2} = \frac{e^4}{6\pi \epsilon_0 (m_0c^2)^2}
\end{equation*}
\subsection{Rayleigh Scattering}
Consider scattering from bound charges in atoms/molecules with wavelength much larger than the size of the scattering, so the dipole approximation is valid. Molecules tend to polarise in an applied field, developing oscillating dipole moments. Typically,
\begin{equation*}
    \vec p = \alpha \vec E
\end{equation*}
where $\alpha$ is the polarisability which depends on the exact form of the scatterer.

Taking two derivatives:
\begin{equation*}
    \ddvec{p} = -\alpha\omega^2 \vec{E_0}^2 \cos(\omega t)
\end{equation*}
and scattered power is:
\begin{equation*}
    \tAvg{\frac{dE}{dt}} = \frac{\mu_0}{6\pi c} \frac{(\alpha \omega^2)^2}{2} |\vec{E_0}|^2
\end{equation*}
and so the scattering cross-section is:
\begin{equation*}
    \sigma_R = \frac{\mu_0 \alpha^2 \omega^4}{6\pi} \propto \frac{1}{\lambda^4} 
\end{equation*}
This frequency-dependent scattering is observed as the different colours of the sky when the sun is at different angles. When the sun is overhead, the blue light is more scattered and so the sky is blue. When the sun is rising or setting, the blue light is scattered away and the red and yellow light (that is not scattered as efficiently) passes through the atmosphere.
\section{Radiation from an Arbitrarily Moving Point Charge}
If a point charge is moving close to the speed of light, we will not be able to use the dipole expansion. Indeed, going to higher orders in the expansion will generate a series that converges very slowly.

This suggests a non-perturbative approach. Consider a single point charge $q$ following a worldline $Y(\tau)$.
\subsection{Lienard-Weichert Potential}
The current is:
\begin{equation*}
    J^\mu(X) = qc \int \delta(X - Y(\tau)) Y^\mu(\tau) d\tau
\end{equation*}
and the $4$-vector potential is:
\begin{equation*}
    A^\mu(x) = \frac{\mu_0}{4\pi} \int J^\mu(Z) 2\delta\left[\eta_{\alpha\beta}(X^\alpha - z^\alpha)(X^\beta - Z^\beta)\right] H(X^0 - Z^0) d^4Z
\end{equation*}
Inserting our current:
\begin{equation*}
    A^\mu(X) = \frac{2\mu_0qc}{4\pi} \int \delta[\eta_{\alpha\beta}R^\alpha(\tau)R^\beta(\tau)] H(R^0(\tau)) \frac{dY}{d\tau} d\tau
\end{equation*}
where $R^\mu(\tau) = X^\mu - Y^\mu(\tau)$. This integral selects the intersection of $Y^\mu(\tau)$ with the past light cone through $X$.

We can think of $\delta[\eta_{\alpha\beta}R^\alpha(\tau) R^\beta(\tau)] H(R^0(\tau))$ as a function of $\tau$. Note that the delta function has two roots, but only one survives the Heaviside function (only one is on the past light cone). We let this root be $\tau_*(X)$. Then:
\begin{equation*}
    \delta[\eta_{\alpha\beta}R^\alpha(\tau) R^\beta(\tau)] H(R^0(\tau)) = \frac{\delta(\tau - \tau_*(X))}{\frac{d}{d\tau} R_\alpha R^\alpha} = \frac{\delta(\tau - \tau_*(X))}{2|R^\alpha(\tau_*) \dot{Y}_\alpha(t_*)}
\end{equation*}
where we set $\dot{Y}$ to be the $\tau$-derivative. Finally, we can integrate over $\tau$ to get:
\begin{equation}
    A^\mu(X) = \frac{\mu_0 q c}{4\pi} \frac{\dot{Y}(\tau_*)}{|R^\alpha(\tau_*) \dot{Y}_\alpha(\tau_*)}
    \label{eqnLienardWeichert}
\end{equation}
which is the Lienard-Weichert potential.

Consider an inertial frame in which:
\begin{equation*}
    \dot{Y}_\alpha(\tau_*) = \gamma(\tau_*) \begin{pmatrix}-c \\ \vec{v}(\tau_*)\end{pmatrix},\qquad R^\alpha(\tau) = \begin{pmatrix}c\delta t(\tau_8) \\ \vec{R}(\tau_*)\end{pmatrix}
\end{equation*}
where $\vec{R} = \vec{x} - \vec{y}(\tau_*)$, and $c \delta t = ct - Y^0(\tau_*) = \abs{\vec{R}}$ because the vector is null, so write this as $R(\tau_*)$.

It follows that:
\begin{equation*}
    |R^\alpha(\tau_*) \dot{Y}_\alpha(\tau_*)| = c\gamma(\tau_*) R(\tau_*) \left[1 - \frac{\vec{v}(\tau_*)}{c} \uvec{R}(\tau_*)\right]
\end{equation*}
This take $R^\mu$ and projects it onto the direction of the $4$-velocity of the particle at the event on the worldline at $\tau_*$. Therefore, in the the frame in which the particle is at rest in this event, $R = c\Delta t = c\abs{\vec{x} - \vec{y}(\tau_*)} = R_{rest}(\tau_*)$.

We can calculate the scalar and vector potentials:
\begin{align*}
    \phi(t, x) &= \frac{q}{4\pi \epsilon_0 R_{rest}(\tau_*)} \frac{1}{1 - \vec{v}(\tau_*) \cdot \uvec{R} / c} \\
    \vec{A}(t, x) &= \frac{\mu_0 q}{4\pi R_{rest}(\tau_*)} \frac{\vec{v}(\tau_*)}{1 - \vec{v}(\tau_*) \cdot \uvec{R} / c}
\end{align*}
Note that $1 - \uvec{R}(\tau_*) \cdot \vec{v}(\tau_*) / c$ is small for $\uvec{R}$ parallel to $\vec{v}$ for $v \sim c$. This means that the fields are concentrated in the direction of motion.
