\section{Simple Methods for Approximating Integrals}
\subsection{Taylor Expansion of the Integrand}
We will consider Stieltjes integrals, which have the form:
\begin{equation*}
    I(x) = \int_{0}^{\infty} \frac{e^{-t}}{1 + xt} dt
\end{equation*}
for $x \geq 0$.

Our aim is to understand the behaviour of $I(x)$ for small $x$. We can Naively Taylor expand the integrand:
\begin{equation*}
    \frac{1}{1 + xt} =\sum_{n=0}^{\infty} (-1)^n x^n t^n
\end{equation*}
by unravelling a geometric series. However, our integral takes $t \to \infty$ and so this expression is not suitable. We could instead consider a finite sum:
\begin{align*}
    \sum_{n=0}^{N} (-xt)^n &= \frac{1 - (-xt)^{N+1}}{1 + xt} \\
    &= \frac{1}{1 + xt} - \underbrace{\frac{(-xt)^{N+1}}{1 + xt}}_{r_N}
\end{align*}
Substituting this into the integral:
\begin{align*}
    I(x) &= \int_{0}^{\infty} e^{-t} \left[\sum_{n=0}^{N} (-1)^n x^n t^n + r_N\right] dt \\
    &= \sum_{n=0}^{N} (-1)^n x^n \int_{0}^{\infty} t^n e^{-t} dt + R_n \\
    &= \sum_{n=0}^{N} (-1)^n x^n n! + R_N
\end{align*}
so we can define $\phi_n = x^n$, an asymptotic sequence. What remains now is to show that $R_N = o(x^N)$ as $x \to 0$. We can do this by considering a ratio:
\begin{align*}
    \frac{R_N}{x^N} &= \frac{x^{N+1}}{x^N} \int_{0}^{\infty} \frac{t^{N+1}e^{-t}}{1 + xt} dt \\
    &\leq x \int_{0}^{\infty} t^{N+1}e^{-t} dt \text{ since $1 + xt > 1$} \\
    &= x(N+1)!
\end{align*}
which tends to $0$ as $x \to 0$ because $N$ is fixed. Our final answer is:
\begin{equation*}
    I(x) \sim \sum_{n=0}^{\infty} (-1)^nn!~x^n \text{ as $n \to 0^+$.}
\end{equation*}
\subsection{Integration by Parts}
We can use the integration by parts formula:
\begin{equation*}
    \int_{a}^{b} f'(t) g(t) dt = \left[f(t)g(t)\right]_a^b - \int_{a}^{b} f(t)g'(t) dt 
\end{equation*}
many times:
\begin{align*}
    I &= \int_{a}^{b} f(t)g^{(n)}(t) dt \\
    &= \sum_{k=1}^{n} (-1)^{k-1} \left[f^{(k-1)}(t) g^{(n-k)}(t)\right]_a^b + (-1)^n \int_{a}^{b} f^{(n)}(t)g(t) dt
\end{align*}
Then if we can bound the resulting integrand by $o(\text{previous term})$, we can give an asymptotic series of boundary terms.
\begin{example}
    Consider:
    \begin{equation*}
        F(x) = \int_{x}^{\infty} e^{-\frac12t^2} dt
    \end{equation*}
    We are interested in its behaviour as $x \to \infty$.

    We rewrite the integrand as $t \times \frac{e^{-t^2 / 2}}{t}$, and perform one integration by parts:
    \begin{equation*}
        F(x) = \left[-\frac1{t} e^{-t^2 / 2}\right]_x^\infty + \int_{x}^{\infty} \frac{e^{-t^2 / 2}}{t^2} dt.
    \end{equation*}
    A one-term asymptotic expansion is therefore $\frac{e^{-x^2 / 2}}{x}$, as long as we can bound the error:
    \begin{align*}
        \int_{x}^{\infty} \frac{e^{-t^2 / 2}}{t^2} dt &= \int_{x}^{\infty} \frac{te^{-t^2 / 2}}{t^3} dt \\
        &\leq \int_{x}^{\infty} \frac{te^{-t^2 / 2}}{x^3} dt \text{ because $t > x$} \\
        &= x^{-3} \int_{x}^{\infty} te^{-t^2 / 2} dt \\
        &= x^{-3} e^{-x^2 / 2}
    \end{align*}
    Therefore the error is $o\left(e^{-x^2 / 2} / x\right)$, and our approximation is valid.

    Continuing this process, we find the following asymptotic series:
    \begin{equation*}
        F(x) \sim \frac{e^{-x^2 / 2}}{x} \sum_{n=0}^{\infty} (-1)^n \frac{(2n-1)!!}{x^{2n}}
    \end{equation*}
\end{example}
\begin{example}
    We consider a more general example:
    \begin{equation*}
        I(x) = \int_{a}^{b} f(t)e^{xt} dt 
    \end{equation*}
    as $x \to \infty$ for $b > a$, $f(b) \neq 0$ and $f$ well-behaved on $[a, b]$. Note that, as $x \to \infty$, $I(x)$ will be dominated by a small contribution near $t = b$ where the exponential term gets very large. As a very rough estimate, we can consider:
    \begin{equation*}
        I(x) \approx f(b) \int_{a}^{b} e^{xt} dt = \frac{f(b) e^{xb}}{x}
    \end{equation*}
    because $e^{xb} \gg e^{xa}$ as $x \to \infty$. A better method is to use repeated integration by parts:
    \begin{align*}
        I(x) &= \sum_{k=1}^{N} (-1)^{k-1} \left[f^{(k-1)}(\lambda) e^{xt} x^{-k}\right]_a^b + \frac{(-1)^N}{x^N} \int_{a}^{b} f^{(n)}(t) e^{xt} dt  \\
        &= e^{xb} \left[\underbrace{\sum_{k=1}^{N} (-1)^{k-1} f^{(k)}(b) x^{-k}}_{J_1} - \underbrace{e^{-x(b-a)} \sum_{k=1}^{N} (-1)^{k-1} f^{(k-1)}(a)x^{-k}}_{J_2}\right. \\
        &+ \left.\underbrace{\frac{(-1)^N}{x^N} \int_{a}^{b} f^{(N)}(t) e^{x(t-b)} dt}_{J_3}\right]
    \end{align*}
    Now $J_1$ can be our $N$-term approximation with $\phi_k = x^{-k}$. We now need to bound $J_2$ and $J_3$ by $o(x^{-N})$:
    \begin{equation*}
        \frac{J_2}{x^{-N}} = e^{-x(b-a)} \sum_{k=1}^{N}(-1)^{k-1} f^{(k-1)}(a) x^{(N-k)}
    \end{equation*}
    which goes to $0$ as $x \to \infty$ by looking at the exponential.
    Next,
    \begin{align*}
        \frac{J_3}{x^{-N}} &\leq \abs{\frac{J_3}{x^{-N}}} \\
        &\leq \int_{a}^{b} \abs{f^{(N)}(t)} e^{x(t-b)} dt \\
        &\leq \max_{a < t < b} \abs{f^{(N)}(t)} \int_{a}^{b} e^{x(t-b)} dt \\
        &\leq \max_{a < t < b} \abs{f^{(N)}(t)} \frac1x \left(1 - e^{-x(b-a)}\right)
    \end{align*}
    which tends to $0$ as $x\to\infty$ as long as $f$ is smooth. Therefore, $J_2$ and $J_3$ are bounded and so we get asymptotic series:
    \begin{equation*}
        I(x) \sim \sum_{k=1}^{\infty} (-1)^{k-1} \frac{f^{(k-1)}(b) e^{xb}}{x^k} \text{ as } x\to\infty.
    \end{equation*}
    \label{exIntParts}
\end{example}
\section{The Gamma Function}
Recall, for $\Re(z) > 0$,
\begin{equation*}
    \Gamma(z) = \int_{0}^{\infty} t^{z-1} e^{-t} dt
\end{equation*}
\begin{propositions}{
        We state the following without proof:
        \label{propsGamma}
    }
    \item $\Gamma$ is analytic; \label{propGammaAnalytic}
    \item $\Gamma(z+1) = z\Gamma(z)$ (recursion relation); \label{propGammRecursion}
    \item $\Gamma(1) = 1$ and $\Gamma(n+1) = n!$; \label{propGammaFactorial}
    \item $\int_{0}^{\infty} t^{z-1} e^{-xt} dt = \frac{\Gamma(z)}{x^z}$; \label{propGammaExtended}
    \item generalised binomial coefficients:
        \begin{equation*}
            \begin{pmatrix}z \\ k\end{pmatrix} = \frac{\Gamma(z+1)}{k! \Gamma(z-k)}
        \end{equation*} 
        here $k$ must be a positive integer. \label{propGammaBinomial}
    \item $\Gamma(z)$ can be analytically continued to the rest of $\C$ using $\gamma(z) = \frac{\gamma(z+1)}{\gamma}$ which works for all but $0$ and the negative integers. \label{propGammaContinue}
\end{propositions}
Note that \thmref{propGammaBinomial} can be used to expand:
\begin{equation*}
    (1 + t)^z = \sum_{k=0}^{\infty} \begin{pmatrix}z \\ k\end{pmatrix} t^k.
\end{equation*}
\section{Watson's Lemma}
\begin{theorem}[Watson's Lemma]
    Consider:
    \begin{equation*}
        F(x) = \int_{0}^{T} f(t)e^{-xt} dt
    \end{equation*}
    as $x \to \infty$ and $T \in (0, \infty]$. Suppose that $f(t) \sim t^\alpha \sum_{n=0}^{\infty} a_n t^{\beta n}$ as $t \to 0^+$ where $\alpha > -1$, $\beta > 0$. Also assume that either:
    \begin{enumerate}
        \item $\abs{f(t)} < Ke^{bt}$ for all $t \in [0, t]$ for some $K, b > 0$; or
        \item $\int_{0}^{T} \abs{f(t)} dt < \infty$.
    \end{enumerate}

    Then $F$ has an asymptotic expansion:
    \begin{equation*}
        F(x) \sim \sum_{n=0}^{\infty} a_n \frac{\Gamma(\alpha + \beta n + 1)}{x^{\alpha + \beta n + 1}}.
    \end{equation*}
    \label{thmWatsonLemma}
\end{theorem}
\begin{proof}
    The inclusion of $e^{-xt}$ in the integrand will zero the integrand unless $t$ is very close to $0$, especially as $x \to \infty$. We will therefore split the integrand:
    \begin{equation*}
        F(x) = \underbrace{\int_{0}^{\epsilon} f(t)e^{-xt} dt}_{F_1(x)} + \underbrace{\int_{\epsilon}^{T} f(t) e^{-xt} dt}_{F_2(x)}
    \end{equation*}
    Therefore, we want to bound $F_2$ and use the asymptotic expansion of $f$ in $F_1$. First we bound $F_2$.
    \begin{case}{$\abs{f(t)} < Ke^{bt}$}
        \begin{align*}
            \abs{F_2(x)} &\leq \int_{\epsilon}^{T} Ke^{bt} e^{-xt} dt  \\
            &= \frac{K}{x-b} e^{(b-x)\epsilon} \\
            &= O\left(\frac{e^{-\epsilon x}}{x}\right)
        \end{align*}
        as $x \to \infty$. Note that the integral converges for $x > b$.
    \end{case}
    \begin{case}{$\abs{f(t)}$ has a bounded integral}
        \begin{align*}
            \abs{F_2}&\leq \int_{\epsilon}^{t} \abs{f(t)} e^{-xt} dt \\
            &\leq e^{-\epsilon x} \int_{\epsilon}^{T} \abs{f(t)} dt \\
            &\leq e^{-\epsilon x} \int_{0}^{T} \abs{f(t)} dt \\
            &= O\left(e^{-\epsilon x}\right)
        \end{align*}
        as $x \to \infty$.
    \end{case}
    Therefore, in both cases, $F_2$ is small beyond all orders in inverse powers of $x$. We now need to show that $F_1$ has the asymptotic series we want, and so equivalently:
    \begin{equation*}
        F_1(x) = \sum_{n=0}^{N} a_n \frac{\Gamma(\alpha + \beta n + 1)}{x^{\alpha + \beta n + 1}} + o(x^{_\alpha - \beta N - 1})
        \tag{RTP}
    \end{equation*}
    Therefore, we consider (using \thmref{propGammaExtended}):
    \begin{align*}
        \abs{F_1 - \sum_{n=0}^{N} a_n \frac{\alpha + \beta n + 1}{x^{\alpha + \beta n + 1}}} &\leq \abs{\int_0^\epsilon f(x) e^{-xt} dt - \int{0}^{\infty} \alpha_n t^{\alpha + \beta n} e^{-xt}} \\
        &\leq \underbrace{\int_0^\epsilon e^{-xt} \abs{f(x) - \sum_{n=0}^{N} \alpha_n t^{\alpha + \beta n}} dt}_{R_1} \\
        &\qquad+ \underbrace{\abs{\int_t^\infty e^{-xt} \sum_{n=0}^{N} a_n t^{\alpha + \beta n}}}_{R_2} \\
    \end{align*}
    Considering the terms individually:
    \begin{align*}
        R_2 &\leq e^{-\eta x} \sum_{n=0}^{N} \abs{a_n} \int_\epsilon^\infty e^{-x(t-\epsilon)} t^{\alpha + \beta n} dt \\
        \intertext{Let $u = t - \epsilon$:}
        &\leq e^{-\eta x} \sum_{n=0}^{N} \abs{a_n} \int_0^\infty e^{-xu} (\epsilon + u)^{\alpha + \beta n} du \\
        \intertext{then note that there exist $C_n, k$ such that $(\epsilon + u)^{\alpha + \beta n} < C_n e^{uk}$:} 
        &\leq e^{-\epsilon x} \sum_{n=0}^{N} |a_n| C_n \int_{0}^{\infty} e^{u(k-x)} du \\
        &\leq e^{-\epsilon x} \sum_{n=0}^{N} \frac{C_n\abs{a_n}}{x - k} = O\left(\frac{e^{-\epsilon x}}{x}\right)
    \end{align*}
    and so we hope that we will be able to neglect this term. For $R_1$, use the asymptotic expansion for $f$:
    \begin{align*}
        f(x) - \sum_{n=0}^{N} a_n t^{\alpha + \beta n} &= o(t^{\alpha + \beta N}) \\
        &- O(t^{\alpha + \beta N + 1}).
    \end{align*}
    Therefore, by the definition of big-$O$ notation, there exists $K_3$ such that:
    \begin{align*}
        R_1 &< K_3 \int_{0}^{\epsilon} e^{-xt} t^{\alpha + \beta (N + 1)} dt \\
        &< K_3 \int_{0}^{\infty} e^{-xt} t^{\alpha + \beta (N + 1)} dt \\
        &< K_3 \frac{\Gamma(\alpha + \beta (N + 1) + 1)}{x^{\alpha + \beta (N + 1) + 1}}
    \end{align*}
    which is on the order $x^{-\alpha - \beta (N + 1) - 1}$, so:
    \begin{equation*}
        R_1 = o\left(x^{-\alpha - \beta N - 1}\right)~\forall N \in \N.
    \end{equation*}
    Now we can see that $F_2 = o(F_1)$, so $F \sim F_1$.
\end{proof}
\begin{example}
    Consider:
    \begin{equation*}
        I(x) = \int_{0}^{\infty} e^{xt} \sin(t) dt
    \end{equation*}
    Note that we already have an exact solution in the form of $(1 + x^2)^{-1}$.

    Consider $f(t) = \sin(t)$ which is exponentially bounded. We use the Taylor series:
    \begin{equation*}
        f(t) = \sum_{n=0}^{\infty} \frac{(-1)^n}{(2n+1)!}t^{2n + 1}
    \end{equation*}
    which as $\alpha = 1, \beta = 2$.

    By \thmref{thmWatsonLemma},
    \begin{align*}
        I(x) &\sim \sum_{n=0}^{\infty} (-1)^n \frac{\Gamma(2n+2)}{(2n+1)!} \frac{1}{x^{2n+2}} \\
        &= \frac{1}{x^2} \sum_{n=0}^{\infty} \frac{(-1)^n}{x^{2n}}
    \end{align*}
    and we can sum this to get the known result $\frac{1}{1 + x^2}$.
\end{example}
\begin{example}
    Consider an ODE:
    \begin{equation*}
        xy'' + (a + 1 - x) y' - ay = 0
    \end{equation*}
    where $a > 0$ is a parameter. The solution is:
    \begin{equation*}
        M(x) = \frac{e^x}{\Gamma(a)} \int_{0}^{1} e^{-xt} (1 - t)^{a-1} dt
    \end{equation*}
    and we want to find and asymptotic expansion for $x \to \infty$.

    In this case $f(t) = (1-t)^{a-1}$. It clearly has a bounded integral on $[0, 1]$. We also need an asymptotic expansion around $t = 0$, for which we can use the generalised binomial expansion:
    \begin{equation*}
        f(t) \sim \sum_{n=0}^{\infty} (-1)^n \frac{\Gamma(a)}{n!~\Gamma(a-n)} t^n
    \end{equation*}
    which has $\alpha= 0, \beta = 1$. By \thmref{thmWatsonLemma},
    \begin{align*}
        M(x) &\sim \frac{e^x}{\Gamma(a)} \sum_{n=0}^{\infty} (-1)^n \frac{\Gamma(a) \Gamma(n+1)}{\Gamma(a-n) n!} \frac{1}{x^{n+1}} \\
        &= e^x \sum_{n=0}^{\infty} (-1)^n \frac{1}{\Gamma(a-n)} \frac{1}{x^{n+1}} \\
    \end{align*}
    which has first term $\frac{e^x}{x\Gamma(a)}$. 
\end{example}
\section{Laplace Integrals and Laplace's Method}
Consider Laplace integrals, of the form:
\begin{equation*}
    I(x) = \int_{a}^{b} f(t)e^{x\phi(t)} dt 
\end{equation*}
where $b > a$ and $f$ and $\phi$ are sufficiently nice, in a way we will define later. We are interested in the behaviour as $x \to \infty$.

\begin{remarks}
    \item \Thmref{thmWatsonLemma} deals with the special case $\phi(t) = -t$.
    \item In general, we have the ``principle of localisation'', which states that the expansion of $I(x)$ as $x \to \infty$ is determined by contributions of arbitrarily small regions around the maxima of $\phi$.
\end{remarks}
We will break up our analysis into different cases for $\phi$.
\subsection{Monotonic \texorpdfstring{$\phi$}{phi}}
Suppose that $\phi$ is monotonic on $[a, b]$.

In the first case, we consider $\phi'(t) > 0$ and $f(b) \neq 0$.
Let $u = \phi(t)$ and note that we can invert this $t = \phi^{-1}(u)$, to get:
\begin{equation*}
    I(x) = \int_{\phi(a)}^{\phi(b)} \frac{f(\phi^{-1}(u))}{\phi'(\phi^{-1}(u))} e^{xu} du 
\end{equation*}
which we can solve by integration by parts (as in example~\ref{exIntParts}):
\begin{equation*}
    I \sim \frac{f(b)e^{x\phi(b)}}{x\phi'(b)} \text{ as } x\to\infty.
\end{equation*}

If instead $\phi'(t) < 0$ and $f(a) \neq 0$, we go through a similar process to find:
\begin{equation*}
    I(x) \sim \frac{f(a) e^{x\phi(a)}}{x|\phi'(a)|}.
\end{equation*}
\subsection{\texorpdfstring{$\phi$}{phi} with One Internal Maximum} \label{sscPhiMaxCase}
Suppose now that $\phi(t)$ has an internal maximum at $t = c$, such that $\phi'(c) = 0$ and $\phi''(c) < 0$. Assume $f(c) \neq 0$. We can consider this as two separate integrals:
\begin{equation*}
    I(x) = \underbrace{\int_{c - \epsilon}^{c + \epsilon} f(t)e^{x\phi(t)} dt}_{I_1} + \underbrace{\int_{\Omega_{\epsilon}} f(t) e^{x\phi(t)} dt}_{I_2}
\end{equation*}
where $\Omega_\epsilon = [a, c - \epsilon] \cup [c + \epsilon, b]$. If $\phi$ has one natural maximum, we can see that $\phi$ is monotonically increasing on $[a, c - \epsilon]$ and decreasing on $[c + \epsilon, b]$. Therefore:
\begin{equation*}
    I_2(x) = O\left(\frac1x e^{x\phi(c + \epsilon)}\right) + O\left(\frac1x e^{x\phi(c - \epsilon)}\right) = o\left(\frac1x e^{x\phi(c)}\right) \text{ as $x \to \infty$}.
\end{equation*}

For $I_1$, we need to assume that $f$ and $\phi$ have enough derivatives so that we expand with a Taylor series around $c$:
\begin{align*}
    \phi(t) &=\phi(c) + \underbrace{\phi'(c)(t - c)}_0 + \frac12 \phi''(c)(t - c)^2 + O((t - c)^3) \\
    f(t) &= f(c) + O(t - c)
\end{align*}
\begin{align*}
    I_1 &= \int_{c - \epsilon}^{c + \epsilon} \left(f(c) + O(t - c)\right)e^{x\phi(c) + \frac12x (t - c)^2 \phi''(c) + O((t - c)^3)} dt \\
    &= f(c) e^{x\phi(c)} \int_{c - \epsilon}^{c + \epsilon} e^{x\phi''(c) (t - c)^2 / 2} (1 + O(t - c)) dt
\end{align*}
In order to generate the leading order term, we will discount the higher order terms here. These would generate the other terms in the asymptotic expansion of $I(x)$ in inverse powers of $x$. Letting $s = \sqrt{-x\phi''(c)}(t-c)$, we find:
\begin{align}
    I_1 &= \frac{f(c) e^{x\phi(c)}}{\sqrt{-x\phi''(c)}} \int_{-s_\epsilon}^{s_\epsilon} e^{s^2 / 2}ds \nonumber\\
    \intertext{then taking $x \to \infty$ (with $\epsilon$ fixed):}
    I_1 &\sim \frac{f(c)e^{x\phi(c)}}{\sqrt{-x\phi''(c)}} \int_{-\infty}^{\infty} e^{-s^2 / 2} ds \nonumber\\
    &= f(c)e^{x\phi(c)}\sqrt{\frac{2\pi}{-x\phi''(c)}} \label{eqnLaplaceMethod}
\end{align}
We can justify replacing $s_\epsilon$ with $\infty$ because $\int_{\delta}^{\infty} e^{-t^2 / 2} dt \sim \frac1\delta e^{-\delta^2 / 2}$ as $\delta \to \infty$ which is sufficiently small. Now we know $I_2 \ll I_1$ as $x \to \infty$, and so $I \sim I_1$ which we have found.
\begin{example}
    Consider:
    \begin{equation*}
        I(x) = \int_{-\infty}^{\infty} e^{-xt^2} e^{at} dt
    \end{equation*}
    which we know has solution $e^{a^2 / 4x}\sqrt{\pi / x}$. To use Laplace's method, we find:
    \begin{equation*}
        f(t) = e^{at},\qquad \phi(t) = -t^2
    \end{equation*}
    and so $\phi$ has an internal maximum at $0$ with $\phi''(0) = -2$. Using \eqnref{eqnLaplaceMethod},
    \begin{equation*}
        I(x) \sim \sqrt{\frac{\pi}{x}} \text{ as } x \to\infty.
    \end{equation*}
\end{example}
\begin{example}[Bessel Functions]
    Consider:
    \begin{equation*}
        K_0(x) = \int_{0}^{\infty} e^{-t - x^2 / 4t} dt
    \end{equation*}
    To be able to use Laplace's method, we must change variables. Consider $\psi = t - x^2 / 4t$. It has maxima at:
    \begin{equation*}
        \psi' = -1 + \frac{x^2}{4t^2} = 0 \implies t_{max} = \frac{x}2.
    \end{equation*}
    This suggests rescaling the problem using $u = \frac{2t}{x}$, to give:
    \begin{equation*}
        K_0(x) = \frac{x}{2} \int_{0}^{\infty} e^{x\left(-\frac{u}{2} - \frac{1}{2u}\right)} du
    \end{equation*}
    which has a much nicer form, $\phi(u) = -\frac12 \left(u + \frac1u\right)$, and has a simple maximum at $u = 1$.
\end{example}
\subsection{More Complicated Cases}
We can also consider the case $\phi(c) = \phi(b) = \max_{[a, b]} \phi(t)$ -- $\phi$ has an internal maximum at $c$ but also achieves this at an endpoint $b$. We require $\phi'(b) \neq 0$, and $f \neq 0$ at $c$ and $b$.

We claim that the internal point dominates the integral. This is because the endpoint contribution has asymptotic expansion proportional to:
\begin{equation*}
    \frac{e^{x\phi(b)}}{x}
\end{equation*}
whereas the internal point contribution has asymptotic expansion proportional to:
\begin{equation*}
    \frac{e^{x\phi(c)}}{x}
\end{equation*}
This means the endpoint's contribution is little-$o$ of the internal point's contribution. Therefore, the endpoint can be ignored when looking for the leading order term. The endpoint may contribute to higher-order terms.

Suppose that $\phi$ has many internal local maxima $c_i$, where $\phi'(c_i) = 0$ and $\phi''(c_i) < 0$. We can say that only the global maximum (on $[a, b]$) matters, because the asymptotic expansions include $e^{x\phi(c_i)}$ and $e^{x_1} \gg e^{x_2}$ when $x_1 > x_2$. If, however, we have two global maxima ($\phi(c_i) = \phi(c_j)$) then we need to include both in the asymptotic expansion.

Suppose now that an endpoint is a global maximum: $\phi'(b) = 0, \phi''(b) < 0$, and $f(b) = 0$. We proceed as in subsection~\ref{sscPhiMaxCase}, but get the integral:
\begin{align*}
    I_1 &= \int_{b-\epsilon}^{b} f(t)e^{x\phi(t)} dt \\
    &\sim \frac{f(b)e^{x\phi(b)}}{\sqrt{x\phi''(b)}} \int_{0}^{\infty} e^{-s^2 / 2} ds \\
    &\sim \frac12 f(b)e^{x\phi(b)} \sqrt{\frac{2\pi}{-x\phi''(b)}}
\end{align*}
which is half of the result of Laplace's method in subsection~\ref{sscPhiMaxCase}.
\begin{example}
    Consider:
    \begin{equation*}
        I(x) = \int_{0}^{3\pi / 2} e^{x\sin^2(t)} dt 
    \end{equation*}
    which immediately gives $f = 1, \phi(t) = \sin^2(t)$. This has multiple global maxima, at $\frac\pi2$ and $\frac{3\pi}2$. We need to include the contribution from $\frac\pi2$ and the endpoint contribution at $\frac{3\pi}{2}$ which, as we have just seen, will need a factor $\frac12$. The result is:
    \begin{equation*}
        I \sim e^x \sqrt{\frac{\pi}{x}} + \frac12 e^x \sqrt{\frac{\pi}{x}} = \frac32 e^x \sqrt{\frac{\pi}{x}}
    \end{equation*}
\end{example}

What if now $\phi$ has an internal maximum that has $p-1$ vanishing derivatives:
\begin{equation*}
    \phi'(c) = \phi''(c) = \cdots = \phi^{(p-1)}(c) = 0,\quad\phi^{(p)}(c) < 0
\end{equation*}
We again proceed as in subsection~\ref{sscPhiMaxCase}, but Taylor expanding $\phi$ will require higher order terms, since $\phi(t + c) = \phi(c) + \frac{\phi^{(p)}(c)}{p!} (t + c)^p$.
\begin{equation*}
    I \sim \int_{c-\epsilon}^{c+\epsilon} f(c)e^{x\phi(c)} e^{x\phi^{(p)}(c)(t-c)^p / p!} dt.
\end{equation*}
We introduce $c = \left[\frac{-x\phi^{(p)}(c)}{p!}^{\frac1p}\right](t-c)^p$:
\begin{align*}
    I &\sim \frac{(p!)^\frac1p}{\left[-x\phi^{(p)}(c)\right]} f(c) e^{x\phi(c)} \int_{-\infty}^{\infty} e^{-s^p} ds \\
    &= \frac{(p!)^\frac1p}{\left[-x\phi^{(p)}(c)\right]} f(c) e^{x\phi(c)} \int_{0}^{\infty} \frac{e^{-u} u^{\frac1p - 1}}{p}~du \text{ where } u = s^p \\
    &= \frac{(p!)^\frac1p}{\left[-x\phi^{(p)}(c)\right]} f(c) e^{x\phi(c)} \times \frac1p \Gamma(1/p) \\
    &= 2f(c)e^{x\phi(c)} \left[\frac{p!}{-x\phi^{(p)}(c)}\right]^\frac1p \Gamma\left(1 + \frac1p\right)
\end{align*}
which is proportional to $x^{-\frac1p} e^{x\phi(c)}$. This tells us that convergence gets slower as the number of vanishing derivatives increases, which reflects the fact that the maximum gets flatter and so a larger region around the maximum will contribute to the integral as $x \to \infty$.
\subsection{Finding Higher-Order Terms}
To find higher-order terms in Laplace's method, we must keep the higher-order terms in the Taylor expansions of $f$ and $\phi$ around $t = c$ where $\phi$ has a global maximum.

\begin{remarks}
    \item This is a difficult process and is better tackled case-by-case for $f$ and $\phi$.
    \item If $f(c) = 0$, we need these higher-order terms anyway to find the leading order term.
\end{remarks}
\begin{example}
    Consider:
    \begin{equation*}
        J(x) = \int_{0}^{\pi} e^{x\sin(t)} dt 
    \end{equation*}
    and suppose we want an asymptotic expansion for $J(x)$ as $x \to \infty$.

    Note that $f = 1$ and $\phi = \sin(t)$, which has an internal maximum at $t = c = \frac\pi2$.

    By our previous methods,
    \begin{equation*}
        J \sim \sqrt{\frac{2\pi}{x}} e^x.
    \end{equation*}
    But to find the higher-order terms, we need the derivation from subsection~\ref{sscPhiMaxCase}:
    \begin{equation*}
        J(x) \sim \int_{\frac\pi2 - \epsilon}^{\frac\pi2 + \epsilon} e^{x\sin(t)} dt 
    \end{equation*}
    To ease the expansions, let $u = t - \frac\pi2$:
    \begin{align*}
        J(x) &\sim \int_{-\epsilon}^\epsilon e^{x\cos(u)}du \\
        &= \int_{-\epsilon}^{\epsilon} \exp\left[x - \frac{xu^2}{2} + \frac{xu^4}{4!} + \cdots\right] du \\
        &= e^x\int_{-\epsilon}^{\epsilon} e^{-xu^2 / 2} e^{xu^4 / 24 + \cdots} \\
        \intertext{Let $s = \sqrt{x} u$:}
        &= \frac{e^x}{\sqrt{x}} \int_{-\epsilon \sqrt{x}}^{\epsilon \sqrt{x}} e^{-s^2 / 2} e^{x^{-1} s^4 / 24 + \cdots} ds \\
        &\sim \frac{e^x}{\sqrt{x}} \int_{-\epsilon \sqrt{x}}^{\epsilon \sqrt{x}} e^{-s^2 / 2} \left(1 + \frac{s^4}{24x} + \text{error}\right) ds \\
        \intertext{The error will be small as $x\to\infty$:}
        &\sim \frac{e^x}{\sqrt{x}} \left[\int_{-\infty}^{\infty} e^{-s^2 / 2} ds + \frac1{24x} \int_{-\infty}^{\infty} s^4 e^{-s^2 / 2} ds + \cdots\right] \\
        &= \sqrt{\frac{2\pi}{x}} \left[1 + \frac{1}{8x} + \cdots\right]
    \end{align*}
    Then to show this is a valid asymptotic approximation we would need to show that the remaining terms are little-$o$ of the previous term.
\end{example}
\begin{example}
    Consider:
    \begin{equation*}
        I(x) = \int_{x}^{\infty} e^{-t-x^2 / t} dt 
    \end{equation*}
    and we want an asymptotic approximation as $x \to \infty$. The maximum of the exponent here is at $t = x$, so we use a co-moving variable $xu = t$ so that $\varphi(u) = -u-u^{-1}$ which has a maximum at $1$. Using this variable change, 
    \begin{equation*}
        I(x) = x \int_{1}^{\infty} e^{x\varphi(u)} du 
    \end{equation*}
    We can calculate derivatives of $\varphi$:
    \begin{align*}
        \varphi(1) &= -2 & \varphi'(1) &= 0 & \varphi''(1) &= -2 & \varphi'''(1) &= 6.
    \end{align*}
    We can now expand $I$:
    \begin{align*}
        I(x) &\sim x \int_{1}^{\infty} e^{-2x-x(u-1)^2 + x(u-1)^3 + \cdots} du \\
    \intertext{considering $s = (u-1)\sqrt{x}$:}
        &\sim \sqrt{x} e^{-2x} \int_{0}^{\epsilon \sqrt{x}} e^{-s^2} \left[1 + \frac{s^3}{\sqrt{x}} + O\left(\frac{s^4}{x}\right)\right] ds \\
        &= \sqrt{x} e^{-2s} \left[\int_{0}^{\infty} e^{-s^2} ds + \frac1{\sqrt{x}} \int_{0}^{\infty} s^3 e^{-s^2} ds + O(x^{-1})\right] \\
        &= \frac{\sqrt{x}e^{-2x}\sqrt{\pi}}{2} \left[1 + (\pi x)^{-\frac12} + O(x^{-1})\right]
    \end{align*}
    where we used:
    \begin{equation*}
        \int_{0}^{\infty} t^{2n+1} e^{-t^2 / a^2} dt = \frac{n!a^{2n+1}}{2}.
    \end{equation*}
\end{example}
\section{Oscillatory Integrals}
\subsection{Introduction and the Riemann-Lebesgue Lemma}
Consider Fourier-type integrals:
\begin{equation*}
    F(x) = \int_{a}^{b} f(t)e^{ix\phi(t)} dt
\end{equation*}
and their behaviour as $x \to \infty$. Here $f$ is referred to as the \underline{amplitude} or \underline{envelope} function. The function $x\phi(t)$ is referred to as the \underline{phase function}, and its $t$ derivative is known as the \underline{local wave number} (for problems involving space) or \underline{frequency} (for problems involving time).

Note that as $x \to \infty$, $\abs{e^{ix\phi(t)}} = 1$ but the integrand will oscillate very rapidly. This leads to ``catastrophic cancellation'' when evaluating the integral. As $x \to \infty$, the wavelength of oscillation will get very small -- much smaller than the length scale of variation of $f$ -- so locally (over one wavelength), $f$ looks mostly constant. Therefore, the integral over one wavelength looks like:
\begin{equation*}
    \int_0^\lambda f(t_\lambda) \sin\left(\frac{2\pi}{\lambda}t\right) dt 
\end{equation*}
which is zero. This is famously formalised by the Riemann-Lebesgue Lemma (for which $\phi = t$).
\begin{theorem}[Riemann-Lebesgue Lemma]
    Let $a < b$ be finite and let $f$ be a function such that $\int_{a}^{b} |f(t)| dt$ is finite. Then:
    \begin{equation*}
        F(x) = \int_{a}^{b} e^{ixt} f(t) dt \to 0 \text{ as } \abs{x} \to \infty
    \end{equation*}
    \label{thmRiemannLebes}
\end{theorem}
In this course we are interested in how fast $f$ goes to $0$ as $x \to \infty$. We first assume that $f$ is differentiable and $f'$ is integrable, and $f(a) \neq 0, f(b) \neq 0$. A first approach is to integrate by parts:
\begin{align*}
    F(x) &= \left[f(t) \frac{e^{ixt}}{ix}\right]_a^b - \frac{1}{ix} \int_{a}^{b} f'(t) e^{ixt} dt  \\
    &= \frac1{ix} \left[f(b) e^{ixb} - f(a) e^{ixa} - \int_{a}^{b} f'(t) e^{ixt} dt \right] \\
    &\sim \frac1{ix} \left[f(b) e^{ixb} - f(a) e^{ixa}\right] 
\end{align*}
by \thmref{thmRiemannLebes}. We see that the behaviour of $f$ is dominated, to leading order, by the behaviour at the endpoints. This is due to imperfect cancellation here.

Indeed, if $f \to 0$ at $x \to \pm \infty$, and we integrate over the whole real line $\R$, then $F$ is small beyond all orders in $x^{-k}$. That is, if $f$ is infinitely differentiable we can integrate by parts $N$ times to show:
\begin{equation*}
    F(x) = o(x^{-N}) ~\forall N \in \N
\end{equation*}
\subsection{Monotonic \texorpdfstring{$\phi$}{phi}}
Consider monotonic $\phi$ on $[a, b]$ such that $\phi'(t) \neq 0$ on $[a, b]$ and $f \neq 0$ at the endpoints. Consider $u = \phi(t)$, where $t = \phi^{-1}(u)$. Now $F$ is:
\begin{align*}
    F(x) &= \int_{\phi(a)}^{\phi(b)} \frac{f(t)}{\phi'(t)}e^{ixu} du \\
    &\sim \frac{1}{ix} \left[\frac{f(t)}{\phi'(t)} e^{ix\phi(t)}\right]_a^b = O(x^{-1})
\end{align*}
which requires $\left(\frac{f}{\phi'}\right)'$ to exist and be integrable.
\subsection{Stationary Phase Method}
Suppose that $\phi$ has one stationary point on $[a, b]$. That is, there exists $c \in [a, b]$ where $\phi'(c) = 0$.

Considering our local wavenumber $x \frac{\partial \phi}{\partial t}$, we see that this is finite as $x\to \infty$, unlike other points. This means we will have small region where catastrophic cancellation does not occur (or at least not as well) around $x = c$. Indeed, we find that this dominates the integral over the endpoints. As $x \to \infty$, this region will still integrate to $0$, but slower than the endpoints. This is the localisation principle for oscillatory integrals: the asymptotic expansion of $F(x)$ is controlled by points of stationary phase (or possibly the endpoints).

Based on this intuition, we split the integral:
\begin{align*}
    F(x) &= \int_{a}^{c-\epsilon} f(t)e^{ix\phi(t)} dt + \int_{c-\epsilon}^{c+\epsilon} f(t)e^{ix\phi(t)} dt + \int_{c+\epsilon}^{b} f(t)e^{ix\phi(t)} dt \\
    &= \int_{c-\epsilon}^{c+\epsilon} f(t)e^{ix\phi(t)} dt + O(x^{-1})
\end{align*}
by considering our existing expansions for the endpoints. We are therefore interested in the integral around $c$.
\begin{align*}
    \phi(t) &= \phi(c) + \phi''(c) \frac{(t-c)^2}{2} + \cdots \\
    f(t) &= f(c) + \cdots
\end{align*}
Then substituting in:
\begin{align*}
    \int_{c-\epsilon}^{c+\epsilon} f(t)e^{ix\phi(t)} dt &\approx f(c) e^{ix\phi(c)} \exp\left[ix\phi''(c) (t-c)^2 / 2\right] \\
    \intertext{defining $s = \sqrt{x|\phi''(c)| / 2}(t-c)$}:
    &\approx \frac{f(c)e^{ix\phi(c)}}{\sqrt{x|\phi''(c)| / 2}} \int_{-\frac12 \sqrt{x|\phi''(c)|}}^{\frac12 \sqrt{x|\phi''(c)|}} e^{i\sigma s^2} ds \\
    &\sim \frac{f(c)e^{ix\phi(c)}}{\sqrt{x|\phi''(c)| / 2}} \int_{-\infty}^{\infty} e^{i\sigma s^2} ds \\
    &= \frac{f(c)e^{ix\phi(c)}}{\sqrt{x|\phi''(c)| / 2}} \sqrt{\pi} e^{i \frac{\pi}{4} \sigma} \\
    &= f(c) e^{ix\phi(c)} \sqrt{\frac{2\pi}{-ix\phi''(c)}} = O(x^{-1/2})
\end{align*}
\begin{example}
    \label{exBesselStatPhase}
    Consider the Bessel function:
    \begin{equation*}
        J_0(r) = \frac{1}{2\pi} \int_{0}^{2\pi} e^{ir\sin(t)} dt 
    \end{equation*}
    which has $f = 1$, $\phi(t) = \sin(t)$, $r = x$. We have stationary points for $\sin$ at $t = \frac\pi2$ and $\frac{3\pi}2$. We evaluate their contributions separately then add them together.

    At $t = \frac\pi2$, the contribution is:
    \begin{equation*}
        \sqrt{\frac{1}{2i\pi r}} e^{ir} = \frac{1}{\sqrt{2\pi r}}e^{i(r - \frac\pi 4)}
    \end{equation*}
    and at $t = \frac{3\pi}{2}$ it is:
    \begin{equation*}
        \sqrt{\frac{1}{-2\pi i r}}e^{ir}
    \end{equation*}
    and so our leading order expansion is:
    \begin{equation*}
        J_0(r) \sim \sqrt{\frac{2}{\pi r}} \cos\left(r - \frac{\pi}{4}\right) \text{ as } r \to \infty
    \end{equation*}
    Note that this $\sim$ is not quite accurate. The Bessel function has many zeroes and so does the cosine. These zeroes do not quite line up and so the ratio of the two functions has many points that are $\infty$. We still have a good approximation here, but the notation is not quite fit for purpose.
\end{example}
\begin{remarks}
    \item All stationary points contribute to the leading term (unlike Laplace's method, where on the the global maximum contributed);
    \item the endpoints can contribute at the next order ($x^{-1}$), they are not exponentially small;
    \item we must be careful with problems in the definition of $\sim$ if the functions oscillate rapidly near zero;
    \item Laplace's method is ``better'' than the stationary phase method for the ease of finding higher-order terms and bounding terms.
\end{remarks}
\section{Method of Steepest Descent}
Consider integrals of the form:
\begin{equation*}
    I(x) = \int_C f(z) e^{x\phi(z)} dz \text{ as } x \to \infty
\end{equation*}
Now $C$ is a contour in the complex plane, and $f, \phi$ are complex-valued functions. We will require them to be analytic in a domain continaing $C$.

Such integrals appear in the definitions of special functions and in many physical applications. Also, oscillatory integrals can be transformed into this form to better calculate higher-order terms than the method of stationary phase.
\subsection{Recap of Complex Methods}
Here $z$ is a complex number and can be written as $p + iq$ for real number $p, q$. We can also split $\phi$ into real and imaginary parts: $\phi(z) = u(p, q) + iv(p, q)$. Using this decomposition:
\begin{equation*}
    I(x) = \int_C f(z) \underbrace{e^{xu(z)}}_{\substack{\text{amplitude}\\\text{part}}} \underbrace{e^{ixv(z)}}_{\substack{\text{oscillatory}\\\text{part}}} dz
\end{equation*}
\begin{proposition}
    If $\phi$ is analytic then:
    \begin{equation*}
        \frac{\partial u}{\partial p} = \frac{\partial v}{\partial q}, \qquad \frac{\partial u}{\partial q} = -\frac{\partial v}{\partial p}
    \end{equation*}
    \label{propCauchyRiemann}
\end{proposition}
If we define $\nabla u = \left(\frac{\partial u}{\partial p}, \frac{\partial u}{\partial q}\right)^T$ and similarly for $v$, \thmref{propCauchyRiemann} gives $\nabla u \cdot \nabla v = 0$. That is, the level contours of $u$ and $v$ are orthogonal. Furthermore, $\nabla v$ is tangent to curves of constant $u$ and $\nabla u$ is tangent to curves of constant $v$. On curves of constant $v$, $u$ is maximally decreasing or increasing.
\begin{theorem}[Cauchy's Theorem]
    Let $C$ be a closed contour and $f$ be a function that is analytic on and inside $C$. Then:
    \begin{equation}
        \oint_C f(z) dz = 0
        \label{eqnClosedIntZero}
    \end{equation}
    \label{thmCauchy}
\end{theorem}
\begin{remark}
    In particular,
    \begin{equation*}
        \oint_C f(z) e^{x\phi(z)} dz = 0
    \end{equation*}
    if also $\phi$ is analytic.
\end{remark}
\begin{corollary}
    If $C_1$ and $C_2$ are curves in $\C$ with the same endpoints (and same direction), then if $f$ is analytic on and between $C_1$ and $C_2$,
    \begin{equation*}
        \int_{C_1} f(z) dz = \int_{C_2} f(z) dz.
    \end{equation*}
    \label{corSameInt}
\end{corollary}
\subsection{Steepest Descent for \texorpdfstring{$u$}{u} With No Critical Points}
Suppose $\nabla u \neq 0$ on the domain.

The simplest case is when $v$ is a constant $v_0$ on $C$. This means $C$ is a level contour of $v$ and is therefore a steepest descent (or ascent) curve of $u$. We can now parameterise $c$ using $t \in [a, b]$:
\begin{align*}
    I(x) &= \int_C f(z) e^{x\phi(z)} dz \\
    &= e^{ixv_0} \int_{a}^{b} e^{xu(t)} \Re\left(f(z) \frac{\partial z}{\partial t}\right)~dt + i e^{i x v_0} \int_{a}^{b} \Im\left(f(t) \frac{\partial z}{\partial t}\right)~dt 
\end{align*}
and we can easily find an asymptotic expansion for this using, for example, \thmref{thmWatsonLemma}.

The other case of this subsection is when $v$ is not constant on $C$. The approach here is to deform $C$ and use \thmref{corSameInt} to instead evaluate the integral on a curve $C'$ on which $v$ is constant.

If $v$ is the same on the endpoints of $C$, then this may work. We just need a level contour passing through both endpoints. In general, however, $v$ is not the same on both endpoints. Even if $v$ takes the same value we may find that there is no level contour passing through both endpoints.

We resolve this issue by deforming $C$ to a contour that follows curves of constant $v$ for most of its length, with some \textit{bridging curves} on which $v$ is not constant. Then we must show that the bridging curves do not contribute to the leading-order terms of the integral.

Furthermore, as the name of the method suggests, we need to traverse the curves of constant $v$ such that $u$ is decreasing as we move away from the endpoints (decreasing as we move away from the start point of the integral, and increasing as we move towards the end point of the integral). This is due to how we want to neglect the bridging curves: choosing the largest value of $u$ at the endpoints we are given (not the endpoints of the bridging curve) means we can estimate the integrals well by considering the behaviour of the integrand at the given endpoints (recall the Principle of Localisation from Laplace's Method).
\begin{example}
    Consider:
    \begin{equation*}
        I(x) = \int_C f(z) e^{xz^2} dz
    \end{equation*}
    so $\phi(z) = z^2$.

    %TODO: Follow this example and draw the diagram. ~25:00 on recording.
    
    Then we deform $C$ such that $C_1$ follows the curve of steepest descent from $z_1$, $C_2$ follows a curve of constant $u$ to meet $C_3$ which traverses a curve of steepest ascent.
    
    Writing:
    \begin{equation*}
        \int_C f(z) e^{xz^2} dz = \underbrace{\int_{C_1} f(z) e^{xz^2} dz}_{I_1} + \underbrace{\int_{C_2} f(z) e^{xz^2} dz}_{I_2} + \underbrace{\int_{C_3} f(z) e^{xz^2} dz}_{I_3}
    \end{equation*}
    then we see that $I_1 = O\left(\frac1x e^{xu(z_1)}\right)$, $I_3 = O\left(\frac1x e^{xu(z_2)}\right)$ and $I_2 = O\left(\frac1x e^{x\tilde{u}}\right)$. Since $\tilde{u} < u(z_1)$ and $u_(z_2)$, we can neglect $I_2$ for the leading order term. Indeed, we can neglect one of $I_1$ and $I_2$ based on the relative size of $u(z_1)$ and $u(z_2)$. Alternatvely, for this example, we could take $C_2$ to $i\infty$ (but this needs $f$ to be well-behaved here).
\end{example}
\begin{example}
    Consider the Bessel function as in example~\ref{exBesselStatPhase}:
    \begin{equation*}
        J_0(r) = \frac1\pi \int_{0}^{\pi} e^{ir\cos(t)} dt
    \end{equation*}
    as $r \to \infty$. We found the first term in the expansion:
    \begin{equation*}
        J_0(r) \sim \sqrt{\frac{2}{\pi r}} \cos\left(r - \frac\pi4\right).
    \end{equation*}
    Take $s = \cos(t)$,
    \begin{equation*}
        J_0(r) = \frac1\pi \int_{-1}^{1} \frac{1}{\sqrt{1 - s^2}} e^{irs} ds
    \end{equation*}
    and interpret this as a contour integral ($s = z \in \C$) along the real axis. $\phi(z) = iz$ and so $u = -q$, $v = p$. The constant $v$ curves of $\phi$ are given by straight lines $p = \text{const}$. The curves of steepest descent for $u$ go upwards. We therefore take $C_1$ to be straight upwards from $z = -1$, $C_2$ horizontal with $\Im(z) = T$, and then $C_3$ straight downwards to $z = 1$:
    \begin{align*}
        C_1&: z = -1 + it,\quad t \in [0, T] \\
        C_2&: z = t + iT,\quad t \in [-1, 1] \\
        C_2&: z = 1 - it,\quad t \in [-T, 0] \\
    \end{align*}
    The integral along $C_1$ is then:
    \begin{align*}
        &i \int_{0}^{T} \frac{e^{ir(-1+it)}}{\sqrt{t(t+2i)}} dt \\
        &= ie^{-ir} \int_{0}^{T} \frac{e^{-rt}}{\sqrt{t(t+2i)}} dt
    \end{align*}
    and the integral along $C_3$ is:
    \begin{align*}
        &-i \int_{-T}^{0} \frac{e^{ir(1-it)}}{\sqrt{t(t+2i)}} dt \\
        &= -ie^{ir} \int_{0}^{-T} \frac{e^{-rt}}{\sqrt{t(t-2i)}} dt
    \end{align*}
    both of which can be evaluating using \thmref{thmWatsonLemma}. The bridging curve integral is:
    \begin{align*}
        &\int_{-1}^{1} \frac{e^{irt - rT}}{\sqrt{1-(t + iT)^2}} dt \\
        \text{Its absolute value} &\leq e^{-rT} \int_{-1}^{1} \abs{\frac{1}{\sqrt{1 - (t + iT)^2}}} dt = O\left(e^{-rT}\right)
    \end{align*}
    which means we can get rid of it. Combining the two remaining integrals gives:
    \begin{equation*}
        J_0 \sim \frac{2}{\pi} \Re\left[e^{ir - i \frac\pi2} \int_{0}^{T} \frac{1}{\sqrt{t(t- 2i)}} dt \right]
    \end{equation*}
    note that $(t - 2i)^{-\frac12}$ has an asymptotic expansion $\sum a_n t^n$ for constants $a_n$ that we can evaluate. \thmref{thmWatsonLemma} applies, with $\alpha = -\frac12$ and $\beta = 1$, to give:
    \begin{equation*}
        J_0(r) = \frac{2}{\pi} \Re\left[e^{ir - i\frac\pi 2} r^{-\frac12} \sum_{n=0}^{\infty} \frac{a_n \Gamma(1 / 2 + n)}{r^n}\right]
    \end{equation*}
    which we can expand:
    \begin{equation*}
        J_0 \sim \sqrt{\frac{2}{\pi r}} \left[\cos\left(r - \frac\pi4\right) + \frac{1}{8r}\sin\left(r - \frac\pi 4\right) - \frac{9}{128r^2} \cos\left(r - \frac\pi 4\right) + \cdots\right]
    \end{equation*}
\end{example}
